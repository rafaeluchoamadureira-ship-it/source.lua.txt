local function E(enumName, value)
	local enumType = Enum[enumName]
	if not enumType then return nil end
	for _, item in ipairs(enumType:GetEnumItems()) do
		if item.Value == value then return item end
	end
	return nil
end

local function SetSafe(instance, property, value)
	if value == nil then return end
	pcall(function() instance[property] = value end)
end

local function Callback(callback, ...)
	if typeof(callback) ~= "function" then return end
	local args = table.pack(...)
	task.spawn(function()
		local ok, err = pcall(function() callback(table.unpack(args, 1, args.n)) end)
		if not ok then warn("[Frank Hub] Callback:", err) end
	end)
end

function HubLibrary.Carregar(nomeDoHub)
	local self = setmetatable({}, HubLibrary)
	self._destroyed = false
	self._connections = {}
	self.Abas = {}
	self.AbaAtiva = nil
	self.ScreenGui = HubLibrary._CreateInterface()
	self.BackGround = self.ScreenGui:WaitForChild("BackGround")
	self.TemplateScrollingFrame = self.BackGround:WaitForChild("ScrollingFrame")
	self.Templates = self.TemplateScrollingFrame:WaitForChild("Templates")
	self.TabButtonsFrame = self.BackGround:WaitForChild("TabButtonsFrame")
	self.FixedButtonsFrame = nil

	for _, child in ipairs(self.BackGround:GetChildren()) do
		if child:IsA("Frame") and child.Name == "TabButtonsFrame" and child ~= self.TabButtonsFrame then
			self.FixedButtonsFrame = child
			break
		end
	end

	local menuName = self.BackGround:FindFirstChild("MenuName")
	if menuName and nomeDoHub ~= nil then menuName.Text = tostring(nomeDoHub) end

	-- O Folder Templates e oculto no funcionamento; ele existe apenas como modelo visual.
	for _, item in ipairs(self.Templates:GetChildren()) do
		if item:IsA("GuiObject") then item.Visible = false end
	end

	-- Drag da janela.
	local top = self.BackGround:FindFirstChild("BackGroundTop")
	if top then
		local dragging = false
		local startMouse
		local startPosition
		table.insert(self._connections, top.InputBegan:Connect(function(input)
			if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
				dragging = true; startMouse = input.Position; startPosition = self.BackGround.Position
			end
		end))
		table.insert(self._connections, top.InputEnded:Connect(function(input)
			if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then dragging = false end
		end))
		table.insert(self._connections, UserInputService.InputChanged:Connect(function(input)
			if self._destroyed or not dragging or not startMouse or not startPosition then return end
			if input.UserInputType ~= Enum.UserInputType.MouseMovement and input.UserInputType ~= Enum.UserInputType.Touch then return end
			local delta = input.Position - startMouse
			self.BackGround.Position = UDim2.new(startPosition.X.Scale, startPosition.X.Offset + delta.X, startPosition.Y.Scale, startPosition.Y.Offset + delta.Y)
		end))
	end

	local close = self.BackGround:FindFirstChild("Close")
	if close and close:IsA("GuiButton") then
		table.insert(self._connections, close.MouseButton1Click:Connect(function() self:Destroy() end))
	end

	-- Melhorias de experiencia sem alterar o desenho base.
	self:_EnhanceInterface()

	return self
end

--==================================================
-- MELHORIAS DA INTERFACE
--==================================================

function HubLibrary:_EnhanceInterface()
	if self._destroyed then
		return
	end

	self._featureConnections = self._featureConnections or {}

	local function connect(signal, callback)
		if not signal then
			return
		end

		local connection = signal:Connect(callback)
		table.insert(self._featureConnections, connection)
	end

	--==================================================
	-- ESCALA RESPONSIVA
	-- Mantem a interface proporcional em resolucoes menores.
	--==================================================

	local BackGround = self.BackGround

	if BackGround then
		local scale = BackGround:FindFirstChild("FrankHubScale")

		if not scale then
			scale = Instance.new("UIScale")
			scale.Name = "FrankHubScale"
			scale.Scale = 1
			scale.Parent = BackGround
		end

		local function atualizarEscala()
			if self._destroyed or not BackGround.Parent then
				return
			end

			local camera = workspace.CurrentCamera
			if not camera then
				return
			end

			local viewport = camera.ViewportSize
			local sx = viewport.X / 900
			local sy = viewport.Y / 700
			local target = math.min(sx, sy)
			target = math.clamp(target, 0.68, 1.15)

			scale.Scale = target
		end

		atualizarEscala()

		local camera = workspace.CurrentCamera
		if camera then
			connect(camera:GetPropertyChangedSignal("ViewportSize"), atualizarEscala)
		end
	end

	--==================================================
	-- ANIMACAO DE HOVER DOS BOTOES
	--==================================================

	local function aplicarHover(button)
		if not button or not button:IsA("GuiButton") then
			return
		end

		local baseColor = button.BackgroundColor3

		connect(button.MouseEnter, function()
			if self._destroyed or not button.Parent then
				return
			end

			local targetColor = Color3.new(
				math.clamp(baseColor.R + 0.06, 0, 1),
				math.clamp(baseColor.G + 0.06, 0, 1),
				math.clamp(baseColor.B + 0.06, 0, 1)
			)

			TweenService:Create(
				button,
				TweenInfo.new(0.12, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
				{BackgroundColor3 = targetColor}
			):Play()
		end)

		connect(button.MouseLeave, function()
			if self._destroyed or not button.Parent then
				return
			end

			TweenService:Create(
				button,
				TweenInfo.new(0.12, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
				{BackgroundColor3 = baseColor}
			):Play()
		end)
	end

	local function aplicarHoverRecursivo(parent)
		if not parent then
			return
		end

		for _, child in ipairs(parent:GetChildren()) do
			if child:IsA("GuiButton") then
				aplicarHover(child)
			end
		end
	end

	aplicarHoverRecursivo(self.BackGround)

	--==================================================
	-- TECLA DE ABRIR/FECHAR
	-- Padrao: RightControl
	--==================================================

	self.ToggleKey = Enum.KeyCode.RightControl

	connect(UserInputService.InputBegan, function(input, gameProcessed)
		if self._destroyed or gameProcessed then
			return
		end

		if input.KeyCode == self.ToggleKey then
			self:ToggleVisible()
		end
	end)

	--==================================================
	-- FECHAR TAMBEM COM ESC
	--==================================================

	connect(UserInputService.InputBegan, function(input)
		if self._destroyed then
			return
		end

		if input.KeyCode == Enum.KeyCode.Escape then
			self:SetVisible(false)
		end
	end)
end

function HubLibrary:SelecionarAba(aba)
	if self._destroyed or not aba then return end

			other.Btn.BackgroundColor3 = Color3.fromRGB(234, 0, 8)
			other.Btn.TextColor3 = Color3.fromRGB(255, 255, 255)
		end
	end

function HubLibrary:AddToggle(texto, callback, valorInicial)
	if self._destroyed or not self.Scroll then return nil end
	local hub = self._hub or self
	local obj = hub:_CloneTemplate("ToggleTemplate", self.Scroll)
	if not obj then return nil end
	local title = obj:FindFirstChild("TitleLabel")
	local button = obj:FindFirstChild("ToggleButton")
	if title then title.Text = tostring(texto) end
	if not button or not button:IsA("TextButton") then obj:Destroy(); return nil end
	local state = valorInicial == true
	local function visual()
		button.BackgroundColor3 = state and Color3.fromRGB(0,255,17) or Color3.fromRGB(0,0,0)
		local line = button:FindFirstChild("ButtonLine")
		if line then line.BackgroundColor3 = state and Color3.fromRGB(255,255,255) or Color3.fromRGB(100,100,100) end
	end
	local function Set(v, fire) state = v == true; visual(); if fire ~= false then Callback(callback, state) end end
	visual()
	table.insert(hub._connections, button.MouseButton1Click:Connect(function() Set(not state, true) end))
	return {Set=function(_,v) Set(v,true) end, Get=function() return state end, Object=obj, Button=button}
end

function HubLibrary:AddSlider(texto, valorMin, valorMax, valorPadrao, callback)
	if self._destroyed or not self.Scroll then return nil end
	local hub = self._hub or self
	valorMin = tonumber(valorMin) or 0; valorMax = tonumber(valorMax) or 100; valorPadrao = tonumber(valorPadrao) or valorMin
	if valorMax < valorMin then valorMin, valorMax = valorMax, valorMin end
	local value = math.clamp(valorPadrao, valorMin, valorMax)
	local obj = hub:_CloneTemplate("SliderTemplate", self.Scroll); if not obj then return nil end
	local title = obj:FindFirstChild("TitleLabel"); local track = obj:FindFirstChild("SliderTrack")
	local fill = track and track:FindFirstChild("FillButton"); local progress = track and track:FindFirstChild("Progress")
	if not track or not fill or not progress then obj:Destroy(); return nil end
	local function visual()
		local pct = (valorMax == valorMin) and 0 or math.clamp((value-valorMin)/(valorMax-valorMin),0,1)
		progress.Size = UDim2.new(pct,0,progress.Size.Y.Scale,progress.Size.Y.Offset)
		fill.Position = UDim2.new(pct,0,fill.Position.Y.Scale,fill.Position.Y.Offset)
		if title then title.Text = tostring(texto) .. " (" .. tostring(value) .. ")" end
	end
	local function setX(x) local w=track.AbsoluteSize.X; if w<=0 then return end; local pct=math.clamp((x-track.AbsolutePosition.X)/w,0,1); value=math.round(valorMin+(valorMax-valorMin)*pct); visual(); Callback(callback,value) end
	local dragging=false
	table.insert(hub._connections, track.InputBegan:Connect(function(i) if i.UserInputType==Enum.UserInputType.MouseButton1 or i.UserInputType==Enum.UserInputType.Touch then dragging=true; setX(i.Position.X) end end))
	table.insert(hub._connections, fill.InputBegan:Connect(function(i) if i.UserInputType==Enum.UserInputType.MouseButton1 or i.UserInputType==Enum.UserInputType.Touch then dragging=true; setX(i.Position.X) end end))
	table.insert(hub._connections, UserInputService.InputEnded:Connect(function(i) if i.UserInputType==Enum.UserInputType.MouseButton1 or i.UserInputType==Enum.UserInputType.Touch then dragging=false end end))
	table.insert(hub._connections, UserInputService.InputChanged:Connect(function(i) if dragging and (i.UserInputType==Enum.UserInputType.MouseMovement or i.UserInputType==Enum.UserInputType.Touch) then setX(i.Position.X) end end))
	visual()
	return {Set=function(_,v) local n=tonumber(v); if not n then return end; value=math.clamp(math.round(n),valorMin,valorMax); visual(); Callback(callback,value) end, Get=function() return value end, Object=obj, Track=track, FillButton=fill}
end

function HubLibrary:AddDropdown(texto, opcoes, callback, valorInicial)
	if self._destroyed or not self.Scroll then return nil end
	local hub=self._hub or self; opcoes=typeof(opcoes)=="table" and opcoes or {}
	local value=valorInicial; if value==nil then value=(#opcoes>0 and opcoes[1] or "Select Option...") end
	local obj=hub:_CloneTemplate("DropdownTemplate", self.Scroll); if not obj then return nil end
	obj.ClipsDescendants=false
	local title=obj:FindFirstChild("TitleLabel"); local button=obj:FindFirstChild("Frame")
	if title then title.Text=tostring(texto) end
	if not button or not button:IsA("TextButton") then obj:Destroy(); return nil end
	local selected=button:FindFirstChild("TitleLabel"); local arrow=button:FindFirstChild("TextLabel")
	local menu=Instance.new("Frame"); menu.Name="ContainerOpcoes"; menu.Size=UDim2.new(0,button.Size.X.Offset,0,math.max(1,#opcoes*30)); menu.Position=UDim2.new(0,button.Position.X.Offset,0,button.Position.Y.Offset+button.Size.Y.Offset+5); menu.BackgroundColor3=Color3.fromRGB(25,25,25); menu.BorderSizePixel=0; menu.Visible=false; menu.ZIndex=100; menu.Parent=obj
	local layout=Instance.new("UIListLayout"); layout.SortOrder=Enum.SortOrder.LayoutOrder; layout.Parent=menu
	local openState=false; local originalSize=obj.Size
	local function updateText() if selected then selected.Text=tostring(value) end; if arrow then arrow.Text=openState and "v" or "^" end end
	local function close() openState=false; menu.Visible=false; updateText(); TweenService:Create(obj,TweenInfo.new(.16),{Size=originalSize}):Play() end
	local function open() if #opcoes==0 then return end; openState=true; menu.Visible=true; updateText(); TweenService:Create(obj,TweenInfo.new(.16),{Size=UDim2.new(originalSize.X.Scale,originalSize.X.Offset,0,originalSize.Y.Offset+#opcoes*30+10)}):Play() end
	updateText()
	table.insert(hub._connections, button.MouseButton1Click:Connect(function() if openState then close() else open() end end))
	for n,opt in ipairs(opcoes) do local b=Instance.new("TextButton"); b.Name="Option"..n; b.LayoutOrder=n; b.Size=UDim2.new(1,0,0,30); b.BackgroundColor3=Color3.fromRGB(25,25,25); b.BorderSizePixel=0; b.Text=tostring(opt); b.TextColor3=Color3.fromRGB(255,255,255); b.Font=Enum.Font.SourceSans; b.TextSize=16; b.AutoButtonColor=false; b.ZIndex=101; b.Parent=menu; table.insert(hub._connections,b.MouseButton1Click:Connect(function() value=opt; close(); Callback(callback,opt) end)) end
	return {Set=function(_,v) value=v; updateText(); Callback(callback,value) end, Get=function() return value end, Open=open, Close=close, Object=obj, Button=button}
end

function HubLibrary:AddInfoCard(tituloCard, iconeId, listaDeTextos)
	if self._destroyed or not self.Scroll then return nil end
	local hub=self._hub or self; local obj=hub:_CloneTemplate("InfoCardTemplate", self.Scroll); if not obj then return nil end
	local title=obj:FindFirstChild("TitleLabel"); local icon=obj:FindFirstChild("Icon"); local list=obj:FindFirstChild("ListInformations")
	if title then title.Text=tostring(tituloCard) end
	if icon and icon:IsA("ImageLabel") and iconeId then icon.Image=tostring(iconeId) end
	listaDeTextos=typeof(listaDeTextos)=="table" and listaDeTextos or {}
	if list then local tpl=list:FindFirstChild("InformationTextTemplate"); if tpl and tpl:IsA("TextLabel") then tpl.Visible=false; for _,c in ipairs(list:GetChildren()) do if c:IsA("TextLabel") and c~=tpl then c:Destroy() end end; for n,t in ipairs(listaDeTextos) do local label=tpl:Clone(); label.Name="InformationText"..n; label.Text=tostring(t); label.Visible=true; label.LayoutOrder=n; label.Parent=list end end end
	return {Object=obj}
end

function HubLibrary:AddButton(texto, callback)
	if self._destroyed or not self.Scroll then return nil end
	local button=Instance.new("TextButton"); button.Name="ButtonTemplate"; button.Size=UDim2.new(0,503,0,50); button.BackgroundColor3=Color3.fromRGB(0,0,0); button.BackgroundTransparency=.45; button.BorderSizePixel=0; button.TextColor3=Color3.fromRGB(255,255,255); button.TextSize=18; button.Text=tostring(texto); button.AutoButtonColor=false; button.Parent=self.Scroll
	table.insert((self._hub or self)._connections, button.MouseButton1Click:Connect(function() Callback(callback) end))
	return {Object=button, Button=button}
end

function HubLibrary:Destroy()
	if self._destroyed then return end
	self._destroyed=true
	for _,c in ipairs(self._connections or {}) do pcall(function() c:Disconnect() end) end
	for _,c in ipairs(self._featureConnections or {}) do pcall(function() c:Disconnect() end) end
	for _,aba in ipairs(self.Abas or {}) do if aba then aba._destroyed=true; if aba.Scroll and aba.Scroll.Parent then aba.Scroll:Destroy() end; if aba.Btn and aba.Btn.Parent then aba.Btn:Destroy() end end end
	self.Abas={}; self.AbaAtiva=nil
	if self.ScreenGui and self.ScreenGui.Parent then self.ScreenGui:Destroy() end
	self.ScreenGui=nil; self.BackGround=nil; self.TemplateScrollingFrame=nil; self.Templates=nil; self.TabButtonsFrame=nil; self.FixedButtonsFrame=nil
end

return HubLibrary

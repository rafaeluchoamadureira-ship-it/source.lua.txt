local URL = "https://raw.githubusercontent.com/rafaeluchoamadureira-ship-it/source.lua.txt/refs/heads/main/source.lua.txt"

local FrankLibrary = loadstring(game:HttpGet(URL))()

local Janela = FrankLibrary.Carregar("Frank Hub v2")

local AbaMain = Janela:AddTab("Principal")

AbaMain:AddToggle("Auto Farm Teste", function(estado)
	print("Toggle:", estado)
end, false)

AbaMain:AddSlider("Velocidade", 16, 250, 16, function(valor)
	local player = game.Players.LocalPlayer
	local character = player.Character
	local humanoid = character and character:FindFirstChildOfClass("Humanoid")

	if humanoid then
		humanoid.WalkSpeed = valor
	end
end)

AbaMain:AddDropdown(
	"Target Areas",
	{"Vila", "Floresta", "Deserto"},
	function(valor)
		print("Selecionado:", valor)
	end
)

-- Converte InputObject.Position para as coordenadas da ScreenGui
-- (a ScreenGui usa IgnoreGuiInset = true).
local function GuiPoint(position)
	local inset = GuiService:GetGuiInset()
	return Vector2.new(position.X, position.Y + inset.Y)
end

local function TrackDrag(input, onMove, onEnd)
	local isTouch = input.UserInputType == Enum.UserInputType.Touch
	local moveConn, endConn

	local function stop()
		if moveConn then
			moveConn:Disconnect()
			moveConn = nil
		end
		if endConn then
			endConn:Disconnect()
			endConn = nil
		end
		if onEnd then
			onEnd()
		end
	end

	moveConn = UserInputService.InputChanged:Connect(function(changed)
		local isMove
		if isTouch then
			isMove = changed == input
		else
			isMove = changed.UserInputType == Enum.UserInputType.MouseMovement
		end
		if isMove then
			onMove(changed.Position)
		end
	end)

	endConn = UserInputService.InputEnded:Connect(function(ended)
		if ended == input or (not isTouch and ended.UserInputType == Enum.UserInputType.MouseButton1) then
			stop()
		end
	end)

	onMove(input.Position)
	return stop
end

local function ColorToTable(color)
	return {
		math.floor(color.R * 255 + 0.5),
		math.floor(color.G * 255 + 0.5),
		math.floor(color.B * 255 + 0.5),
	}
end

local function TableToColor(value)
	if typeof(value) == "Color3" then
		return value
	end
	if type(value) == "table" then
		local r, g, b = tonumber(value[1]), tonumber(value[2]), tonumber(value[3])
		if r and g and b then
			return Color3.fromRGB(math.clamp(r, 0, 255), math.clamp(g, 0, 255), math.clamp(b, 0, 255))
		end
	end
	return nil
end

local PROFILE_DIR = "FrankHub/Profiles"

local function EnsureFolder(path)
	if type(makefolder) ~= "function" then
		return false
	end
	local current = ""
	for part in string.gmatch(path, "[^/]+") do
		current = (current == "") and part or (current .. "/" .. part)
		local exists = false
		if type(isfolder) == "function" then
			local ok, result = pcall(isfolder, current)
			exists = ok and result == true
		end
		if not exists then
			pcall(makefolder, current)
		end
	end
	return true
end

local function ProfilePath(name)
	local safe = tostring(name):gsub("[^%w%._%-]", "_")
	return PROFILE_DIR .. "/" .. safe .. ".json"
end

	local query = string.lower(tostring(self._searchQuery or ""))
	return query:match("^%s*(.-)%s*$") or ""
end

function HubLibrary:_ApplyEntry(entry, query)
	local object = entry.Object
	if not object or not object.Parent then
		return
	end
	local matched = true
	if query ~= "" then
		if entry.Kind == "Section" or entry.Kind == "Spacer" then
			matched = false
		else
			local haystack = string.lower((entry.Title or "") .. " " .. (entry.Description or ""))
			matched = string.find(haystack, query, 1, true) ~= nil
		end
	end
	-- Secoes recolhidas escondem seus itens (a busca ignora o recolhimento).
	local collapsed = query == "" and entry.Section ~= nil and entry.Section.Collapsed == true
	entry.Visible = matched
	object.Visible = matched and not entry.ForceHidden and not collapsed
end

function HubLibrary:_ApplySearch()
	if self._destroyed then
		return
	end
	local query = self:_SearchQuery()
	for _, entry in ipairs(self._registry) do
		self:_ApplyEntry(entry, query)
	end
end

''')

	name = tostring(name or "Default")

function HubLibrary:LoadProfile(name)
	name = tostring(name or "Default")
	local snapshot = self._profiles[name]

	if not snapshot and FileApiAvailable() then
		local path = ProfilePath(name)
		local okExists, exists = pcall(isfile, path)
		if okExists and exists then
			local okRead, contents = pcall(readfile, path)
			if okRead then
				local okDecode, decoded = pcall(function()
					return HttpService:JSONDecode(contents)
				end)
				if okDecode and type(decoded) == "table" then
					snapshot = decoded
					self._profiles[name] = decoded
				end
			end
		end
	end

	if not snapshot then
		return false
	end
	self._currentProfile = name
	return self:Restore(snapshot)
end

function HubLibrary:DeleteProfile(name)
	name = tostring(name or "Default")
	if name == "Default" then
		return false
	end
	self._profiles[name] = nil
	if self._currentProfile == name then
		self._currentProfile = "Default"
	end
	if type(delfile) == "function" and type(isfile) == "function" then
		local path = ProfilePath(name)
		local ok, exists = pcall(isfile, path)
		if ok and exists then
			pcall(delfile, path)
		end
	end
	return true
end

function HubLibrary:GetProfiles()
	local found = {}
	for name in pairs(self._profiles) do
		found[name] = true
	end
	if type(listfiles) == "function" then
		local ok, files = pcall(listfiles, PROFILE_DIR)
		if ok and type(files) == "table" then
			for _, path in ipairs(files) do
				local base = tostring(path):match("([^/\\]+)%.json$")
				if base then
					found[base] = true
				end
			end
		end
	end
	local names = {}
	for name in pairs(found) do
		table.insert(names, name)
	end
	table.sort(names)
	return names
end

		if c then
			local ch, cs, cv = c:ToHSV()
			setHSV(ch, cs, cv, silent)
		end
	end
	control.Get = function()
		return currentColor()
	end
	control._serialize = function()
		return ColorToTable(currentColor())
	end
	control._deserialize = function(value, silent)
		control:Set(value, silent)
	end

	local function dragArea(hit, handler)
		hub:_Conn(hit.InputBegan, function(input)
			if input.UserInputType ~= Enum.UserInputType.MouseButton1
				and input.UserInputType ~= Enum.UserInputType.Touch then
				return
			end
			local page = self.Scroll
			page.ScrollingEnabled = false
			TrackDrag(input, function(position)
				handler(GuiPoint(position))
			end, function()
				if page.Parent then
					page.ScrollingEnabled = true
				end
			end)
		end)
	end

	dragArea(svHit, function(point)
		local pos, size = sv.AbsolutePosition, sv.AbsoluteSize
		if size.X <= 0 or size.Y <= 0 then
			return
		end
		setHSV(h, (point.X - pos.X) / size.X, 1 - (point.Y - pos.Y) / size.Y)
	end)
	dragArea(hueHit, function(point)
		local pos, size = hueBar.AbsolutePosition, hueBar.AbsoluteSize
		if size.Y <= 0 then
			return
		end
		setHSV((point.Y - pos.Y) / size.Y, s, v)
	end)

	local opened = false
	local function resize()
		hub:Tween(card, { Size = UDim2.new(1, 0, 0, opened and (PANEL_Y + PANEL_H + 12) or CARD_H) }, 0.16)
	end
	local function closeSelf()
		if opened then
			opened = false
			resize()
		end
		hub:_ClearPopup(closeSelf)
	end
	local function openSelf()
		hub:_SetPopup(closeSelf, card)
		opened = true
		resize()
	end
	control.Open = openSelf
	control.Close = closeSelf

	hub:_Conn(swatch.MouseButton1Click, function()
		if opened then
			closeSelf()
		else
			openSelf()
		end
	end)
	hub:_Conn(card.Destroying, function()
		hub:_ClearPopup(closeSelf)
	end)

	return self:_Register(control, "ColorPicker", texto, {
		Description = descricao,
		Default = ColorToTable(initial),
		StateKey = stateKey,
		TitleLabel = titleLabel,
	})
end

function Tab:AddProgressBar(texto, valorInicial, maximo, descricao)
	if self._destroyed then
		return nil
	end
	local hub = self._hub
	local max = tonumber(maximo) or 100
	if max <= 0 then
		max = 100
	end
	local value = math.clamp(tonumber(valorInicial) or 0, 0, max)

	local card = self:_NewCard(54, "ProgressBar")

	local title = New("TextLabel", {
		Name = "TitleLabel",
		BackgroundTransparency = 1,
		Position = UDim2.fromOffset(12, 6),
		Size = UDim2.new(1, -90, 0, 22),
		Text = tostring(texto),
		TextSize = 17,
		FontFace = FONT_BOLD,
		TextXAlignment = Enum.TextXAlignment.Left,
		TextTruncate = Enum.TextTruncate.AtEnd,
	}, card)
	hub:_T(title, "TextColor3", "Text")

	local percent = New("TextLabel", {
		Name = "Percent",
		AnchorPoint = Vector2.new(1, 0),
		BackgroundTransparency = 1,
		Position = UDim2.new(1, -12, 0, 6),
		Size = UDim2.fromOffset(70, 22),
		TextSize = 17,
		FontFace = FONT_BOLD,
		TextXAlignment = Enum.TextXAlignment.Right,
	}, card)
	hub:_T(percent, "TextColor3", "Accent")

	local bar = New("Frame", {
		Name = "Bar",
		Position = UDim2.fromOffset(12, 34),
		Size = UDim2.new(1, -24, 0, 10),
		BorderSizePixel = 0,
	}, card)
	Corner(bar, 5)
	local barStroke = Stroke(bar, 1.5)
	local fill = New("Frame", {
		Name = "Fill",
		Size = UDim2.new(0, 0, 1, 0),
		BorderSizePixel = 0,
	}, bar)
	Corner(fill, 5)
	hub:_Themed(bar, function(t)
		bar.BackgroundColor3 = t.Control
		barStroke.Color = t.Border
		fill.BackgroundColor3 = t.Success
	end)

	local control = { Object = card }

	local function render(animate)
		local pct = value / max
		percent.Text = tostring(math.floor(pct * 100 + 0.5)) .. "%"
		local size = UDim2.new(pct, 0, 1, 0)
		if animate then
			hub:Tween(fill, { Size = size }, 0.15)
		else
			fill.Size = size
		end
	end
	render(false)

	control.Set = function(_, newValue)
		value = math.clamp(tonumber(newValue) or value, 0, max)
		render(true)
	end
	control.Get = function()
		return value
	end
	control.SetMax = function(_, newMax)
		newMax = tonumber(newMax)
		if newMax and newMax > 0 then
			max = newMax
			value = math.clamp(value, 0, max)
			render(true)
		end
	end

	return self:_Register(control, "ProgressBar", texto, {
		Description = descricao,
		Persist = false,
		TitleLabel = title,
	})
end

function Tab:AddStatusLabel(titulo, texto, tipo, descricao)
	if self._destroyed then
		return nil
	end
	local hub = self._hub
	local card = self:_NewCard(36, "StatusLabel")
	local titleLabel = self:_Title(card, titulo, -190)

	local status = New("TextLabel", {
		Name = "Status",
		AnchorPoint = Vector2.new(1, 0.5),
		BackgroundTransparency = 1,
		Position = UDim2.new(1, -12, 0.5, 0),
		Size = UDim2.fromOffset(170, 26),
		Text = tostring(texto or ""),
		TextSize = 17,
		FontFace = FONT_BOLD,
		TextXAlignment = Enum.TextXAlignment.Right,
		TextTruncate = Enum.TextTruncate.AtEnd,
	}, card)

	local kind = tipo
	hub:_Themed(status, function(t)
		local colors = {
			Success = t.Success,
			Warning = t.Warning,
			Error = t.Danger,
			Danger = t.Danger,
			Info = t.Accent,
		}
		status.TextColor3 = colors[tostring(kind)] or t.MutedText
	end)

	local control = { Object = card }
	control.SetStatus = function(_, newText, newKind)
		status.Text = tostring(newText or "")
		if newKind ~= nil then
			kind = newKind
		end
		hub:_ApplyTheme()
	end
	control.GetStatus = function()
		return status.Text
	end

	return self:_Register(control, "StatusLabel", titulo, {
		Description = descricao,
		Persist = false,
		TitleLabel = titleLabel,
	})
end

--==================================================
-- AUTO-TESTE
--==================================================

function HubLibrary:SelfTest()
	if self._destroyed then
		return false, {}
	end

	local results = {}
	local function check(name, condition)
		table.insert(results, { Name = name, Passed = condition == true })
	end
	local function try(name, fn)
		local ok, err = pcall(fn)
		if not ok then
			check(name .. " (erro: " .. tostring(err) .. ")", false)
		end
	end

	local tab = self:AddTab("__SelfTest")

	try("toggle", function()
		local c = tab:AddToggle("T", nil, false)
		c:Set(true, true)
		check("toggle Set/Get", c:Get() == true)
		c:SetTitle("T2")
		check("toggle SetTitle", c:GetTitle() == "T2")
	end)

	try("slider", function()
		local c = tab:AddSlider("S", 0, 100, 10, nil, nil, nil, 5)
		c:Set(37, true)
		check("slider passo", c:Get() == 35)
	end)

	try("dropdown", function()
		local c = tab:AddDropdown("D", { "A", "B", "C" }, nil, "A")
		c:Set("B", true)
		check("dropdown Set/Get", c:Get() == "B")
		c:SetOptions({ "X", "Y" })
		check("dropdown SetOptions", c:Get() == "X")
	end)

	try("dropdown multi", function()
		local c = tab:AddDropdown("M", { "A", "B", "C" }, nil, { "A" }, nil, nil, true)
		c:Set({ "B", "C" }, true)
		check("multi Set/Get", #c:Get() == 2)
	end)

	try("textbox", function()
		local c = tab:AddTextBox("N", nil, 1, nil, nil, nil, true)
		c:Set(5, true)
		c:Set("abc", true)
		check("textbox numerico", c:Get() == 5)
	end)

	try("colorpicker", function()
		local c = tab:AddColorPicker("C", nil, Color3.fromRGB(255, 0, 0))
		c:Set(Color3.fromRGB(0, 255, 0), true)
		check("colorpicker Set/Get", c:Get().G > 0.9)
	end)

	try("keybind", function()
		local c = tab:AddKeybind("K", Enum.KeyCode.F5)
		c:Set(Enum.KeyCode.F6, true)
		check("keybind Set/Get", c:Get() == Enum.KeyCode.F6)
	end)

	try("progress", function()
		local c = tab:AddProgressBar("P", 10, 100)
		c:Set(50)
		check("progress Set/Get", c:Get() == 50)
	end)

	try("section", function()
		local sec = tab:AddSection("Sec", nil, true)
		sec:SetCollapsed(true)
		local item = tab:AddLabel("L")
		check("section recolhida esconde item", item.Object.Visible == false)
		sec:SetCollapsed(false)
		check("section aberta mostra item", item.Object.Visible == true)
	end)

	try("config", function()
		local snapshot = self:Snapshot()
		local ok = pcall(function()
			return HttpService:JSONEncode(snapshot)
		end)
		check("snapshot -> JSON", ok)
		check("restore", self:Restore(snapshot, true))
	end)

	tab:Destroy()

	local passed = 0
	for _, result in ipairs(results) do
		if result.Passed then
			passed = passed + 1
		else
			warn("[Frank Hub] SelfTest FALHOU:", result.Name)
		end
	end
	print(string.format("[Frank Hub] SelfTest: %d/%d ok", passed, #results))
	return passed == #results, results
end

return HubLibrary

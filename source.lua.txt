--==================================================
-- CONFIGURAÇÕES
--==================================================

local COR_FUNDO = Color3.fromRGB(25, 25, 25)
local COR_TOPO = Color3.fromRGB(215, 20, 20)
local COR_BORDA = Color3.fromRGB(0, 0, 0)

local COR_CARD = Color3.fromRGB(35, 35, 35)
local COR_CONTROLE = Color3.fromRGB(20, 20, 20)

local COR_VERDE = Color3.fromRGB(0, 235, 60)

local TAMANHO_BORDA = 3

--==================================================
-- FUNÇÕES AUXILIARES
--==================================================

local function AplicarBorda(parent, espessura)
	local stroke = Instance.new("UIStroke")
	stroke.Thickness = espessura or TAMANHO_BORDA
	stroke.Color = COR_BORDA
	stroke.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
	stroke.Parent = parent

	return stroke
end

local function Arredondar(parent, raio)
	local corner = Instance.new("UICorner")
	corner.CornerRadius = UDim.new(0, raio or 6)
	corner.Parent = parent

	return corner
end

local function ExecutarCallback(callback, ...)
	if typeof(callback) ~= "function" then
		return
	end

	local args = table.pack(...)

	task.spawn(function()
		local sucesso, erro = pcall(function()
			callback(table.unpack(args, 1, args.n))
		end)

		if not sucesso then
			warn("[HubLibrary] Erro no callback:", erro)
		end
	end)
end

local function NormalizarImagem(id)
	if id == nil then
		return "rbxassetid://0"
	end

	local texto = tostring(id)

	if texto == "" then
		return "rbxassetid://0"
	end

	if string.find(texto, "rbxassetid://", 1, true) then
		return texto
	end

	if string.find(texto, "rbxassetid.com", 1, true) then
		return texto
	end

	return "rbxassetid://" .. texto
end

--==================================================
-- CARREGAR HUB
--==================================================

function HubLibrary.Carregar(nomeDoHub)
	local self = setmetatable({}, HubLibrary)

	self.Abas = {}
	self.AbaAtiva = nil
	self._destroyed = false

	--==================================================
	-- SCREEN GUI
	--==================================================

	local PlayerGui = LocalPlayer:WaitForChild("PlayerGui")

	-- Remove uma versão antiga do mesmo menu
	local GuiAntiga = PlayerGui:FindFirstChild("Menu Hub")

	if GuiAntiga then
		GuiAntiga:Destroy()
	end

	local ScreenGui = Instance.new("ScreenGui")

	ScreenGui.Name = "Menu Hub"
	ScreenGui.ResetOnSpawn = false
	ScreenGui.ZIndexBehavior = Enum.ZIndexBehavior.Sibling

	ScreenGui.Parent = PlayerGui

	self.ScreenGui = ScreenGui

	--==================================================
	-- BACKGROUND
	--==================================================

	local BackGround = Instance.new("Frame")

	BackGround.Name = "Background"

	BackGround.Size = UDim2.new(0, 550, 0, 420)
	BackGround.Position = UDim2.new(0.5, -275, 0.5, -210)

	BackGround.BackgroundColor3 = COR_FUNDO
	BackGround.BorderSizePixel = 0

	BackGround.Parent = ScreenGui

	AplicarBorda(BackGround)
	Arredondar(BackGround, 8)

	self.BackGround = BackGround

	--==================================================
	-- TOPO
	--==================================================

	local BackGroundTop = Instance.new("Frame")

	BackGroundTop.Name = "BackGroundTop"

	BackGroundTop.Size = UDim2.new(1, 0, 0, 45)

	BackGroundTop.BackgroundColor3 = COR_TOPO
	BackGroundTop.BorderSizePixel = 0

	BackGroundTop.Parent = BackGround

	AplicarBorda(BackGroundTop)
	Arredondar(BackGroundTop, 8)

	--==================================================
	-- NOME
	--==================================================

	local MenuName = Instance.new("TextLabel")

	MenuName.Name = "MenuName"

	MenuName.Size = UDim2.new(0.7, 0, 1, 0)
	MenuName.Position = UDim2.new(0, 15, 0, 0)

	MenuName.BackgroundTransparency = 1

	MenuName.Text = nomeDoHub or "Frank Hub"

	MenuName.TextColor3 = Color3.fromRGB(255, 255, 255)

	MenuName.Font = Enum.Font.SourceSansItalic
	MenuName.TextSize = 26

	MenuName.TextXAlignment = Enum.TextXAlignment.Left

	MenuName.Parent = BackGroundTop

	local textStroke = Instance.new("UIStroke")

	textStroke.Thickness = 2
	textStroke.Color = COR_BORDA

	textStroke.Parent = MenuName

	--==================================================
	-- BOTÃO FECHAR
	--==================================================

	local Close = Instance.new("TextButton")

	Close.Name = "Close"

	Close.Size = UDim2.new(0, 35, 0, 35)
	Close.Position = UDim2.new(1, -45, 0.5, -17.5)

	Close.BackgroundColor3 = Color3.fromRGB(255, 70, 70)

	Close.Text = "X"

	Close.Font = Enum.Font.SourceSansBold
	Close.TextSize = 22

	Close.TextColor3 = Color3.fromRGB(255, 255, 255)

	Close.Parent = BackGroundTop

	AplicarBorda(Close, 2)
	Arredondar(Close, 6)

	Close.MouseButton1Click:Connect(function()
		if self._destroyed then
			return
		end

		self:Destroy()
	end)

	--==================================================
	-- ARRASTAR JANELA
	--==================================================

	local arrastando = false
	local inicioMouse = nil
	local inicioPosicao = nil

	BackGroundTop.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then

			arrastando = true

			inicioMouse = input.Position
			inicioPosicao = BackGround.Position
		end
	end)

	BackGroundTop.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then

			arrastando = false
		end
	end)

	UserInputService.InputChanged:Connect(function(input)
		if not arrastando then
			return
		end

		if input.UserInputType ~= Enum.UserInputType.MouseMovement
			and input.UserInputType ~= Enum.UserInputType.Touch then
			return
		end

		if not inicioMouse or not inicioPosicao then
			return
		end

		local delta = input.Position - inicioMouse

		BackGround.Position = UDim2.new(
			inicioPosicao.X.Scale,
			inicioPosicao.X.Offset + delta.X,
			inicioPosicao.Y.Scale,
			inicioPosicao.Y.Offset + delta.Y
		)
	end)

	--==================================================
	-- ABAS
	--==================================================

	local TabButtonsFrame = Instance.new("Frame")

	TabButtonsFrame.Name = "TabButtonsFrame"

	TabButtonsFrame.Size = UDim2.new(0, 110, 1, -65)
	TabButtonsFrame.Position = UDim2.new(0, 10, 0, 55)

	TabButtonsFrame.BackgroundTransparency = 1

	TabButtonsFrame.Parent = BackGround

	local tabLayout = Instance.new("UIListLayout")

	tabLayout.Padding = UDim.new(0, 8)
	tabLayout.SortOrder = Enum.SortOrder.LayoutOrder

	tabLayout.Parent = TabButtonsFrame

	self.TabButtonsFrame = TabButtonsFrame

	return self
end

--==================================================
-- ADD TAB
--==================================================

function HubLibrary:AddTab(nomeDaAba)
	if self._destroyed then
		return nil
	end

	if not self.TabButtonsFrame or not self.BackGround then
		return nil
	end

	local abaSelf = {}

	--==================================================
	-- BOTÃO DA ABA
	--==================================================

	local TabButton = Instance.new("TextButton")

	TabButton.Name = tostring(nomeDaAba) .. "Button"

	TabButton.Size = UDim2.new(1, 0, 0, 35)

	TabButton.BackgroundColor3 = Color3.fromRGB(40, 40, 40)

	TabButton.Text = tostring(nomeDaAba)

	TabButton.TextColor3 = Color3.fromRGB(200, 200, 200)

	TabButton.Font = Enum.Font.SourceSansBold
	TabButton.TextSize = 16

	TabButton.Parent = self.TabButtonsFrame

	AplicarBorda(TabButton, 2)
	Arredondar(TabButton, 5)

	--==================================================
	-- SCROLL DA ABA
	--==================================================

	local ScrollingFrame = Instance.new("ScrollingFrame")

	ScrollingFrame.Name = tostring(nomeDaAba) .. "Scroll"

	ScrollingFrame.Size = UDim2.new(1, -145, 1, -65)

	ScrollingFrame.Position = UDim2.new(0, 130, 0, 55)

	ScrollingFrame.BackgroundTransparency = 1

	ScrollingFrame.BorderSizePixel = 0

	ScrollingFrame.ScrollBarThickness = 6

	ScrollingFrame.ScrollBarImageColor3 = COR_TOPO

	ScrollingFrame.CanvasSize = UDim2.new(0, 0, 0, 0)

	ScrollingFrame.AutomaticCanvasSize = Enum.AutomaticSize.None

	ScrollingFrame.Visible = false

	ScrollingFrame.Parent = self.BackGround

	local scrollLayout = Instance.new("UIListLayout")

	scrollLayout.Padding = UDim.new(0, 10)

	scrollLayout.SortOrder = Enum.SortOrder.LayoutOrder

	scrollLayout.Parent = ScrollingFrame

	scrollLayout:GetPropertyChangedSignal("AbsoluteContentSize"):Connect(function()
		if ScrollingFrame.Parent then
			ScrollingFrame.CanvasSize = UDim2.new(
				0,
				0,
				0,
				scrollLayout.AbsoluteContentSize.Y + 20
			)
		end
	end)

	local scrollPadding = Instance.new("UIPadding")

	scrollPadding.PaddingTop = UDim.new(0, 5)
	scrollPadding.PaddingBottom = UDim.new(0, 10)
	scrollPadding.PaddingLeft = UDim.new(0, 5)
	scrollPadding.PaddingRight = UDim.new(0, 5)

	scrollPadding.Parent = ScrollingFrame

	--==================================================
	-- SELECIONAR ABA
	--==================================================

	local function SelecionarAba()
		if self._destroyed then
			return
		end

		for _, aba in ipairs(self.Abas) do
			if aba.Scroll and aba.Scroll.Parent then
				aba.Scroll.Visible = false
			end

			if aba.Btn and aba.Btn.Parent then
				aba.Btn.BackgroundColor3 = Color3.fromRGB(40, 40, 40)
				aba.Btn.TextColor3 = Color3.fromRGB(200, 200, 200)
			end
		end

		ScrollingFrame.Visible = true

		TabButton.BackgroundColor3 = COR_TOPO
		TabButton.TextColor3 = Color3.fromRGB(255, 255, 255)

		self.AbaAtiva = abaSelf
	end

	TabButton.MouseButton1Click:Connect(SelecionarAba)

	--==================================================
	-- REGISTRAR ABA
	--==================================================

	table.insert(self.Abas, {
		Scroll = ScrollingFrame,
		Btn = TabButton,
		Object = abaSelf,
	})

	--==================================================
	-- REFERÊNCIAS DA ABA
	--==================================================

	abaSelf.ScrollingFrame = ScrollingFrame
	abaSelf.Scroll = ScrollingFrame
	abaSelf.Btn = TabButton
	abaSelf._hub = self
	abaSelf._destroyed = false

	--==================================================
	-- PRIMEIRA ABA
	--==================================================

	if #self.Abas == 1 then
		SelecionarAba()
	end

	--==================================================
	-- RETORNO
	--==================================================

	return setmetatable(
		abaSelf,
		{
			__index = HubLibrary
		}
	)
end

--==================================================

function HubLibrary:AddToggle(texto, callback, valorInicial)
	if self._destroyed or not self.ScrollingFrame then
		return nil
	end

	local ativo = valorInicial == true

	local ToggleTemplate = Instance.new("Frame")

	ToggleTemplate.Name = "ToggleTemplate"

	ToggleTemplate.Size = UDim2.new(1, -10, 0, 45)

	ToggleTemplate.BackgroundColor3 = COR_CARD

	ToggleTemplate.BorderSizePixel = 0

	ToggleTemplate.Parent = self.ScrollingFrame

	AplicarBorda(ToggleTemplate)
	Arredondar(ToggleTemplate, 6)

	--==================================================
	-- TEXTO
	--==================================================

	local TitleLabel = Instance.new("TextLabel")

	TitleLabel.Name = "TitleLabel"

	TitleLabel.Size = UDim2.new(0.7, 0, 1, 0)

	TitleLabel.Position = UDim2.new(0, 12, 0, 0)

	TitleLabel.BackgroundTransparency = 1

	TitleLabel.Text = tostring(texto)

	TitleLabel.TextColor3 = Color3.fromRGB(255, 255, 255)

	TitleLabel.Font = Enum.Font.SourceSansBold
	TitleLabel.TextSize = 20

	TitleLabel.TextXAlignment = Enum.TextXAlignment.Left

	TitleLabel.Parent = ToggleTemplate

	--==================================================
	-- BOTÃO
	--==================================================

	local ToggleButton = Instance.new("TextButton")

	ToggleButton.Name = "ToggleButton"

	ToggleButton.Size = UDim2.new(0, 26, 0, 26)

	ToggleButton.Position = UDim2.new(1, -38, 0.5, -13)

	ToggleButton.BackgroundColor3 = ativo
		and COR_VERDE
		or COR_CONTROLE

	ToggleButton.Text = ""

	ToggleButton.Parent = ToggleTemplate

	AplicarBorda(ToggleButton, 2)
	Arredondar(ToggleButton, 6)

	--==================================================
	-- SET
	--==================================================

	local function Set(valor, dispararCallback)
		ativo = valor == true

		local corAlvo = ativo
			and COR_VERDE
			or COR_CONTROLE

		TweenService:Create(
			ToggleButton,
			TweenInfo.new(0.12),
			{
				BackgroundColor3 = corAlvo
			}
		):Play()

		if dispararCallback ~= false then
			ExecutarCallback(callback, ativo)
		end
	end

	ToggleButton.MouseButton1Click:Connect(function()
		Set(not ativo, true)
	end)

	--==================================================
	-- OBJETO RETORNADO
	--==================================================

	local controle = {
		Set = function(_, valor)
			Set(valor, true)
		end,

		Get = function()
			return ativo
		end,

		Object = ToggleTemplate,
		Button = ToggleButton,
	}

	return controle
end

--==================================================
-- SLIDER
--==================================================

function HubLibrary:AddSlider(
	texto,
	valorMin,
	valorMax,
	valorPadrao,
	callback
)
	if self._destroyed or not self.ScrollingFrame then
		return nil
	end

	valorMin = tonumber(valorMin) or 0
	valorMax = tonumber(valorMax) or 100
	valorPadrao = tonumber(valorPadrao) or valorMin

	if valorMax < valorMin then
		valorMin, valorMax = valorMax, valorMin
	end

	valorPadrao = math.clamp(valorPadrao, valorMin, valorMax)

	local valorAtual = valorPadrao

	--==================================================
	-- CONTAINER
	--==================================================

	local SliderTemplate = Instance.new("Frame")

	SliderTemplate.Name = "SliderTemplate"

	SliderTemplate.Size = UDim2.new(1, -10, 0, 60)

	SliderTemplate.BackgroundColor3 = COR_CARD

	SliderTemplate.BorderSizePixel = 0

	SliderTemplate.Parent = self.ScrollingFrame

	AplicarBorda(SliderTemplate)
	Arredondar(SliderTemplate, 6)

	--==================================================
	-- TEXTO
	--==================================================

	local TitleLabel = Instance.new("TextLabel")

	TitleLabel.Name = "TitleLabel"

	TitleLabel.Size = UDim2.new(1, -20, 0, 25)

	TitleLabel.Position = UDim2.new(0, 12, 0, 3)

	TitleLabel.BackgroundTransparency = 1

	TitleLabel.TextColor3 = Color3.fromRGB(255, 255, 255)

	TitleLabel.Font = Enum.Font.SourceSansBold

	TitleLabel.TextSize = 18

	TitleLabel.TextXAlignment = Enum.TextXAlignment.Left

	TitleLabel.Parent = SliderTemplate

	--==================================================
	-- TRACK
	--==================================================

	local SliderTrack = Instance.new("Frame")

	SliderTrack.Name = "SliderTrack"

	SliderTrack.Size = UDim2.new(1, -30, 0, 12)

	SliderTrack.Position = UDim2.new(0, 15, 0, 38)

	SliderTrack.BackgroundColor3 = Color3.fromRGB(15, 15, 15)

	SliderTrack.BorderSizePixel = 0

	SliderTrack.Parent = SliderTemplate

	AplicarBorda(SliderTrack, 2)

	Arredondar(SliderTrack, 6)

	--==================================================
	-- PROGRESSO
	--==================================================

	local Progress = Instance.new("Frame")

	Progress.Name = "Progress"

	Progress.Size = UDim2.new(0, 0, 1, 0)

	Progress.BackgroundColor3 = COR_VERDE

	Progress.BorderSizePixel = 0

	Progress.Parent = SliderTrack

	Arredondar(Progress, 6)

	--==================================================
	-- BOTÃO
	--==================================================

	local FillButton = Instance.new("TextButton")

	FillButton.Name = "FillButton"

	FillButton.AnchorPoint = Vector2.new(0.5, 0.5)

	FillButton.Size = UDim2.new(0, 16, 0, 20)

	FillButton.Position = UDim2.new(0, 0, 0.5, 0)

	FillButton.BackgroundColor3 = Color3.fromRGB(255, 255, 255)

	FillButton.Text = ""

	FillButton.AutoButtonColor = false

	FillButton.Parent = SliderTrack

	AplicarBorda(FillButton, 2)

	Arredondar(FillButton, 8)

	--==================================================
	-- ATUALIZAR VALOR
	--==================================================

	local function AtualizarVisual()
		local percentual = 0

		if valorMax ~= valorMin then
			percentual = (valorAtual - valorMin) / (valorMax - valorMin)
		end

		percentual = math.clamp(percentual, 0, 1)

		Progress.Size = UDim2.new(
			percentual,
			0,
			1,
			0
		)

		FillButton.Position = UDim2.new(
			percentual,
			0,
			0.5,
			0
		)

		TitleLabel.Text = tostring(texto) .. " (" .. tostring(valorAtual) .. ")"
	end

	--==================================================
	-- ALTERAR PELO MOUSE
	--==================================================

	local function AtualizarPorPosicao(posicaoX)
		local largura = SliderTrack.AbsoluteSize.X

		if largura <= 0 then
			return
		end

		local percentual =
			(posicaoX - SliderTrack.AbsolutePosition.X)
			/ largura

		percentual = math.clamp(percentual, 0, 1)

		valorAtual = math.round(
			valorMin
				+ ((valorMax - valorMin) * percentual)
		)

		AtualizarVisual()

		ExecutarCallback(callback, valorAtual)
	end

	local arrastando = false

	SliderTrack.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then

			arrastando = true

			AtualizarPorPosicao(input.Position.X)
		end
	end)

	FillButton.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then

			arrastando = true

			AtualizarPorPosicao(input.Position.X)
		end
	end)

	UserInputService.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then

			arrastando = false
		end
	end)

	UserInputService.InputChanged:Connect(function(input)
		if not arrastando then
			return
		end

		if input.UserInputType == Enum.UserInputType.MouseMovement
			or input.UserInputType == Enum.UserInputType.Touch then

			AtualizarPorPosicao(input.Position.X)
		end
	end)

	--==================================================
	-- OBJETO RETORNADO
	--==================================================

	AtualizarVisual()

	local controle = {
		Set = function(_, valor)
			valor = tonumber(valor)

			if not valor then
				return
			end

			valorAtual = math.clamp(
				math.round(valor),
				valorMin,
				valorMax
			)

			AtualizarVisual()

			ExecutarCallback(callback, valorAtual)
		end,

		Get = function()
			return valorAtual
		end,

		Object = SliderTemplate,
	}

	return controle
end

--==================================================
-- DROPDOWN
--==================================================

function HubLibrary:AddDropdown(
	texto,
	opcoes,
	callback,
	valorInicial
)
	if self._destroyed or not self.ScrollingFrame then
		return nil
	end

	if typeof(opcoes) ~= "table" then
		opcoes = {}
	end

	local valorAtual = valorInicial

	if valorAtual == nil and #opcoes > 0 then
		valorAtual = opcoes[1]
	end

	--==================================================
	-- CONTAINER
	--==================================================

	local DropdownTemplate = Instance.new("Frame")

	DropdownTemplate.Name = "DropdownTemplate"

	DropdownTemplate.Size = UDim2.new(1, -10, 0, 45)

	DropdownTemplate.BackgroundColor3 = COR_CARD

	DropdownTemplate.BorderSizePixel = 0

	DropdownTemplate.ClipsDescendants = true

	DropdownTemplate.Parent = self.ScrollingFrame

	AplicarBorda(DropdownTemplate)
	Arredondar(DropdownTemplate, 6)

	--==================================================
	-- TÍTULO
	--==================================================

	local TitleLabel = Instance.new("TextLabel")

	TitleLabel.Name = "TitleLabel"

	TitleLabel.Size = UDim2.new(0.4, 0, 0, 45)

	TitleLabel.Position = UDim2.new(0, 12, 0, 0)

	TitleLabel.BackgroundTransparency = 1

	TitleLabel.Text = tostring(texto)

	TitleLabel.TextColor3 = Color3.fromRGB(255, 255, 255)

	TitleLabel.Font = Enum.Font.SourceSansBold

	TitleLabel.TextSize = 18

	TitleLabel.TextXAlignment = Enum.TextXAlignment.Left

	TitleLabel.Parent = DropdownTemplate

	--==================================================
	-- BOTÃO
	--==================================================

	local DropdownButton = Instance.new("TextButton")

	DropdownButton.Name = "DropdownButton"

	DropdownButton.Size = UDim2.new(0, 200, 0, 30)

	DropdownButton.Position = UDim2.new(1, -210, 0, 7.5)

	DropdownButton.BackgroundColor3 = COR_CONTROLE

	DropdownButton.TextColor3 = Color3.fromRGB(180, 180, 180)

	DropdownButton.Font = Enum.Font.SourceSans

	DropdownButton.TextSize = 16

	DropdownButton.Parent = DropdownTemplate

	AplicarBorda(DropdownButton, 2)
	Arredondar(DropdownButton, 5)

	if valorAtual ~= nil then
		DropdownButton.Text = tostring(valorAtual)
	else
		DropdownButton.Text = "Select Option..."
	end

	--==================================================
	-- OPÇÕES
	--==================================================

	local ContainerOpcoes = Instance.new("Frame")

	ContainerOpcoes.Name = "ContainerOpcoes"

	ContainerOpcoes.Size = UDim2.new(
		0,
		200,
		0,
		math.max(#opcoes * 30, 1)
	)

	ContainerOpcoes.Position = UDim2.new(
		1,
		-210,
		0,
		45
	)

	ContainerOpcoes.BackgroundColor3 = Color3.fromRGB(25, 25, 25)

	ContainerOpcoes.BorderSizePixel = 0

	ContainerOpcoes.Visible = false

	ContainerOpcoes.ZIndex = 10

	ContainerOpcoes.Parent = DropdownTemplate

	AplicarBorda(ContainerOpcoes, 2)

	Arredondar(ContainerOpcoes, 5)

	local optionLayout = Instance.new("UIListLayout")

	optionLayout.SortOrder = Enum.SortOrder.LayoutOrder

	optionLayout.Parent = ContainerOpcoes

	--==================================================
	-- ABRIR / FECHAR
	--==================================================

	local aberto = false

	local function Fechar()
		aberto = false

		ContainerOpcoes.Visible = false

		TweenService:Create(
			DropdownTemplate,
			TweenInfo.new(0.2),
			{
				Size = UDim2.new(1, -10, 0, 45)
			}
		):Play()
	end

	local function Abrir()
		if #opcoes == 0 then
			return
		end

		aberto = true

		ContainerOpcoes.Visible = true

		local alturaAlvo =
			45 + (#opcoes * 30) + 10

		TweenService:Create(
			DropdownTemplate,
			TweenInfo.new(0.2),
			{
				Size = UDim2.new(
					1,
					-10,
					0,
					alturaAlvo
				)
			}
		):Play()
	end

	DropdownButton.MouseButton1Click:Connect(function()
		if aberto then
			Fechar()
		else
			Abrir()
		end
	end)

	--==================================================
	-- CRIAR OPÇÕES
	--==================================================

	for _, opcao in ipairs(opcoes) do
		local OptBtn = Instance.new("TextButton")

		OptBtn.Name = "Option"

		OptBtn.Size = UDim2.new(1, 0, 0, 30)

		OptBtn.BackgroundColor3 = Color3.fromRGB(25, 25, 25)

		OptBtn.BorderSizePixel = 0

		OptBtn.Text = tostring(opcao)

		OptBtn.TextColor3 = Color3.fromRGB(255, 255, 255)

		OptBtn.Font = Enum.Font.SourceSans

		OptBtn.TextSize = 16

		OptBtn.ZIndex = 11

		OptBtn.Parent = ContainerOpcoes

		OptBtn.MouseButton1Click:Connect(function()
			valorAtual = opcao

			DropdownButton.Text = tostring(opcao)

			Fechar()

			ExecutarCallback(callback, opcao)
		end)
	end

	--==================================================
	-- OBJETO RETORNADO
	--==================================================

	local controle = {
		Set = function(_, valor)
			valorAtual = valor

			DropdownButton.Text =
				valor == nil
				and "Select Option..."
				or tostring(valor)

			ExecutarCallback(callback, valorAtual)
		end,

		Get = function()
			return valorAtual
		end,

		Open = function()
			Abrir()
		end,

		Close = function()
			Fechar()
		end,

		Object = DropdownTemplate,
	}

	return controle
end

function HubLibrary:AddInfoCard(

	iconeId,
	listaDeTextos

	if self._destroyed or not self.ScrollingFrame then

	InfoCardTemplate.BorderSizePixel = 0

	Icon.Position = UDim2.new(0, 12, 0, 45)

	ListInformations.Name = "ListInformations"

	end

	return {
		Object = InfoCardTemplate,
	}
end

--==================================================
-- BUTTON
--==================================================

function HubLibrary:AddButton(texto, callback)
	if self._destroyed or not self.ScrollingFrame then
		return nil
	end

	local ButtonTemplate = Instance.new("TextButton")

	ButtonTemplate.Name = "ButtonTemplate"

	ButtonTemplate.Size = UDim2.new(1, -10, 0, 45)

	ButtonTemplate.BackgroundColor3 = COR_CARD

	ButtonTemplate.BorderSizePixel = 0

	ButtonTemplate.Text = tostring(texto)

	ButtonTemplate.TextColor3 =
		Color3.fromRGB(255, 255, 255)

	ButtonTemplate.Font = Enum.Font.SourceSansBold

	ButtonTemplate.TextSize = 18

	ButtonTemplate.AutoButtonColor = false

	ButtonTemplate.Parent = self.ScrollingFrame

	AplicarBorda(ButtonTemplate)
	Arredondar(ButtonTemplate, 6)

	ButtonTemplate.MouseEnter:Connect(function()
		TweenService:Create(
			ButtonTemplate,
			TweenInfo.new(0.12),
			{
				BackgroundColor3 =
					Color3.fromRGB(50, 50, 50)
			}
		):Play()
	end)

	ButtonTemplate.MouseLeave:Connect(function()
		TweenService:Create(
			ButtonTemplate,
			TweenInfo.new(0.12),
			{
				BackgroundColor3 = COR_CARD
			}
		):Play()
	end)

	ButtonTemplate.MouseButton1Click:Connect(function()
		ExecutarCallback(callback)
	end)

	return {
		Object = ButtonTemplate,
	}
end

--==================================================
-- DESTROY
--==================================================

function HubLibrary:Destroy()
	if self._destroyed then
		return
	end

	self._destroyed = true

	if self.ScreenGui then
		self.ScreenGui:Destroy()
	end

	self.ScreenGui = nil
	self.BackGround = nil
	self.TabButtonsFrame = nil
	self.Abas = {}
	self.AbaAtiva = nil
end

--==================================================
-- RETORNA A BIBLIOTECA
--==================================================

return HubLibrary

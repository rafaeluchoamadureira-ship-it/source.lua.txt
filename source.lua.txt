-- HUBLIBRARY ADAPTADA PARA A INTERFACE FEITA A MAO
--
-- Hierarquia esperada:
--
-- Menu Hub [ScreenGui]
-- └─ BackGround [Frame]
--    ├─ BackGroundTop [Frame]
--    ├─ TabButtonsFrame [Frame]
--    │  ├─ UIListLayout
--    │  └─ Aba1 [TextButton]
--    ├─ ScrollingFrame [ScrollingFrame]
--    │  ├─ UIListLayout
--    │  └─ Templates [Folder]
--    │     ├─ InfoCardTemplate [Frame]
--    │     ├─ ToggleTemplate [Frame]
--    │     ├─ DropdownTemplate [Frame]
--    │     └─ SliderTemplate [Frame]
--    ├─ Close [TextButton]
--    ├─ TabButtonsFrame [Frame]
--    │  ├─ UIListLayout
--    │  ├─ DiscordButton [TextButton]
--    │  └─ ConfigButton [TextButton]
--    └─ MenuName [TextLabel]
--
-- Observacao:
-- Ha DOIS TabButtonsFrame com o mesmo nome.
-- A biblioteca encontra o primeiro pela presenca de "Aba1"
-- e trata o outro como a barra de botoes fixos.
--
-- Os templates existentes no Studio sao usados como base visual.
-- A biblioteca NAO recria o design dos controles.
--==================================================

local COR_FUNDO = Color3.fromRGB(25, 25, 25)
local COR_TOPO = Color3.fromRGB(215, 20, 20)
local COR_BORDA = Color3.fromRGB(0, 0, 0)

local COR_CARD = Color3.fromRGB(35, 35, 35)
local COR_CONTROLE = Color3.fromRGB(20, 20, 20)
local COR_VERDE = Color3.fromRGB(0, 235, 60)

local TAMANHO_BORDA = 3

local COR_TAB_NORMAL = Color3.fromRGB(40, 40, 40)
local COR_TAB_TEXTO_NORMAL = Color3.fromRGB(200, 200, 200)
local COR_TAB_TEXTO_ATIVA = Color3.fromRGB(255, 255, 255)

local function AplicarBorda(parent, espessura)
	if not parent then
		return nil
	end

	local stroke = parent:FindFirstChildOfClass("UIStroke")

	if stroke then
		if espessura then
			stroke.Thickness = espessura
		end
		return stroke
	end

	stroke = Instance.new("UIStroke")
	stroke.Thickness = espessura or TAMANHO_BORDA
	stroke.Color = COR_BORDA
	stroke.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
	stroke.Parent = parent

	return stroke
end

local function Arredondar(parent, raio)
	if not parent then
		return nil
	end

	local corner = parent:FindFirstChildOfClass("UICorner")

	if corner then
		return corner
	end

	corner = Instance.new("UICorner")
	corner.CornerRadius = UDim.new(0, raio or 6)
	corner.Parent = parent

	return corner
end

local function ExecutarCallback(callback, ...)
	if typeof(callback) ~= "function" then
		return
	end

	local args = table.pack(...)

	task.spawn(function()
		local sucesso, erro = pcall(function()
			callback(table.unpack(args, 1, args.n))
		end)

		if not sucesso then
			warn("[HubLibrary] Erro no callback:", erro)
		end
	end)
end

local function NormalizarImagem(id)
	if id == nil then
		return "rbxassetid://0"
	end

	local texto = tostring(id)

	if texto == "" then
		return "rbxassetid://0"
	end

	if string.find(texto, "rbxassetid://", 1, true) then
		return texto
	end

	if string.find(texto, "rbxasset.com", 1, true) then
		return texto
	end

	return "rbxassetid://" .. texto
end

local function Conectar(connections, signal, callback)
	if not signal then
		return nil
	end

	local connection = signal:Connect(callback)
	table.insert(connections, connection)

	return connection
end

local function DesconectarLista(connections)
	if not connections then
		return
	end

	for _, connection in ipairs(connections) do
		if typeof(connection) == "RBXScriptConnection" then
			pcall(function()
				connection:Disconnect()
			end)
		end
	end

	table.clear(connections)
end

local function EncontrarTabFrame(backGround)
	local candidatos = {}

	for _, child in ipairs(backGround:GetChildren()) do
		if child:IsA("Frame") and child.Name == "TabButtonsFrame" then
			table.insert(candidatos, child)
		end
	end

	for _, frame in ipairs(candidatos) do
		if frame:FindFirstChild("Aba1") then
			return frame
		end
	end

	return candidatos[1]
end

local function EncontrarFrameFixos(backGround, tabFrame)
	for _, child in ipairs(backGround:GetChildren()) do
		if child:IsA("Frame")
			and child.Name == "TabButtonsFrame"
			and child ~= tabFrame then
			return child
		end
	end

	return nil
end

local function EncontrarTemplate(templates, nome)
	if not templates then
		return nil
	end

	local object = templates:FindFirstChild(nome)

	if object and object:IsA("GuiObject") then
		return object
	end

	return nil
end

local function EncontrarTitulo(template)
	if not template then
		return nil
	end

	local title = template:FindFirstChild("TitleLabel")

	if title and title:IsA("TextLabel") then
		return title
	end

	return template:FindFirstChildWhichIsA("TextLabel", true)
end

local function EncontrarToggleButton(template)
	if not template then
		return nil
	end

	local button = template:FindFirstChild("ToggleButton")

	if button and button:IsA("TextButton") then
		return button
	end

	return template:FindFirstChildWhichIsA("TextButton", true)
end

local function EncontrarDropdownButton(template)
	if not template then
		return nil
	end

	-- Na hierarquia informada, o TextButton se chama "Frame".
	local button = template:FindFirstChild("Frame")

	if button and button:IsA("TextButton") then
		return button
	end

	return template:FindFirstChildWhichIsA("TextButton", true)
end

local function EncontrarDropdownTexto(button)
	if not button then
		return nil
	end

	local label = button:FindFirstChild("TextLabel")

	if label and label:IsA("TextLabel") then
		return label
	end

	for _, child in ipairs(button:GetChildren()) do
		if child:IsA("TextLabel") then
			return child
		end
	end

	return nil
end

local function EsconderTemplates(templates)
	if not templates then
		return
	end

	for _, child in ipairs(templates:GetChildren()) do
		if child:IsA("GuiObject") then
			child.Visible = false
		end
	end
end

local function CopiarAparenciaScrolling(origem, destino)
	destino.Size = origem.Size
	destino.Position = origem.Position
	destino.AnchorPoint = origem.AnchorPoint
	destino.BackgroundColor3 = origem.BackgroundColor3
	destino.BackgroundTransparency = origem.BackgroundTransparency
	destino.BorderColor3 = origem.BorderColor3
	destino.BorderSizePixel = origem.BorderSizePixel
	destino.ClipsDescendants = origem.ClipsDescendants
	destino.LayoutOrder = origem.LayoutOrder
	destino.ScrollBarImageColor3 = origem.ScrollBarImageColor3
	destino.ScrollBarImageTransparency = origem.ScrollBarImageTransparency
	destino.ScrollBarThickness = origem.ScrollBarThickness
	destino.ScrollBarImageTransparency = origem.ScrollBarImageTransparency
	destino.ScrollingDirection = origem.ScrollingDirection
	destino.AutomaticCanvasSize = Enum.AutomaticSize.None
	destino.CanvasSize = UDim2.new(0, 0, 0, 0)
	destino.Visible = false

	pcall(function()
		destino.BottomImage = origem.BottomImage
	end)

	pcall(function()
		destino.MidImage = origem.MidImage
	end)

	pcall(function()
		destino.TopImage = origem.TopImage
	end)
end

local function CriarPagina(baseScrolling, backGround, nome)
	local page = Instance.new("ScrollingFrame")
	page.Name = tostring(nome) .. "Scroll"
	page.Parent = backGround

	CopiarAparenciaScrolling(baseScrolling, page)

	for _, child in ipairs(baseScrolling:GetChildren()) do
		if child:IsA("UIPadding") then
			local padding = child:Clone()
			padding.Parent = page
		end
	end

	local layoutBase = baseScrolling:FindFirstChildOfClass("UIListLayout")
	local layout = Instance.new("UIListLayout")

	if layoutBase then
		layout.Padding = layoutBase.Padding
		layout.FillDirection = layoutBase.FillDirection
		layout.HorizontalAlignment = layoutBase.HorizontalAlignment
		layout.VerticalAlignment = layoutBase.VerticalAlignment
		layout.SortOrder = layoutBase.SortOrder
		layout.Wraps = layoutBase.Wraps
	else
		layout.Padding = UDim.new(0, 10)
		layout.SortOrder = Enum.SortOrder.LayoutOrder
	end

	layout.Parent = page

	return page, layout
end

--==================================================
-- CARREGAR A INTERFACE EXISTENTE
--==================================================

function HubLibrary.Carregar(nomeDoHub, screenGuiOpcional)
	local self = setmetatable({}, HubLibrary)

	self.Abas = {}
	self.AbaAtiva = nil
	self._destroyed = false
	self._connections = {}
	self._itemConnections = {}

	local PlayerGui = LocalPlayer:WaitForChild("PlayerGui")

	local ScreenGui = screenGuiOpcional

	if not ScreenGui then
		ScreenGui = PlayerGui:FindFirstChild("Menu Hub")
	end

	if not ScreenGui then
		error('[HubLibrary] "Menu Hub" nao foi encontrada no PlayerGui.')
	end

	if not ScreenGui:IsA("ScreenGui") then
		error('[HubLibrary] "Menu Hub" precisa ser uma ScreenGui.')
	end

	local BackGround = ScreenGui:FindFirstChild("BackGround")

	if not BackGround or not BackGround:IsA("Frame") then
		error('[HubLibrary] "BackGround" nao encontrada.')
	end

	local ScrollingFrame = BackGround:FindFirstChild("ScrollingFrame")

	if not ScrollingFrame or not ScrollingFrame:IsA("ScrollingFrame") then
		error('[HubLibrary] "ScrollingFrame" nao encontrada.')
	end

	local Templates = ScrollingFrame:FindFirstChild("Templates")

	if not Templates or not Templates:IsA("Folder") then
		error('[HubLibrary] "Templates" nao encontrada dentro de ScrollingFrame.')
	end

	local TabButtonsFrame = EncontrarTabFrame(BackGround)

	if not TabButtonsFrame then
		error('[HubLibrary] O primeiro "TabButtonsFrame" nao foi encontrado.')
	end

	local FixedButtonsFrame =
		EncontrarFrameFixos(BackGround, TabButtonsFrame)

	self.ScreenGui = ScreenGui
	self.BackGround = BackGround
	self.Templates = Templates
	self.TemplateScrollingFrame = ScrollingFrame
	self.TabButtonsFrame = TabButtonsFrame
	self.FixedButtonsFrame = FixedButtonsFrame

	--==================================================
	-- IMPORTANTE:
	-- O ScrollingFrame original fica escondido.
	-- Ele funciona como base visual para as paginas.
	--==================================================

	ScrollingFrame.Visible = false
	EsconderTemplates(Templates)

	local menuName = BackGround:FindFirstChild("MenuName")

	if menuName and menuName:IsA("TextLabel") then
		menuName.Text = tostring(nomeDoHub or "Frank Hub")
	end

	--==================================================
	-- FECHAR
	--==================================================

	local closeButton = BackGround:FindFirstChild("Close")

	if closeButton and closeButton:IsA("GuiButton") then
		Conectar(self._connections, closeButton.MouseButton1Click, function()
			if not self._destroyed then
				self:Destroy()
			end
		end)
	end

	--==================================================
	-- ARRASTAR
	--==================================================

	local BackGroundTop = BackGround:FindFirstChild("BackGroundTop")

	if BackGroundTop and BackGroundTop:IsA("GuiObject") then
		local arrastando = false
		local inicioMouse = nil
		local inicioPosicao = nil

		Conectar(self._connections, BackGroundTop.InputBegan, function(input)
			if input.UserInputType == Enum.UserInputType.MouseButton1
				or input.UserInputType == Enum.UserInputType.Touch then

				arrastando = true
				inicioMouse = input.Position
				inicioPosicao = BackGround.Position
			end
		end)

		Conectar(self._connections, BackGroundTop.InputEnded, function(input)
			if input.UserInputType == Enum.UserInputType.MouseButton1
				or input.UserInputType == Enum.UserInputType.Touch then

				arrastando = false
			end
		end)

		Conectar(self._connections, UserInputService.InputChanged, function(input)
			if self._destroyed or not arrastando then
				return
			end

			if input.UserInputType ~= Enum.UserInputType.MouseMovement
				and input.UserInputType ~= Enum.UserInputType.Touch then
				return
			end

			if not inicioMouse or not inicioPosicao then
				return
			end

			local delta = input.Position - inicioMouse

			BackGround.Position = UDim2.new(
				inicioPosicao.X.Scale,
				inicioPosicao.X.Offset + delta.X,
				inicioPosicao.Y.Scale,
				inicioPosicao.Y.Offset + delta.Y
			)
		end)
	end

	--==================================================
	-- BOTOES FIXOS
	--==================================================

	if FixedButtonsFrame then
		local DiscordButton = FixedButtonsFrame:FindFirstChild("DiscordButton")
		local ConfigButton = FixedButtonsFrame:FindFirstChild("ConfigButton")

		if DiscordButton and DiscordButton:IsA("GuiButton") then
			Conectar(self._connections, DiscordButton.MouseButton1Click, function()
				ExecutarCallback(self.OnDiscord)
			end)
		end

		if ConfigButton and ConfigButton:IsA("GuiButton") then
			Conectar(self._connections, ConfigButton.MouseButton1Click, function()
				if self.ConfigCallback then
					ExecutarCallback(self.ConfigCallback)
					return
				end

				if self.ConfigTab then
					self:SelecionarAba(self.ConfigTab)
				end
			end)
		end
	end

	return self
end

--==================================================
-- SELECIONAR ABA
--==================================================

function HubLibrary:SelecionarAba(aba)
	if self._destroyed or not aba then
		return
	end

	for _, outraAba in ipairs(self.Abas) do
		if outraAba.Scroll and outraAba.Scroll.Parent then
			outraAba.Scroll.Visible = false
		end

		if outraAba.Btn and outraAba.Btn.Parent then
			outraAba.Btn.BackgroundColor3 = COR_TAB_NORMAL
			outraAba.Btn.TextColor3 = COR_TAB_TEXTO_NORMAL
		end
	end

	if aba.Scroll and aba.Scroll.Parent then
		aba.Scroll.Visible = true
	end

	if aba.Btn and aba.Btn.Parent then
		aba.Btn.BackgroundColor3 = COR_TOPO
		aba.Btn.TextColor3 = COR_TAB_TEXTO_ATIVA
	end

	self.AbaAtiva = aba

	ExecutarCallback(aba._onSelected, aba)
end

function HubLibrary:AddTab(nomeDaAba)
	if self._destroyed then
		return nil
	end

	local templateButton = self.TabButtonsFrame:FindFirstChild("Aba1")

	if not templateButton or not templateButton:IsA("TextButton") then
		warn('[HubLibrary] "Aba1" nao encontrada.')
		return nil
	end

	-- "Aba1" e o template desenhado a mao.
	templateButton.Visible = false

	local TabButton = templateButton:Clone()
	TabButton.Name = tostring(nomeDaAba) .. "Button"
	TabButton.Text = tostring(nomeDaAba)
	TabButton.Visible = true
	TabButton.Parent = self.TabButtonsFrame

	local ScrollingPage, PageLayout = CriarPagina(
		self.TemplateScrollingFrame,
		self.BackGround,
		nomeDaAba
	)

	local abaSelf = {
		ScrollingFrame = ScrollingPage,
		Scroll = ScrollingPage,
		Btn = TabButton,
		_hub = self,
		_destroyed = false,
		Name = tostring(nomeDaAba),
		Layout = PageLayout,
	}

	-- Se a aba tiver sido destruida, impede callbacks.
	local clickConnection = TabButton.MouseButton1Click:Connect(function()
		if not self._destroyed and not abaSelf._destroyed then
			self:SelecionarAba(abaSelf)
		end
	end)

	table.insert(self._connections, clickConnection)
	table.insert(self.Abas, abaSelf)

	-- Primeira aba fica ativa automaticamente.
	if #self.Abas == 1 then
		self:SelecionarAba(abaSelf)
	end

	return setmetatable(abaSelf, {
		__index = HubLibrary
	})
end

-- TOGGLE
--==================================================

function HubLibrary:AddToggle(texto, callback, valorInicial)
	if self._destroyed or not self.ScrollingFrame then
		return nil
	end

	local hub = self._hub or self
	local parentScroll = self.Scroll or self.ScrollingFrame

	local template = EncontrarTemplate(hub.Templates, "ToggleTemplate")

	if not template then
		return nil
	end

	template.Visible = false

	local ToggleTemplate = template:Clone()
	ToggleTemplate.Visible = true
	ToggleTemplate.Parent = parentScroll

	local TitleLabel = EncontrarTitulo(ToggleTemplate)
	local ToggleButton = EncontrarToggleButton(ToggleTemplate)

	if TitleLabel and TitleLabel:IsA("TextLabel") then
		TitleLabel.Text = tostring(texto)
	end

	if not ToggleButton then
		warn("[HubLibrary] ToggleButton nao encontrado.")
		ToggleTemplate:Destroy()
		return nil
	end

	local ativo = valorInicial == true

	local function AtualizarVisual()
		local cor = ativo and COR_VERDE or COR_CONTROLE

		ToggleButton.BackgroundColor3 = cor

		local ButtonLine = ToggleButton:FindFirstChild("ButtonLine")

		if ButtonLine and ButtonLine:IsA("GuiObject") then
			ButtonLine.BackgroundColor3 = cor
		end
	end

	local function Set(valor, dispararCallback)
		ativo = valor == true
		AtualizarVisual()

		if dispararCallback ~= false then
			ExecutarCallback(callback, ativo)
		end
	end

	AtualizarVisual()

	Conectar(hub._itemConnections, ToggleButton.MouseButton1Click, function()
		if ToggleTemplate.Parent then
			Set(not ativo, true)
		end
	end)

	return {
		Set = function(_, valor)
			Set(valor, true)
		end,

		Get = function()
			return ativo
		end,

		Object = ToggleTemplate,
		Button = ToggleButton,
	}
end

--==================================================
-- SLIDER
--==================================================

function HubLibrary:AddSlider(
	texto,
	valorMin,
	valorMax,
	valorPadrao,
	callback
)
	if self._destroyed or not self.ScrollingFrame then
		return nil
	end

	local hub = self._hub or self
	local parentScroll = self.Scroll or self.ScrollingFrame

	valorMin = tonumber(valorMin) or 0
	valorMax = tonumber(valorMax) or 100
	valorPadrao = tonumber(valorPadrao) or valorMin

	if valorMax < valorMin then
		valorMin, valorMax = valorMax, valorMin
	end

	valorPadrao = math.clamp(
		valorPadrao,
		valorMin,
		valorMax
	)

	local valorAtual = valorPadrao

	local template = EncontrarTemplate(
		hub.Templates,
		"SliderTemplate"
	)

	if not template then
		return nil
	end

	template.Visible = false

	local SliderTemplate = template:Clone()
	SliderTemplate.Visible = true
	SliderTemplate.Parent = parentScroll

	local TitleLabel = EncontrarTitulo(SliderTemplate)
	local SliderTrack = SliderTemplate:FindFirstChild("SliderTrack")

	if not SliderTrack then
		warn("[HubLibrary] SliderTrack nao encontrado.")
		SliderTemplate:Destroy()
		return nil
	end

	local FillButton = SliderTrack:FindFirstChild("FillButton")
	local Progress = SliderTrack:FindFirstChild("Progress")

	if not FillButton
		or not FillButton:IsA("TextButton")
		or not Progress
		or not Progress:IsA("GuiObject") then

		warn("[HubLibrary] FillButton/Progress nao encontrado.")
		SliderTemplate:Destroy()
		return nil
	end

	if TitleLabel and TitleLabel:IsA("TextLabel") then
		TitleLabel.Text =
			tostring(texto) .. " (" .. tostring(valorAtual) .. ")"
	end

	local function AtualizarVisual()
		local percentual = 0

		if valorMax ~= valorMin then
			percentual =
				(valorAtual - valorMin)
				/ (valorMax - valorMin)
		end

		percentual = math.clamp(percentual, 0, 1)

		Progress.Size =
			UDim2.new(percentual, 0, 1, 0)

		FillButton.Position =
			UDim2.new(percentual, 0, 0.5, 0)

		if TitleLabel and TitleLabel:IsA("TextLabel") then
			TitleLabel.Text =
				tostring(texto) .. " (" .. tostring(valorAtual) .. ")"
		end
	end

	local function DefinirPorX(posicaoX)
		if not SliderTrack.Parent then
			return
		end

		local largura = SliderTrack.AbsoluteSize.X

		if largura <= 0 then
			return
		end

		local percentual =
			(posicaoX - SliderTrack.AbsolutePosition.X)
			/ largura

		percentual = math.clamp(percentual, 0, 1)

		valorAtual = math.round(
			valorMin
			+ ((valorMax - valorMin) * percentual)
		)

		AtualizarVisual()
		ExecutarCallback(callback, valorAtual)
	end

	local arrastando = false

	Conectar(hub._itemConnections, SliderTrack.InputBegan, function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then

			arrastando = true
			DefinirPorX(input.Position.X)
		end
	end)

	Conectar(hub._itemConnections, FillButton.InputBegan, function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then

			arrastando = true
			DefinirPorX(input.Position.X)
		end
	end)

	Conectar(hub._itemConnections, UserInputService.InputEnded, function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then

			arrastando = false
		end
	end)

	Conectar(hub._itemConnections, UserInputService.InputChanged, function(input)
		if not arrastando then
			return
		end

		if input.UserInputType ~= Enum.UserInputType.MouseMovement
			and input.UserInputType ~= Enum.UserInputType.Touch then
			return
		end

		DefinirPorX(input.Position.X)
	end)

	AtualizarVisual()

	return {
		Set = function(_, valor)
			local numero = tonumber(valor)

			if numero == nil then
				return
			end

			valorAtual = math.clamp(
				math.round(numero),
				valorMin,
				valorMax
			)

			AtualizarVisual()
			ExecutarCallback(callback, valorAtual)
		end,

		Get = function()
			return valorAtual
		end,

		Object = SliderTemplate,
		Track = SliderTrack,
		FillButton = FillButton,
	}
end

--==================================================
-- DROPDOWN
--==================================================

function HubLibrary:AddDropdown(
	texto,
	opcoes,
	callback,
	valorInicial
)
	if self._destroyed or not self.ScrollingFrame then
		return nil
	end

	local hub = self._hub or self
	local parentScroll = self.Scroll or self.ScrollingFrame

	opcoes =
		typeof(opcoes) == "table"
		and opcoes
		or {}

	local valorAtual = valorInicial

	if valorAtual == nil then
		if #opcoes > 0 then
			valorAtual = opcoes[1]
		else
			valorAtual = "Select Option..."
		end
	end

	local template =
		EncontrarTemplate(
			hub.Templates,
			"DropdownTemplate"
		)

	if not template then
		return nil
	end

	template.Visible = false

	local DropdownTemplate = template:Clone()
	DropdownTemplate.Visible = true
	DropdownTemplate.ClipsDescendants = false
	DropdownTemplate.Parent = parentScroll

	local TitleLabel = DropdownTemplate:FindFirstChild("TitleLabel")
	local DropdownButton =
		EncontrarDropdownButton(DropdownTemplate)

	if TitleLabel and TitleLabel:IsA("TextLabel") then
		TitleLabel.Text = tostring(texto)
	end

	if not DropdownButton then
		warn("[HubLibrary] Botao do Dropdown nao encontrado.")
		DropdownTemplate:Destroy()
		return nil
	end

	local SelectedLabel =
		EncontrarDropdownTexto(DropdownButton)

	local ContainerOpcoes = Instance.new("Frame")
	ContainerOpcoes.Name = "ContainerOpcoes"
	ContainerOpcoes.Size = UDim2.new(
		DropdownButton.Size.X.Scale,
		DropdownButton.Size.X.Offset,
		0,
		math.max(#opcoes * 30, 1)
	)

	ContainerOpcoes.Position = UDim2.new(
		DropdownButton.Position.X.Scale,
		DropdownButton.Position.X.Offset,
		0,
		DropdownButton.Position.Y.Offset
			+ DropdownButton.Size.Y.Offset
			+ 5
	)

	ContainerOpcoes.BackgroundColor3 =
		Color3.fromRGB(25, 25, 25)

	ContainerOpcoes.BorderSizePixel = 0
	ContainerOpcoes.Visible = false
	ContainerOpcoes.ZIndex = 100
	ContainerOpcoes.Parent = DropdownTemplate

	AplicarBorda(ContainerOpcoes, 2)
	Arredondar(ContainerOpcoes, 5)

	local OptionLayout = Instance.new("UIListLayout")
	OptionLayout.SortOrder = Enum.SortOrder.LayoutOrder
	OptionLayout.Parent = ContainerOpcoes

	local aberto = false
	local tamanhoOriginal = DropdownTemplate.Size

	local function AtualizarTexto()
		local textoSelecionado =
			valorAtual == nil
			and "Select Option..."
			or tostring(valorAtual)

		if SelectedLabel and SelectedLabel:IsA("TextLabel") then
			SelectedLabel.Text = textoSelecionado
		elseif DropdownButton:IsA("TextButton") then
			DropdownButton.Text = textoSelecionado
		end
	end

	local function Fechar()
		aberto = false
		ContainerOpcoes.Visible = false

		TweenService:Create(
			DropdownTemplate,
			TweenInfo.new(0.18),
			{
				Size = tamanhoOriginal
			}
		):Play()
	end

	local function Abrir()
		if #opcoes == 0 then
			return
		end

		aberto = true
		ContainerOpcoes.Visible = true

		local alturaExtra =
			(#opcoes * 30) + 10

		TweenService:Create(
			DropdownTemplate,
			TweenInfo.new(0.18),
			{
				Size = UDim2.new(
					tamanhoOriginal.X.Scale,
					tamanhoOriginal.X.Offset,
					0,
					tamanhoOriginal.Y.Offset + alturaExtra
				)
			}
		):Play()
	end

	Conectar(
		hub._itemConnections,
		DropdownButton.MouseButton1Click,
		function()
			if aberto then
				Fechar()
			else
				Abrir()
			end
		end
	)

	for indice, opcao in ipairs(opcoes) do
		local OptionButton = Instance.new("TextButton")
		OptionButton.Name = "Option" .. tostring(indice)
		OptionButton.Size = UDim2.new(1, 0, 0, 30)
		OptionButton.BackgroundColor3 =
			Color3.fromRGB(25, 25, 25)
		OptionButton.BorderSizePixel = 0
		OptionButton.Text = tostring(opcao)
		OptionButton.TextColor3 =
			Color3.fromRGB(255, 255, 255)
		OptionButton.Font = Enum.Font.SourceSans
		OptionButton.TextSize = 16
		OptionButton.AutoButtonColor = false
		OptionButton.ZIndex = 101
		OptionButton.LayoutOrder = indice
		OptionButton.Parent = ContainerOpcoes

		Conectar(
			hub._itemConnections,
			OptionButton.MouseEnter,
			function()
				if OptionButton.Parent then
					TweenService:Create(
						OptionButton,
						TweenInfo.new(0.1),
						{
							BackgroundColor3 =
								Color3.fromRGB(45, 45, 45)
						}
					):Play()
				end
			end
		)

		Conectar(
			hub._itemConnections,
			OptionButton.MouseLeave,
			function()
				if OptionButton.Parent then
					TweenService:Create(
						OptionButton,
						TweenInfo.new(0.1),
						{
							BackgroundColor3 =
								Color3.fromRGB(25, 25, 25)
						}
					):Play()
				end
			end
		)

		Conectar(
			hub._itemConnections,
			OptionButton.MouseButton1Click,
			function()
				valorAtual = opcao
				AtualizarTexto()
				Fechar()
				ExecutarCallback(callback, opcao)
			end
		)
	end

	AtualizarTexto()

	return {
		Set = function(_, valor)
			valorAtual = valor
			AtualizarTexto()

			ExecutarCallback(
				callback,
				valorAtual
			)
		end,

		Get = function()
			return valorAtual
		end,

		Open = function()
			Abrir()
		end,

		Close = function()
			Fechar()
		end,

		Object = DropdownTemplate,
		Button = DropdownButton,
	}
end

--==================================================
-- INFO CARD
--==================================================

function HubLibrary:AddInfoCard(
	tituloCard,
	iconeId,
	listaDeTextos
)
	if self._destroyed or not self.ScrollingFrame then
		return nil
	end

	local hub = self._hub or self
	local parentScroll = self.Scroll or self.ScrollingFrame

	listaDeTextos =
		typeof(listaDeTextos) == "table"
		and listaDeTextos
		or {}

	local template =
		EncontrarTemplate(
			hub.Templates,
			"InfoCardTemplate"
		)

	if not template then
		return nil
	end

	template.Visible = false

	local InfoCardTemplate = template:Clone()
	InfoCardTemplate.Visible = true
	InfoCardTemplate.Parent = parentScroll

	local TitleLabel =
		InfoCardTemplate:FindFirstChild("TitleLabel")

	local Icon =
		InfoCardTemplate:FindFirstChild("Icon")

	local ListInformations =
		InfoCardTemplate:FindFirstChild(
			"ListInformations"
		)

	if TitleLabel and TitleLabel:IsA("TextLabel") then
		TitleLabel.Text = tostring(tituloCard)
	end

	if Icon and Icon:IsA("ImageLabel") then
		Icon.Image =
			NormalizarImagem(iconeId)
	end

	if ListInformations
		and ListInformations:IsA("GuiObject") then

		local InformationTemplate =
			ListInformations:FindFirstChild(
				"InformationTextTemplate"
			)

		if InformationTemplate
			and InformationTemplate:IsA("TextLabel") then

			InformationTemplate.Visible = false

			for _, child in ipairs(
				ListInformations:GetChildren()
			) do
				if child:IsA("TextLabel")
					and child ~= InformationTemplate then
					child:Destroy()
				end
			end

			for indice, texto in ipairs(listaDeTextos) do
				local InformationText =
					InformationTemplate:Clone()

				InformationText.Name =
					"InformationText" .. tostring(indice)

				InformationText.Text =
					tostring(texto)

				InformationText.Visible = true
				InformationText.LayoutOrder = indice
				InformationText.Parent =
					ListInformations
			end
		end
	end

	return {
		Object = InfoCardTemplate
	}
end

--==================================================
-- BUTTON
-- Se existir ButtonTemplate na pasta Templates,
-- ele sera usado. Caso contrario, cria um fallback.
--==================================================

function HubLibrary:AddButton(texto, callback)
	if self._destroyed or not self.ScrollingFrame then
		return nil
	end

	local hub = self._hub or self
	local parentScroll = self.Scroll or self.ScrollingFrame

	local template =
		EncontrarTemplate(
			hub.Templates,
			"ButtonTemplate"
		)

	local ButtonTemplate

	if template then
		template.Visible = false

		ButtonTemplate = template:Clone()
		ButtonTemplate.Visible = true
		ButtonTemplate.Parent = parentScroll
	else
		ButtonTemplate = Instance.new("TextButton")
		ButtonTemplate.Name = "ButtonTemplate"
		ButtonTemplate.Size = UDim2.new(1, -10, 0, 45)
		ButtonTemplate.BackgroundColor3 = COR_CARD
		ButtonTemplate.BorderSizePixel = 0
		ButtonTemplate.TextColor3 =
			Color3.fromRGB(255, 255, 255)
		ButtonTemplate.Font =
			Enum.Font.SourceSansBold
		ButtonTemplate.TextSize = 18
		ButtonTemplate.AutoButtonColor = false
		ButtonTemplate.Parent = parentScroll

		AplicarBorda(ButtonTemplate)
		Arredondar(ButtonTemplate, 6)
	end

	if ButtonTemplate:IsA("TextButton") then
		ButtonTemplate.Text = tostring(texto)
	end

	Conectar(
		hub._itemConnections,
		ButtonTemplate.MouseEnter,
		function()
			if ButtonTemplate.Parent then
				TweenService:Create(
					ButtonTemplate,
					TweenInfo.new(0.12),
					{
						BackgroundColor3 =
							Color3.fromRGB(50, 50, 50)
					}
				):Play()
			end
		end
	)

	Conectar(
		hub._itemConnections,
		ButtonTemplate.MouseLeave,
		function()
			if ButtonTemplate.Parent then
				TweenService:Create(
					ButtonTemplate,
					TweenInfo.new(0.12),
					{
						BackgroundColor3 =
							template
							and template.BackgroundColor3
							or COR_CARD
					}
				):Play()
			end
		end
	)

	Conectar(
		hub._itemConnections,
		ButtonTemplate.MouseButton1Click,
		function()
			ExecutarCallback(callback)
		end
	)

	return {
		Object = ButtonTemplate,
		Button = ButtonTemplate,
	}
end

--==================================================
-- DESTRUIR
--==================================================

function HubLibrary:Destroy()
	if self._destroyed then
		return
	end

	self._destroyed = true

	DesconectarLista(self._connections)
	DesconectarLista(self._itemConnections)

	if self.Abas then
		for _, aba in ipairs(self.Abas) do
			if aba then
				aba._destroyed = true

				if aba.Scroll
					and aba.Scroll.Parent then
					aba.Scroll:Destroy()
				end

				if aba.Btn
					and aba.Btn.Parent then
					aba.Btn:Destroy()
				end
			end
		end
	end

	self.Abas = {}
	self.AbaAtiva = nil

	-- A ScreenGui criada manualmente pelo usuario NAO e destruida.
	self.ScreenGui = nil
	self.BackGround = nil
	self.Templates = nil
	self.TemplateScrollingFrame = nil
	self.ScrollingFrame = nil
	self.TabButtonsFrame = nil
	self.FixedButtonsFrame = nil
end

return HubLibrary

local function UDim2Exact(xs, xo, ys, yo)
	return UDim2.new(xs, xo, ys, yo)
end

local function UDimExact(scale, offset)
	return UDim.new(scale, offset)
end

local function Color3Exact(r, g, b)
	return Color3.new(r, g, b)
end

local function SetSafe(instance, property, value)
	if instance == nil then
		return
	end

	pcall(function()
		instance[property] = value
	end)
end

local function AplicarBorda(parent, thickness, color, borderPosition)
	if not parent then
		return nil
	end

	local stroke = parent:FindFirstChildOfClass("UIStroke")

	if not stroke then
		stroke = Instance.new("UIStroke")
		stroke.Parent = parent
	end

	stroke.Thickness = thickness or 3
	stroke.Color = color or COR_BORDA

	if borderPosition ~= nil then
		SetSafe(stroke, "BorderStrokePosition", borderPosition)
	end

	SetSafe(stroke, "ApplyStrokeMode", Enum.ApplyStrokeMode.Border)
	SetSafe(stroke, "LineJoinMode", Enum.LineJoinMode.Round)

	return stroke
end

local function CriarStroke(parent, name, thickness, color, applyMode, borderPosition, lineJoin)
	local stroke = Instance.new("UIStroke")
	stroke.Name = name or "UIStroke"
	stroke.Thickness = thickness or 3
	stroke.Color = color or COR_BORDA
	stroke.Parent = parent

	if applyMode ~= nil then
		SetSafe(stroke, "ApplyStrokeMode", applyMode)
	end

	if borderPosition ~= nil then
		SetSafe(stroke, "BorderStrokePosition", borderPosition)
	end

	if lineJoin ~= nil then
		SetSafe(stroke, "LineJoinMode", lineJoin)
	end

	return stroke
end

local function CriarCorner(parent, radius)
	local corner = Instance.new("UICorner")
	corner.Name = "UICorner"

	local r = radius or 5
	corner.CornerRadius = UDim.new(0, r)
	corner.Parent = parent

	return corner
end

local function ExecutarCallback(callback, ...)
	if typeof(callback) ~= "function" then
		return
	end

	local args = table.pack(...)

	task.spawn(function()
		local ok, err = pcall(function()
			callback(table.unpack(args, 1, args.n))
		end)

		if not ok then
			warn("[Frank Hub] Callback:", err)
		end
	end)
end

local function Conectar(lista, signal, callback)
	if not signal then
		return nil
	end

	local connection = signal:Connect(callback)
	table.insert(lista, connection)
	return connection
end

local function DesconectarLista(lista)
	if not lista then
		return
	end

	for _, connection in ipairs(lista) do
		if typeof(connection) == "RBXScriptConnection" then
			pcall(function()
				connection:Disconnect()
			end)
		end
	end

	table.clear(lista)
end

local function FontFromFamily(family, weight, italic)
	local style = italic and Enum.FontStyle.Italic or Enum.FontStyle.Normal
	local fontWeight

	if (weight or 400) >= 900 then
		fontWeight = Enum.FontWeight.Black
	elseif (weight or 400) >= 700 then
		fontWeight = Enum.FontWeight.Bold
	elseif (weight or 400) >= 600 then
		fontWeight = Enum.FontWeight.SemiBold
	else
		fontWeight = Enum.FontWeight.Regular
	end

	local ok, result = pcall(function()
		return Font.new(family, fontWeight, style)
	end)

	if ok then
		return result
	end

	return Font.new("rbxasset://fonts/families/Arial.json", fontWeight, style)
end

local function SetTextFont(instance, family, weight, italic)
	if not instance then
		return
	end

	local font = FontFromFamily(family, weight, italic)

	SetSafe(instance, "FontFace", font)

	-- Fallback para ambientes que ainda expõem Font.
	if family:find("GothamSSm", 1, true) then
		SetSafe(instance, "Font", Enum.Font.GothamBold)
	elseif weight >= 700 then
		SetSafe(instance, "Font", italic and Enum.Font.SourceSansItalic or Enum.Font.SourceSansBold)
	else
		SetSafe(instance, "Font", Enum.Font.SourceSans)
	end
end

local function SetLabelProps(label, props)
	for property, value in pairs(props) do
		SetSafe(label, property, value)
	end
end

local function EncontrarFilho(parent, name, className)
	if not parent then
		return nil
	end

	local child = parent:FindFirstChild(name)

	if child and (not className or child:IsA(className)) then
		return child
	end

	return nil
end

local function EncontrarPrimeiro(parent, className)
	if not parent then
		return nil
	end

	return parent:FindFirstChildWhichIsA(className, true)
end

local function DestruirSeExistir(parent, name)
	local old = parent and parent:FindFirstChild(name)
	if old then
		old:Destroy()
	end
end

function HubLibrary:_CreatePage(nome)
	local page = Instance.new("ScrollingFrame")
	page.Name = tostring(nome) .. "Scroll"

	local base = self.TemplateScrollingFrame

	page.Size = base.Size
	page.Position = base.Position
	page.AnchorPoint = base.AnchorPoint
	page.BackgroundColor3 = base.BackgroundColor3
	page.BackgroundTransparency = base.BackgroundTransparency
	page.BorderSizePixel = base.BorderSizePixel
	page.ScrollBarThickness = base.ScrollBarThickness
	page.ScrollBarImageColor3 = base.ScrollBarImageColor3
	page.ScrollBarImageTransparency = base.ScrollBarImageTransparency
	page.AutomaticCanvasSize = Enum.AutomaticSize.None
	page.CanvasSize = UDim2.new(0, 0, 0, 0)
	page.ScrollingDirection = Enum.ScrollingDirection.XY
	page.Visible = false
	page.ZIndex = base.ZIndex
	page.Parent = self.BackGround

	local oldPadding = base:FindFirstChildOfClass("UIPadding")

	if oldPadding then
		local padding = oldPadding:Clone()
		padding.Parent = page
	end

	local layout = Instance.new("UIListLayout")
	layout.Name = "UIListLayout"

	local baseLayout = base:FindFirstChildOfClass("UIListLayout")

	if baseLayout then
		layout.Padding = baseLayout.Padding
		layout.FillDirection = baseLayout.FillDirection
		layout.HorizontalAlignment = baseLayout.HorizontalAlignment
		layout.VerticalAlignment = baseLayout.VerticalAlignment
		layout.SortOrder = baseLayout.SortOrder
	else
		layout.Padding = UDim.new(0, 6)
		layout.FillDirection = Enum.FillDirection.Vertical
		layout.HorizontalAlignment = Enum.HorizontalAlignment.Center
		layout.VerticalAlignment = Enum.VerticalAlignment.Center
		layout.SortOrder = Enum.SortOrder.LayoutOrder
	end

	layout.Parent = page

	local connection = layout:GetPropertyChangedSignal("AbsoluteContentSize"):Connect(function()
		if not self._destroyed and page.Parent then
			page.CanvasSize = UDim2.new(
				0,
				0,
				0,
				math.max(0, layout.AbsoluteContentSize.Y + 20)
			)
		end
	end)

	table.insert(self._connections, connection)

	return page
end

function HubLibrary:_CloneTemplate(name, page)
	local template = self.Templates and self.Templates:FindFirstChild(name)

	if not template then
		warn("[Frank Hub] Template ausente:", name)
		return nil
	end

	local clone = template:Clone()
	clone.Visible = true
	clone.Parent = page

	return clone
end

--==================================================
-- SELECAO DE ABA
--==================================================

function HubLibrary:SelecionarAba(aba)
	if self._destroyed or not aba then
		return
	end

	for _, other in ipairs(self.Abas) do
		if other.Scroll and other.Scroll.Parent then
			other.Scroll.Visible = false
		end

		if other.Btn and other.Btn.Parent then
			other.Btn.BackgroundColor3 = COR_TAB_NORMAL
			other.Btn.TextColor3 = COR_TEXTO
		end
	end

	if aba.Scroll and aba.Scroll.Parent then
		aba.Scroll.Visible = true
	end

	if aba.Btn and aba.Btn.Parent then
		aba.Btn.BackgroundColor3 = COR_TAB_ATIVA
		aba.Btn.TextColor3 = COR_TEXTO
	end

	self.AbaAtiva = aba

	ExecutarCallback(aba._onSelected, aba)
end

	return setmetatable(abaSelf, {

--==================================================
-- ADD TOGGLE
--==================================================

function HubLibrary:AddToggle(texto, callback, valorInicial)
	if self._destroyed or not self.Scroll then
		return nil
	end

	local hub = self._hub or self
	local clone = hub:_CloneTemplate("ToggleTemplate", self.Scroll)

	if not clone then
		return nil
	end

	local TitleLabel = clone:FindFirstChild("TitleLabel")
	local ToggleButton = clone:FindFirstChild("ToggleButton")

	if TitleLabel and TitleLabel:IsA("TextLabel") then
		TitleLabel.Text = tostring(texto)
	end

	if not ToggleButton or not ToggleButton:IsA("TextButton") then
		clone:Destroy()
		warn("[Frank Hub] ToggleButton nao encontrado.")
		return nil
	end

	local ativo = valorInicial == true

	local function Atualizar()
		local cor = ativo and COR_VERDE or COR_CONTROLE
		ToggleButton.BackgroundColor3 = cor

		local line = ToggleButton:FindFirstChild("ButtonLine")

		if line and line:IsA("GuiObject") then
			line.BackgroundColor3 =
				ativo and Color3.fromRGB(255, 255, 255)
				or Color3.fromRGB(100, 100, 100)

			if line:IsA("Frame") then
				line.Visible = true
			end
		end
	end

	local function Set(valor, disparar)
		ativo = valor == true

		local targetColor = ativo
			and COR_VERDE
			or COR_CONTROLE

		if ToggleButton.Parent then
			TweenService:Create(
				ToggleButton,
				TweenInfo.new(0.12, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
				{
					BackgroundColor3 = targetColor
				}
			):Play()
		end

		Atualizar()

		if disparar ~= false then
			ExecutarCallback(callback, ativo)
		end
	end

	Atualizar()

	Conectar(
		hub._itemConnections,
		ToggleButton.MouseButton1Click,
		function()
			if clone.Parent then
				Set(not ativo, true)
			end
		end
	)

	return {
		Set = function(_, valor)
			Set(valor, true)
		end,

		Get = function()
			return ativo
		end,

		Object = clone,
		Button = ToggleButton,
	}
end

--==================================================
-- ADD SLIDER
--==================================================

function HubLibrary:AddSlider(
	texto,
	valorMin,
	valorMax,
	valorPadrao,
	callback
)
	if self._destroyed or not self.Scroll then
		return nil
	end

	local hub = self._hub or self

	valorMin = tonumber(valorMin) or 0
	valorMax = tonumber(valorMax) or 100
	valorPadrao = tonumber(valorPadrao) or valorMin

	if valorMax < valorMin then
		valorMin, valorMax = valorMax, valorMin
	end

	valorPadrao = math.clamp(
		valorPadrao,
		valorMin,
		valorMax
	)

	local valorAtual = valorPadrao

	local clone = hub:_CloneTemplate("SliderTemplate", self.Scroll)

	if not clone then
		return nil
	end

	local TitleLabel = clone:FindFirstChild("TitleLabel")
	local SliderTrack = clone:FindFirstChild("SliderTrack")

	if not SliderTrack then
		clone:Destroy()
		warn("[Frank Hub] SliderTrack nao encontrado.")
		return nil
	end

	local FillButton =
		SliderTrack:FindFirstChild("FillButton")

	local Progress =
		SliderTrack:FindFirstChild("Progress")

	if not FillButton
		or not FillButton:IsA("TextButton")
		or not Progress
		or not Progress:IsA("GuiObject") then

		clone:Destroy()
		warn("[Frank Hub] Slider incompleto.")
		return nil
	end

	local function Atualizar()
		local percentual = 0

		if valorMax ~= valorMin then
			percentual =
				(valorAtual - valorMin)
				/ (valorMax - valorMin)
		end

		percentual = math.clamp(percentual, 0, 1)

		Progress.Size = UDim2.new(
			percentual,
			0,
			Progress.Size.Y.Scale,
			Progress.Size.Y.Offset
		)

		FillButton.Position = UDim2.new(
			percentual,
			0,
			FillButton.Position.Y.Scale,
			FillButton.Position.Y.Offset
		)

		if TitleLabel and TitleLabel:IsA("TextLabel") then
			TitleLabel.Text =
				tostring(texto)
				.. " ("
				.. tostring(valorAtual)
				.. ")"
		end
	end

	local function DefinirPorX(x)
		if not SliderTrack.Parent then
			return
		end

		local largura =
			SliderTrack.AbsoluteSize.X

		if largura <= 0 then
			return
		end

		local percentual =
			(x - SliderTrack.AbsolutePosition.X)
			/ largura

		percentual =
			math.clamp(percentual, 0, 1)

		valorAtual = math.round(
			valorMin
			+ ((valorMax - valorMin) * percentual)
		)

		Atualizar()
		ExecutarCallback(callback, valorAtual)
	end

	local dragging = false

	Conectar(
		hub._itemConnections,
		SliderTrack.InputBegan,
		function(input)
			if input.UserInputType == Enum.UserInputType.MouseButton1
				or input.UserInputType == Enum.UserInputType.Touch then

				dragging = true
				DefinirPorX(input.Position.X)
			end
		end
	)

	Conectar(
		hub._itemConnections,
		FillButton.InputBegan,
		function(input)
			if input.UserInputType == Enum.UserInputType.MouseButton1
				or input.UserInputType == Enum.UserInputType.Touch then

				dragging = true
				DefinirPorX(input.Position.X)
			end
		end
	)

	Conectar(
		hub._itemConnections,
		UserInputService.InputEnded,
		function(input)
			if input.UserInputType == Enum.UserInputType.MouseButton1
				or input.UserInputType == Enum.UserInputType.Touch then

				dragging = false
			end
		end
	)

	Conectar(
		hub._itemConnections,
		UserInputService.InputChanged,
		function(input)
			if not dragging then
				return
			end

			if input.UserInputType ~= Enum.UserInputType.MouseMovement
				and input.UserInputType ~= Enum.UserInputType.Touch then
				return
			end

			DefinirPorX(input.Position.X)
		end
	)

	Atualizar()

	return {
		Set = function(_, valor)
			local numero = tonumber(valor)

			if numero == nil then
				return
			end

			valorAtual = math.clamp(
				math.round(numero),
				valorMin,
				valorMax
			)

			Atualizar()
			ExecutarCallback(callback, valorAtual)
		end,

		Get = function()
			return valorAtual
		end,

		Object = clone,
		Track = SliderTrack,
		FillButton = FillButton,
	}
end

--==================================================
-- ADD DROPDOWN
--==================================================

function HubLibrary:AddDropdown(
	texto,
	opcoes,
	callback,
	valorInicial
)
	if self._destroyed or not self.Scroll then
		return nil
	end

	local hub = self._hub or self

	opcoes =
		typeof(opcoes) == "table"
		and opcoes
		or {}

	local valorAtual = valorInicial

	if valorAtual == nil then
		if #opcoes > 0 then
			valorAtual = opcoes[1]
		else
			valorAtual = "Select Option..."
		end
	end

	local clone =
		hub:_CloneTemplate(
			"DropdownTemplate",
			self.Scroll
		)

	if not clone then
		return nil
	end

	clone.ClipsDescendants = false

	local TitleLabel = clone:FindFirstChild("TitleLabel")
	local DropdownButton =
		clone:FindFirstChild("Frame")

	if TitleLabel and TitleLabel:IsA("TextLabel") then
		TitleLabel.Text = tostring(texto)
	end

	if not DropdownButton
		or not DropdownButton:IsA("TextButton") then

		clone:Destroy()
		warn("[Frank Hub] Dropdown Button nao encontrado.")
		return nil
	end

	local SelectedLabel =
		DropdownButton:FindFirstChild("TitleLabel")

	local Arrow =
		DropdownButton:FindFirstChild("TextLabel")

	local Container = Instance.new("Frame")
	Container.Name = "ContainerOpcoes"
	Container.Size = UDim2.new(
		0,
		DropdownButton.Size.X.Offset,
		0,
		math.max(1, #opcoes * 30)
	)
	Container.Position = UDim2.new(
		0,
		DropdownButton.Position.X.Offset,
		0,
		DropdownButton.Position.Y.Offset
			+ DropdownButton.Size.Y.Offset
			+ 5
	)
	Container.BackgroundColor3 =
		Color3.fromRGB(25, 25, 25)
	Container.BorderSizePixel = 0
	Container.Visible = false
	Container.ZIndex = 100
	Container.Parent = clone

	CriarStroke(
		Container,
		"UIStroke",
		2,
		COR_BORDA,
		Enum.ApplyStrokeMode.Border,
		Enum.BorderStrokePosition.Outer
	)

	CriarCorner(Container, 5)

	local Layout =
		Instance.new("UIListLayout")

	Layout.Name = "UIListLayout"
	Layout.FillDirection =
		Enum.FillDirection.Vertical
	Layout.SortOrder =
		Enum.SortOrder.LayoutOrder
	Layout.Parent = Container

	local opened = false
	local originalSize = clone.Size

	local function AtualizarTexto()
		local current =
			valorAtual == nil
			and "Select Option..."
			or tostring(valorAtual)

		if SelectedLabel
			and SelectedLabel:IsA("TextLabel") then

			SelectedLabel.Text = current
		end

		if Arrow
			and Arrow:IsA("TextLabel") then

			Arrow.Text =
				opened and "v" or "^"
		end
	end

	local function Fechar()
		opened = false
		Container.Visible = false

		AtualizarTexto()

		TweenService:Create(
			clone,
			TweenInfo.new(
				0.16,
				Enum.EasingStyle.Quad,
				Enum.EasingDirection.Out
			),
			{
				Size = originalSize
			}
		):Play()
	end

	local function Abrir()
		if #opcoes == 0 then
			return
		end

		opened = true
		Container.Visible = true
		AtualizarTexto()

		TweenService:Create(
			clone,
			TweenInfo.new(
				0.16,
				Enum.EasingStyle.Quad,
				Enum.EasingDirection.Out
			),
			{
				Size = UDim2.new(
					originalSize.X.Scale,
					originalSize.X.Offset,
					0,
					originalSize.Y.Offset
						+ (#opcoes * 30)
						+ 10
				)
			}
		):Play()
	end

	AtualizarTexto()

	Conectar(
		hub._itemConnections,
		DropdownButton.MouseButton1Click,
		function()
			if opened then
				Fechar()
			else
				Abrir()
			end
		end
	)

	for indice, opcao in ipairs(opcoes) do
		local button =
			Instance.new("TextButton")

		button.Name =
			"Option" .. tostring(indice)

		button.LayoutOrder = indice
		button.Size =
			UDim2.new(1, 0, 0, 30)

		button.BackgroundColor3 =
			Color3.fromRGB(25, 25, 25)

		button.BorderSizePixel = 0
		button.Text = tostring(opcao)
		button.TextColor3 =
			Color3.fromRGB(255, 255, 255)
		button.TextSize = 16
		button.Font = Enum.Font.SourceSans
		button.ZIndex = 101
		button.AutoButtonColor = false
		button.Parent = Container

		Conectar(
			hub._itemConnections,
			button.MouseEnter,
			function()
				button.BackgroundColor3 =
					Color3.fromRGB(45, 45, 45)
			end
		)

		Conectar(
			hub._itemConnections,
			button.MouseLeave,
			function()
				button.BackgroundColor3 =
					Color3.fromRGB(25, 25, 25)
			end
		)

		Conectar(
			hub._itemConnections,
			button.MouseButton1Click,
			function()
				valorAtual = opcao
				Fechar()
				ExecutarCallback(
					callback,
					opcao
				)
			end
		)
	end

	return {
		Set = function(_, valor)
			valorAtual = valor
			AtualizarTexto()
			ExecutarCallback(
				callback,
				valorAtual
			)
		end,

		Get = function()
			return valorAtual
		end,

		Open = function()
			Abrir()
		end,

		Close = function()
			Fechar()
		end,

		Object = clone,
		Button = DropdownButton,
	}
end

	tituloCard,
	iconeId,
	listaDeTextos
)
	if self._destroyed or not self.Scroll then
		return nil
	end

	local hub = self._hub or self

	listaDeTextos =
		typeof(listaDeTextos) == "table"
		and listaDeTextos
		or {}

	local clone =
		hub:_CloneTemplate(
			"InfoCardTemplate",
			self.Scroll
		)

	if not clone then
		return nil
	end

	local TitleLabel =
		clone:FindFirstChild("TitleLabel")

	local Icon =
		clone:FindFirstChild("Icon")

	local ListInformations =
		clone:FindFirstChild(
			"ListInformations"
		)

	if TitleLabel
		and TitleLabel:IsA("TextLabel") then

		TitleLabel.Text =
			tostring(tituloCard)
	end

	if Icon
		and Icon:IsA("ImageLabel") then

		Icon.Image =
			tostring(iconeId or
				"rbxasset://textures/ui/GuiImagePlaceholder.png")
	end

	if ListInformations
		and ListInformations:IsA("GuiObject") then

		local template =
			ListInformations:FindFirstChild(
				"InformationTextTemplate"
			)

		if template
			and template:IsA("TextLabel") then

			template.Visible = false

			for _, child in ipairs(
				ListInformations:GetChildren()
			) do
				if child:IsA("TextLabel")
					and child ~= template then

					child:Destroy()
				end
			end

			for indice, texto in ipairs(
				listaDeTextos
			) do

				local info =
					template:Clone()

				info.Name =
					"InformationText"
					.. tostring(indice)

				info.Text =
					tostring(texto)

				info.LayoutOrder =
					indice

				info.Visible = true
				info.Parent =
					ListInformations
			end
		end
	end

	return {
		Object = clone
	}
end

-- DESTROY
--==================================================

function HubLibrary:Destroy()
	if self._destroyed then
		return
	end

	self._destroyed = true

	DesconectarLista(self._connections)
	DesconectarLista(self._itemConnections)

	for _, aba in ipairs(self.Abas) do
		if aba then
			aba._destroyed = true

			if aba.Scroll and aba.Scroll.Parent then
				aba.Scroll:Destroy()
			end

			if aba.Btn and aba.Btn.Parent then
				aba.Btn:Destroy()
			end
		end
	end

	self.Abas = {}
	self.AbaAtiva = nil

	-- Nao destroi a ScreenGui original de fora;
	-- esta instancia foi criada pela propria biblioteca.
	if self.ScreenGui and self.ScreenGui.Parent then
		self.ScreenGui:Destroy()
	end

	self.ScreenGui = nil
	self.BackGround = nil
	self.TemplateScrollingFrame = nil
	self.Templates = nil
	self.TabButtonsFrame = nil
	self.FixedButtonsFrame = nil
	self._connections = {}
	self._itemConnections = {}
end

local function CloneTheme(theme)
	local out = {}
	for key, value in pairs(theme) do
		out[key] = value
	end
	return out
end

local function NormalizeKey(key)
	if typeof(key) == "EnumItem" then
		if key.EnumType == Enum.KeyCode then
			return key
		end
		return nil
	end
	if type(key) == "string" then
		local value = Enum.KeyCode[key]
		return value
	end
	return nil
end

local function DeepCopy(value)
	if type(value) ~= "table" then
		return value
	end
	local copy = {}
	for key, item in pairs(value) do
		copy[key] = DeepCopy(item)
	end
	return copy
end

local function IsCallable(fn)
	return type(fn) == "function"
end

local function SafeJSONEncode(value)
	local ok, result = pcall(function()
		return HttpService:JSONEncode(value)
	end)
	if ok then
		return result
	end
	return nil, result
end

local function SafeJSONDecode(text)
	local ok, result = pcall(function()
		return HttpService:JSONDecode(text)
	end)
	if ok then
		return result
	end
	return nil, result
end

end

local function GetHub(self)
	return self and (self._hub or self)
end

local function EnsureEnhancementState(self)
	self._registry = self._registry or {}
	self._registryById = self._registryById or {}
	self._state = self._state or {}
	self._stateListeners = self._stateListeners or {}
	self._events = self._events or {}
	self._themes = self._themes or {}
	self._profiles = self._profiles or {}
	self._nextId = self._nextId or 0
	self._enhancementConnections = self._enhancementConnections or {}
	self._searchQuery = self._searchQuery or ""
	self._animationsEnabled = self._animationsEnabled ~= false
	self._responsiveEnabled = self._responsiveEnabled ~= false
	self._minimized = self._minimized == true
	self._currentProfile = self._currentProfile or "Default"
	self._autoSave = self._autoSave == true
	self._configFile = self._configFile or "FrankHub_Default.json"
	self._enhanced = true
	self._openPopup = self._openPopup or nil
	self._popupIgnoreUntil = self._popupIgnoreUntil or 0
	self._batchDepth = self._batchDepth or 0
	self._watermark = self._watermark or nil
end

local function NextId(self, prefix)
	self._nextId = (self._nextId or 0) + 1
	return string.format("%s_%04d", prefix or "Item", self._nextId)
end

local function ConnectEnhancement(self, signal, callback)
	if not signal then
		return nil
	end
	local ok, connection = pcall(function()
		return signal:Connect(callback)
	end)
	if ok and connection then
		table.insert(self._enhancementConnections, connection)
		return connection
	end
	return nil
end

local function DisconnectEnhancements(self)
	if not self._enhancementConnections then
		return
	end
	for _, connection in ipairs(self._enhancementConnections) do
		if typeof(connection) == "RBXScriptConnection" then
			pcall(function()
				connection:Disconnect()
			end)
		end
	end
	table.clear(self._enhancementConnections)
end

local function RegisterItem(self, control, kind, options)
	local hub = GetHub(self)
	if not hub or not control or not control.Object then
		return control
	end
	EnsureEnhancementState(hub)

	local entry = {
		Id = (options and options.Id) or NextId(hub, kind or "Item"),
		Kind = kind or "Item",
		Control = control,
		Object = control.Object,
		Title = options and options.Title or "",
		Description = options and options.Description or nil,
		Tab = self,
		Visible = true,
		Default = options and DeepCopy(options.Default) or nil,
		CreatedAt = os.clock(),
	}

	control.Id = entry.Id
	control.Kind = entry.Kind
	control.Title = entry.Title
	control.Description = entry.Description
	control._entry = entry

	table.insert(hub._registry, entry)
	hub._registryById[entry.Id] = entry
	return control
end

local function GetVisibleText(entry)
	local object = entry and entry.Object
	if not object or not object.Parent then
		return ""
	end
	local title = entry.Title or ""
	if object:IsA("TextButton") or object:IsA("TextLabel") then
		title = title ~= "" and title or object.Text
	end
	local titleLabel = object:FindFirstChild("TitleLabel", true)
	if titleLabel and titleLabel:IsA("TextLabel") then
		title = titleLabel.Text
	end
	return string.lower(tostring(title or ""))
end

local function RecalculateCanvas(hub, tab)
	if not tab then
		return
	end
	local page = tab.Scroll or tab.ScrollingFrame
	if not page or not page.Parent then
		return
	end
	local layout = page:FindFirstChildOfClass("UIListLayout")
	if layout then
		page.CanvasSize = UDim2.new(0, 0, 0, math.max(0, layout.AbsoluteContentSize.Y + 20))
	end
end

local function ApplySearch(hub)
	local query = string.lower(tostring(hub._searchQuery or "")):gsub("^%s+", ""):gsub("%s+$", "")
	local activeTab = hub.AbaAtiva
	for _, entry in ipairs(hub._registry or {}) do
		local object = entry.Object
		local sameTab = not activeTab or entry.Tab == activeTab
		local matched = true
		if query ~= "" then
			local haystack = GetVisibleText(entry)
			matched = string.find(haystack, query, 1, true) ~= nil
		end
		entry.Visible = matched
		if object and object.Parent then
			if sameTab then
				object.Visible = matched
			else
				object.Visible = true
			end
		end
	end
	RecalculateCanvas(hub, activeTab)
end

local function ThemeObject(hub, object, kind)
	if not object or not object.Parent then
		return
	end
	local theme = hub._theme
	if object:IsA("Frame") or object:IsA("TextButton") or object:IsA("TextLabel") then
		if kind == "Window" then
			object.BackgroundColor3 = theme.Background
			if object:IsA("Frame") then
				object.BackgroundTransparency = theme.BackgroundTransparency
			end
		elseif kind == "Top" then
			object.BackgroundColor3 = theme.Top
		elseif kind == "Card" then
			object.BackgroundColor3 = theme.Card
		elseif kind == "Control" then
			object.BackgroundColor3 = theme.Control
		end
	end
end

local function MergeTheme(base, custom)
	local merged = CloneTheme(base)
	for key, value in pairs(custom or {}) do
		merged[key] = value
	end
	return merged
end

function HubLibrary:RegisterTheme(name, theme)
	local hub = GetHub(self)
	if type(name) ~= "string" or type(theme) ~= "table" then
		return false
	end
	hub._themes[name] = MergeTheme(THEMES.Crimson, theme)
	return true
end

function HubLibrary:SetTheme(themeOrName)
	local hub = GetHub(self)
	if not hub then
		return false
	end
	local nextTheme
	if type(themeOrName) == "string" then
		nextTheme = hub._themes[themeOrName] or THEMES[themeOrName]
	else
		nextTheme = themeOrName
	end
	if type(nextTheme) ~= "table" then
		return false
	end
	hub._theme = MergeTheme(THEMES.Crimson, nextTheme)
	if type(themeOrName) == "string" then
		hub._themeName = themeOrName
	end
	
	ThemeObject(hub, hub.BackGround, "Window")
	local top = hub.BackGround and hub.BackGround:FindFirstChild("BackGroundTop")
	if top then ThemeObject(hub, top, "Top") end
	for _, entry in ipairs(hub._registry or {}) do
		local object = entry.Object
		ThemeObject(hub, object, "Card")
		local control = object and object:FindFirstChild("ToggleButton", true)
		if control then ThemeObject(hub, control, "Control") end
		local dropdown = object and object:FindFirstChild("Frame")
		if dropdown and dropdown:IsA("TextButton") then ThemeObject(hub, dropdown, "Control") end
		local track = object and object:FindFirstChild("SliderTrack", true)
		if track then ThemeObject(hub, track, "Control") end
		local button = object and object:IsA("TextButton") and object
		if button then
			button.TextColor3 = hub._theme.Text
		end
		local titleLabel = object and object:FindFirstChild("TitleLabel", true)
		if titleLabel and titleLabel:IsA("TextLabel") then
			titleLabel.TextColor3 = hub._theme.Text
		end
	end
	if hub._notificationHolder then
		for _, child in ipairs(hub._notificationHolder:GetChildren()) do
			if child:IsA("Frame") then
				child.BackgroundColor3 = hub._theme.Card
			end
		end
	end
	if hub._searchFrame then
		hub._searchFrame.BackgroundColor3 = hub._theme.Control
		local stroke = hub._searchFrame:FindFirstChildOfClass("UIStroke")
		if stroke then stroke.Color = hub._theme.Border end
	end
	if hub._searchInput then
		hub._searchInput.TextColor3 = hub._theme.Text
		hub._searchInput.PlaceholderColor3 = hub._theme.MutedText
	end
	if hub._minimizeButton then
		hub._minimizeButton.BackgroundColor3 = hub._theme.Accent
		hub._minimizeButton.TextColor3 = hub._theme.Text
	end
	if hub._tooltip then
		hub._tooltip.BackgroundColor3 = hub._theme.Tooltip
		if hub._tooltipText then hub._tooltipText.TextColor3 = hub._theme.Text end
	end
	if hub._uiScale then hub:SetResponsive(hub._responsiveEnabled) end
	hub:Fire("ThemeChanged", hub._theme)
	return true
end

function HubLibrary:GetTheme()
	local hub = GetHub(self)
	return hub and DeepCopy(hub._theme) or nil
end

function HubLibrary:On(eventName, callback)
	local hub = GetHub(self)
	if type(eventName) ~= "string" or not IsCallable(callback) then
		return nil
	end
	hub._events[eventName] = hub._events[eventName] or {}
	local entry = { Callback = callback, Connected = true }
	table.insert(hub._events[eventName], entry)
	local connection = {
		Disconnect = function()
			entry.Connected = false
		end,
	}
	return connection
end

function HubLibrary:Fire(eventName, ...)
	local hub = GetHub(self)
	local listeners = hub and hub._events and hub._events[eventName]
	if not listeners then
		return
	end
	for _, entry in ipairs(listeners) do
		if entry.Connected then
			ExecutarCallback(entry.Callback, ...)
		end
	end
end

function HubLibrary:GetState(key, fallback)
	local hub = GetHub(self)
	if hub._state[key] == nil then
		return fallback
	end
	return hub._state[key]
end

function HubLibrary:SetState(key, value, fireEvent)
	local hub = GetHub(self)
	local old = hub._state[key]
	hub._state[key] = value
	if fireEvent ~= false and old ~= value then
		local listeners = hub._stateListeners[key] or {}
		for _, callback in ipairs(listeners) do
			ExecutarCallback(callback, value, old, key)
		end
		hub:Fire("StateChanged", key, value, old)
	end
	return value
end

function HubLibrary:BindState(key, callback)
	local hub = GetHub(self)
	if type(key) ~= "string" or not IsCallable(callback) then
		return nil
	end
	hub._stateListeners[key] = hub._stateListeners[key] or {}
	table.insert(hub._stateListeners[key], callback)
	return {
		Disconnect = function()
			for index, fn in ipairs(hub._stateListeners[key] or {}) do
				if fn == callback then
					table.remove(hub._stateListeners[key], index)
					break
				end
			end
		end,
	}
end

function HubLibrary:SubscribeState(key, callback)
	return self:BindState(key, callback)
end

local function AddStateBinding(hub, control, key, fireCallback)
	if not hub or not control or not key then
		return
	end
	hub._state[key] = control.Get and control:Get() or hub._state[key]
	control.StateKey = key
	local originalSet = control.Set
	if type(originalSet) == "function" then
		control.Set = function(selfControl, value, silent)
			local result = originalSet(selfControl, value)
			hub:SetState(key, value, false)
			if fireCallback and not silent then
				hub:Fire("StateBindingChanged", key, value, control)
			end
			return result
		end
	end
	hub:BindState(key, function(value)
		if control.Get and control.Get() ~= value and type(originalSet) == "function" then
			originalSet(control, value)
		end
	end)
end

		if input.KeyCode == Enum.KeyCode.Unknown then return end
		key = input.KeyCode
		listening = false
		update()
		ExecutarCallback(callback, key)
		hub:Fire("KeybindChanged", tostring(texto), key)
	end)
	ConnectEnhancement(hub, UserInputService.InputBegan, function(input, processed)
		if hub._destroyed or processed then return end
		if input.UserInputType == Enum.UserInputType.Keyboard and input.KeyCode == key then
			ExecutarCallback(callback, key)
			hub:Fire("KeybindPressed", tostring(texto), key)
		end
	end)

	local result = {
		Object = object,
		Button = keyButton,
		Get = function() return key end,
		Set = function(_, newKey)
			local normalized = NormalizeKey(newKey)
			if normalized then
				key = normalized
				update()
			end
			return key
		end,
	}
	RegisterItem(self, result, "Keybind", {
		Title = texto,
		Description = descricao,
		Default = key.Name,
	})
	if descricao then self:AddTooltip(result, descricao) end
	return result
end

function HubLibrary:Snapshot()
	local hub = GetHub(self)
	local data = {
		Version = 3,
		Theme = hub._themeName or "Crimson",
		State = DeepCopy(hub._state),
		Values = {},
	}
	for _, entry in ipairs(hub._registry or {}) do
		local control = entry.Control
		if control and type(control.Get) == "function" then
			local ok, value = pcall(function() return control:Get() end)
			if ok then
				data.Values[entry.Id] = value
			end
		end
	end
	return data
end

function HubLibrary:Restore(snapshot, silent)
	local hub = GetHub(self)
	if type(snapshot) ~= "table" then return false end
	if snapshot.State then
		hub._state = DeepCopy(snapshot.State)
	end
	for id, value in pairs(snapshot.Values or {}) do
		local entry = hub._registryById[id]
		if entry and entry.Control and type(entry.Control.Set) == "function" then
			pcall(function()
				entry.Control:Set(value, silent)
			end)
		end
	end
	if snapshot.Theme then
		hub:SetTheme(snapshot.Theme)
	end
	hub:Refresh()
	hub:Fire("ConfigRestored", snapshot)
	return true
end

function HubLibrary:ExportConfig(pretty)
	local payload = self:Snapshot()
	local encoded, err = SafeJSONEncode(payload)
	if not encoded then
		warn("[Frank Hub] Falha ao exportar config:", err)
		return nil
	end
	if pretty then
		return encoded
	end
	return encoded
end

function HubLibrary:ImportConfig(json, silent)
	local data, err = SafeJSONDecode(tostring(json or ""))
	if not data then
		warn("[Frank Hub] JSON invalido:", err)
		return false
	end
	return self:Restore(data, silent)
end

function HubLibrary:SetConfigFile(fileName)
	local hub = GetHub(self)
	local safe = tostring(fileName or "FrankHub_Default.json")
	safe = safe:gsub("[^%w%._%-]", "_")
	if not safe:lower():match("%.json$") then
		safe = safe .. ".json"
	end
	hub._configFile = safe
	return safe
end

function HubLibrary:SaveConfig(fileName)
	local hub = GetHub(self)
	if fileName then hub:SetConfigFile(fileName) end
	if not SafeFileAvailable() then
		return false, "writefile/isfile/readfile indisponivel neste executor"
	end
	local payload = self:ExportConfig()
	if not payload then return false, "falha no JSON" end
	local ok, err = pcall(function()
		writefile(hub._configFile, payload)
	end)
	if not ok then
		return false, err
	end
	hub:Notify("Config", "Config salva: " .. hub._configFile, 2.5, "Success")
	hub:Fire("ConfigSaved", hub._configFile)
	return true
end

function HubLibrary:LoadConfig(fileName, silent)
	local hub = GetHub(self)
	if fileName then hub:SetConfigFile(fileName) end
	if not SafeFileAvailable() then
		return false, "readfile/isfile indisponivel neste executor"
	end
	if not isfile(hub._configFile) then
		return false, "arquivo nao encontrado"
	end
	local ok, contents = pcall(function()
		return readfile(hub._configFile)
	end)
	if not ok then return false, contents end
	local restored = self:ImportConfig(contents, silent)
	if restored then
		hub:Notify("Config", "Config carregada: " .. hub._configFile, 2.5, "Success")
		hub:Fire("ConfigLoaded", hub._configFile)
	end
	return restored
end

		TweenInfo.new(
			duration,
			easingStyle or Enum.EasingStyle.Quad,

		properties or {}

	tween:Play()
	return tween

function HubLibrary:SetResponsive(enabled)
	local hub = GetHub(self)
	hub._responsiveEnabled = enabled ~= false
	local scale = hub._uiScale
	if not scale then
		return hub._responsiveEnabled
	end
	local viewport = workspace.CurrentCamera and workspace.CurrentCamera.ViewportSize or Vector2.new(1920, 1080)
	local factor = math.min(viewport.X / 1920, viewport.Y / 1080)
	if hub._responsiveEnabled then
		factor = math.clamp(factor, 0.72, 1.1)
	else
		factor = 1
	end
	scale.Scale = factor
	return factor
end

function HubLibrary:Minimizar(force)
	local hub = GetHub(self)
	if not hub.BackGround then return false end
	local minimize = force
	if minimize == nil then minimize = not hub._minimized end
	hub._minimized = minimize == true
	for _, tab in ipairs(hub.Abas or {}) do
		if tab.Scroll and tab.Scroll.Parent then
			tab.Scroll.Visible = not hub._minimized and tab == hub.AbaAtiva
		end
	end
	if hub._searchFrame then
		hub._searchFrame.Visible = (not hub._minimized) and (hub._searchEnabled ~= false)
	end
	if hub._minimizeButton then
		hub._minimizeButton.Text = hub._minimized and "+" or "_"
	end
	hub:Fire("MinimizedChanged", hub._minimized)
	return hub._minimized
end

		hub:SetResponsive(hub._responsiveEnabled)

			hub:SetResponsive(hub._responsiveEnabled)

	hub:SetTheme("Crimson")

	hub:Fire("Loaded", nomeDoHub)
	return hub

function HubLibrary.Carregar(nomeDoHub)
	if HubLibrary._Active and not HubLibrary._Active._destroyed then
		pcall(function()
			HubLibrary._Active:Destroy()
		end)
		HubLibrary._Active = nil
	end

	local hub = _OriginalCarregar(nomeDoHub)
	if not hub then
		return nil
	end

	EnhanceAfterLoad(hub, nomeDoHub)

	-- A interface original do RBXM ja possuia uma aba Farm.
	-- Criamos a aba automaticamente no modo externo para que
	-- a biblioteca abra completa mesmo quando nenhum script de
	-- exemplo foi executado depois do loader.
	if #(hub.Abas or {}) == 0 then
		local farm = hub:AddTab("Farm")
		if farm then
			farm:AddInfoCard(
				"Egg Predictor",
				"rbxasset://textures/ui/GuiImagePlaceholder.png",
				{"3.03x", "Predictor ativo"}
			)
			farm:AddToggle("Auto Steal", nil, false)
			farm:AddDropdown(
				"Target Areas",
				{"Area 1", "Area 2", "Area 3"},
				nil,
				"Area 1"
			)
			farm:AddSlider("Min Value To Steal", 0, 100, 25, nil)
			hub:SelecionarAba(farm)
		end
	end

	hub:Refresh()
	HubLibrary._Active = hub
	return hub
end

function HubLibrary:SelecionarAba(aba)
	local hub = GetHub(self)
	local previous = hub.AbaAtiva
	_OriginalSelecionarAba(hub, aba)
	if not aba then return end
	if hub._searchInput then
		hub._searchInput.Text = hub._searchQuery or ""
	end
	ApplySearch(hub)
	if aba.Scroll and previous ~= aba and hub._animationsEnabled then
		local targetPos = aba.Scroll.Position
		aba.Scroll.Position = targetPos + UDim2.fromOffset(16, 0)
		local tween = TweenService:Create(aba.Scroll, TweenInfo.new(0.18, Enum.EasingStyle.Quint, Enum.EasingDirection.Out), {
			Position = targetPos,
		})
		tween:Play()
	end
	hub:Fire("TabChanged", aba, previous)
end

function HubLibrary:Validate()
	local hub = GetHub(self)
	local issues = {}
	for _, entry in ipairs(hub._registry or {}) do
		if not entry.Object or not entry.Object.Parent then
			table.insert(issues, "Item removido: " .. tostring(entry.Id))
		end
	end
	if #issues > 0 then
		warn("[Frank Hub] Validate encontrou " .. tostring(#issues) .. " problema(s).")
	end
	return #issues == 0, issues
end

function HubLibrary:CleanupRegistry()
	local hub = GetHub(self)
	local cleaned = {}
	for _, entry in ipairs(hub._registry or {}) do
		if entry.Object and entry.Object.Parent then
			table.insert(cleaned, entry)
		else
			hub._registryById[entry.Id] = nil
		end
	end
	hub._registry = cleaned
	return #cleaned
end

function HubLibrary.Unload()
	local hub = HubLibrary._Active
	if hub and not hub._destroyed then
		hub:Destroy()
	end
	HubLibrary._Active = nil
	return true
end

function HubLibrary:Destroy()
	local hub = GetHub(self)
	DisconnectEnhancements(hub)
	if hub._tooltip and hub._tooltip.Parent then hub._tooltip:Destroy() end
	if hub._notificationHolder and hub._notificationHolder.Parent then hub._notificationHolder:Destroy() end
	hub._events = {}
	hub._stateListeners = {}
	hub._registry = {}
	hub._registryById = {}
	if HubLibrary._Active == hub then
		HubLibrary._Active = nil
	end
	_OriginalDestroy(hub)
end

local function V6Hub(self) return self and (self._hub or self) end
local function V6OpenPopup(hub, object, closeFn)
	hub._openPopup={Object=object,Close=closeFn}
	hub._popupIgnoreUntil=os.clock()+0.22
end
local function V6ClosePopup(hub)
	local popup=hub._openPopup
	hub._openPopup=nil
	if popup and type(popup.Close)=="function" then pcall(popup.Close) end
end
local function V6InstallInputManager(hub)
	if hub._v6InputInstalled then return end
	hub._v6InputInstalled=true
	ConnectEnhancement(hub,UserInputService.InputBegan,function(input,processed)
		if hub._destroyed or processed then return end
		if input.UserInputType~=Enum.UserInputType.MouseButton1 and input.UserInputType~=Enum.UserInputType.Touch then return end
		local popup=hub._openPopup
		if not popup or not popup.Object or not popup.Object.Parent then hub._openPopup=nil; return end
		if os.clock()<(hub._popupIgnoreUntil or 0) then return end
		local p=input.Position
		local inside=false
		local ok,objects=pcall(function() return GuiService:GetGuiObjectsAtPosition(p.X,p.Y) end)
		if ok and type(objects)=="table" then
			for _,obj in ipairs(objects) do
				if obj==popup.Object or obj:IsDescendantOf(popup.Object) then inside=true; break end
			end
		end
		if not inside then
			local pos,size=popup.Object.AbsolutePosition,popup.Object.AbsoluteSize
			inside=p.X>=pos.X and p.X<=pos.X+size.X and p.Y>=pos.Y and p.Y<=pos.Y+size.Y
		end
		if not inside then V6ClosePopup(hub) end
	end)
end
local function V6ButtonFX(hub,button,normal,hover,pressed)
	if not button or not button:IsA("GuiButton") then return end
	button.AutoButtonColor=false
	ConnectEnhancement(hub,button.MouseEnter,function() if not button:GetAttribute("FrankDisabled") then hub:Tween(button,{BackgroundColor3=hover},0.10) end end)
	ConnectEnhancement(hub,button.MouseLeave,function() if not button:GetAttribute("FrankDisabled") then hub:Tween(button,{BackgroundColor3=normal},0.10) end end)
	ConnectEnhancement(hub,button.MouseButton1Down,function() if not button:GetAttribute("FrankDisabled") then hub:Tween(button,{BackgroundColor3=pressed},0.06) end end)
	ConnectEnhancement(hub,button.MouseButton1Up,function() if not button:GetAttribute("FrankDisabled") then hub:Tween(button,{BackgroundColor3=hover},0.06) end end)
end

local function V6BuildDropdown(self,texto,opcoes,callback,valorInicial)
	if self._destroyed or not self.Scroll then return nil end
	local hub=V6Hub(self)
	opcoes=type(opcoes)=="table" and opcoes or {}
	local value=valorInicial
	if value==nil then value=opcoes[1] or "Select Option..." end
	local clone=hub:_CloneTemplate("DropdownTemplate",self.Scroll)
	if not clone then return nil end
	clone.ClipsDescendants=false
	local title=clone:FindFirstChild("TitleLabel")
	local button=clone:FindFirstChild("Frame")
	if title and title:IsA("TextLabel") then title.Text=tostring(texto) end
	if not button or not button:IsA("TextButton") then clone:Destroy(); return nil end
	local selected=button:FindFirstChild("TitleLabel")
	local arrow=button:FindFirstChild("TextLabel")
	local baseSize=clone.Size
	local opened=false
	local container=Instance.new("Frame")
	container.Name="ContainerOpcoes"
	container.Position=UDim2.new(0,button.Position.X.Offset,0,button.Position.Y.Offset+button.Size.Y.Offset+5)
	container.Size=UDim2.new(0,button.Size.X.Offset,0,math.max(30,#opcoes*30+6))
	container.BackgroundColor3=hub._theme and hub._theme.Control or Color3.fromRGB(20,20,20)
	container.BorderSizePixel=0
	container.Visible=false
	container.ZIndex=1000
	container.Active=true
	container.Parent=clone
	CriarCorner(container,6)
	CriarStroke(container,"UIStroke",2,hub._theme and hub._theme.Border or COR_BORDA,Enum.ApplyStrokeMode.Border,Enum.BorderStrokePosition.Outer)
	local layout=Instance.new("UIListLayout")
	layout.SortOrder=Enum.SortOrder.LayoutOrder
	layout.Padding=UDim.new(0,1)
	layout.Parent=container
	local padding=Instance.new("UIPadding")
	padding.PaddingTop=UDim.new(0,3); padding.PaddingBottom=UDim.new(0,3); padding.Parent=container
	local items={}
	local function render()
		if selected and selected:IsA("TextLabel") then selected.Text=tostring(value) end
		if arrow and arrow:IsA("TextLabel") then arrow.Text=opened and "v" or "^" end
		for _,item in ipairs(items) do
			item.Button.BackgroundColor3=(tostring(item.Value)==tostring(value)) and (hub._theme and hub._theme.Accent or COR_TAB_ATIVA) or (hub._theme and hub._theme.Control or Color3.fromRGB(25,25,25))
		end
	end
	local function close()
		opened=false
		if hub._openPopup and hub._openPopup.Object==container then hub._openPopup=nil end
		container.Visible=false
		render()
		hub:Tween(clone,{Size=baseSize},0.14)
	end
	local function open()
		if #opcoes==0 then return end
		if hub._openPopup and hub._openPopup.Object~=container then V6ClosePopup(hub) end
		opened=true; container.Visible=true; V6OpenPopup(hub,container,close); render()
		hub:Tween(clone,{Size=UDim2.new(baseSize.X.Scale,baseSize.X.Offset,0,baseSize.Y.Offset+math.min(#opcoes,8)*30+14)},0.14)
		task.delay(0.18,function() if opened then hub._popupIgnoreUntil=0 end end)
	end
	for i,option in ipairs(opcoes) do
		local b=Instance.new("TextButton")
		b.Name="Option"..i; b.LayoutOrder=i; b.Size=UDim2.new(1,-6,0,29); b.BackgroundColor3=hub._theme and hub._theme.Control or Color3.fromRGB(25,25,25); b.BorderSizePixel=0; b.Text=tostring(option); b.TextColor3=hub._theme and hub._theme.Text or COR_TEXTO; b.TextSize=15; b.Font=Enum.Font.SourceSans; b.TextXAlignment=Enum.TextXAlignment.Left; b.AutoButtonColor=false; b.Active=true; b.ZIndex=1001; b.Parent=container
		local pad=Instance.new("UIPadding"); pad.PaddingLeft=UDim.new(0,9); pad.Parent=b; CriarCorner(b,4)
		local item={Button=b,Value=option}; table.insert(items,item)
		V6ButtonFX(hub,b,b.BackgroundColor3,hub._theme and hub._theme.ControlHover or Color3.fromRGB(45,45,45),hub._theme and hub._theme.AccentHover or COR_TAB_ATIVA)
		Conectar(hub._itemConnections,b.Activated,function()
			if not opened or b:GetAttribute("FrankDisabled") then return end
			value=option; render(); ExecutarCallback(callback,value); close()
		end)
	end
	Conectar(hub._itemConnections,button.Activated,function() if button:GetAttribute("FrankDisabled") then return end; if opened then close() else open() end end)
	render()
	return {Set=function(_,v,silent) value=v; render(); if silent~=true then ExecutarCallback(callback,value) end end,Get=function() return value end,Open=open,Close=close,Toggle=function() if opened then close() else open() end end,IsOpen=function() return opened end,Object=clone,Button=button,Options=opcoes}
end

function HubLibrary:AddMultiDropdown(texto,opcoes,callback,valorInicial,descricao,stateKey)
	local tab=self; local hub=V6Hub(tab)
	if tab._destroyed or not tab.Scroll then return nil end
	opcoes=type(opcoes)=="table" and opcoes or {}
	local selected={}
	if type(valorInicial)=="table" then for _,v in ipairs(valorInicial) do selected[tostring(v)]=v end end
	local clone=hub:_CloneTemplate("DropdownTemplate",tab.Scroll); if not clone then return nil end
	clone.ClipsDescendants=false
	local title=clone:FindFirstChild("TitleLabel"); local button=clone:FindFirstChild("Frame")
	if title and title:IsA("TextLabel") then title.Text=tostring(texto) end
	if not button or not button:IsA("TextButton") then clone:Destroy(); return nil end

	local layout=Instance.new("UIListLayout"); layout.SortOrder=Enum.SortOrder.LayoutOrder; layout.Padding=UDim.new(0,1); layout.Parent=container
	local padding=Instance.new("UIPadding"); padding.PaddingTop=UDim.new(0,3); padding.PaddingBottom=UDim.new(0,3); padding.Parent=container
	local items={}
	local function values() local out={}; for _,v in pairs(selected) do table.insert(out,v) end; table.sort(out,function(a,b)return tostring(a)<tostring(b) end); return out end
	local function render() local out=values(); if label and label:IsA("TextLabel") then label.Text=#out==0 and "Select..." or (#out==1 and tostring(out[1]) or tostring(#out).." selecionados") end; if arrow and arrow:IsA("TextLabel") then arrow.Text=opened and "v" or "^" end; for _,it in ipairs(items) do local on=selected[tostring(it.Value)]~=nil; it.Button.BackgroundColor3=on and (hub._theme and hub._theme.Accent or COR_TAB_ATIVA) or (hub._theme and hub._theme.Control or Color3.fromRGB(25,25,25)); it.Mark.Text=on and "✓" or "" end end
	local function close() opened=false; if hub._openPopup and hub._openPopup.Object==container then hub._openPopup=nil end; container.Visible=false; render(); hub:Tween(clone,{Size=baseSize},0.14) end
	local function open() if #opcoes==0 then return end; if hub._openPopup and hub._openPopup.Object~=container then V6ClosePopup(hub) end; opened=true; container.Visible=true; V6OpenPopup(hub,container,close); render(); hub:Tween(clone,{Size=UDim2.new(baseSize.X.Scale,baseSize.X.Offset,0,baseSize.Y.Offset+math.min(#opcoes,8)*30+14)},0.14); task.delay(0.18,function() if opened then hub._popupIgnoreUntil=0 end end) end
	for i,option in ipairs(opcoes) do
		local b=Instance.new("TextButton"); b.Name="Option"..i; b.LayoutOrder=i; b.Size=UDim2.new(1,-6,0,29); b.BackgroundColor3=hub._theme and hub._theme.Control or Color3.fromRGB(25,25,25); b.BorderSizePixel=0; b.Text=tostring(option); b.TextColor3=hub._theme and hub._theme.Text or COR_TEXTO; b.TextSize=15; b.Font=Enum.Font.SourceSans; b.TextXAlignment=Enum.TextXAlignment.Left; b.AutoButtonColor=false; b.Active=true; b.ZIndex=1001; b.Parent=container
		local pad=Instance.new("UIPadding"); pad.PaddingLeft=UDim.new(0,9); pad.Parent=b; CriarCorner(b,4)
		local mark=Instance.new("TextLabel"); mark.Name="Check"; mark.BackgroundTransparency=1; mark.Position=UDim2.new(1,-28,0,0); mark.Size=UDim2.fromOffset(24,29); mark.TextSize=17; mark.TextColor3=hub._theme and hub._theme.Text or COR_TEXTO; mark.ZIndex=1002; mark.Parent=b
		local item={Button=b,Value=option,Mark=mark}; table.insert(items,item); V6ButtonFX(hub,b,b.BackgroundColor3,hub._theme and hub._theme.ControlHover or Color3.fromRGB(45,45,45),hub._theme and hub._theme.AccentHover or COR_TAB_ATIVA)
		Conectar(hub._itemConnections,b.Activated,function() if not opened or b:GetAttribute("FrankDisabled") then return end; local k=tostring(option); if selected[k]~=nil then selected[k]=nil else selected[k]=option end; render(); ExecutarCallback(callback,values()) end)

	if stateKey then control.StateKey=tostring(stateKey); hub:SetState(tostring(stateKey),values(),false) end
	return control
end

	local hub=V6Hub(self); local needle=tostring(idOrTitle or ""):lower()
	for _,e in ipairs(hub._registry or {}) do if tostring(e.Id):lower()==needle or tostring(e.Title):lower()==needle then return e.Control end end
	return nil
end

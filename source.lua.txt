Tab.__index = function(_, key)

	if own ~= nil then
		return own
	end

		return function(selfTab, ...)
			return hubMethod(selfTab._hub, ...)
		end
	end
	return nil
end

local function MergeTheme(base, custom)
	local merged = {}
	for key, value in pairs(base) do
		merged[key] = value
	end
	for key, value in pairs(custom or {}) do
		merged[key] = value
	end
	return merged
end

-- Temas custom precisam virar JSON para entrar no config.
local function SerializeTheme(theme)
	local out = {}
	for key, value in pairs(theme) do
		if typeof(value) == "Color3" then
			out[key] = {
				math.floor(value.R * 255 + 0.5),
				math.floor(value.G * 255 + 0.5),
				math.floor(value.B * 255 + 0.5),
			}
		elseif type(value) == "number" then
			out[key] = value
		end
	end
	return out
end

local function DeserializeTheme(data)
	local out = {}
	if type(data) ~= "table" then
		return out
	end
	for key, value in pairs(data) do
		if THEMES.Crimson[key] ~= nil then
			if type(value) == "table" and #value == 3 then
				out[key] = Color3.fromRGB(
					tonumber(value[1]) or 0,
					tonumber(value[2]) or 0,
					tonumber(value[3]) or 0
				)
			elseif type(value) == "number" then
				out[key] = value
			end
		end
	end
	return out
end

local function New(className, props, parent)
	local inst = Instance.new(className)
	if props then
		for key, value in pairs(props) do
			inst[key] = value
		end
	end
	if parent then
		inst.Parent = parent
	end
	return inst
end

local function Corner(parent, radius)
	return New("UICorner", { CornerRadius = UDim.new(0, radius or 5) }, parent)
end

local function Stroke(parent, thickness, color)
	local stroke = New("UIStroke", {
		Thickness = thickness or 2,
		Color = color or Color3.new(0, 0, 0),
		ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
	})
	pcall(function()
		stroke.BorderStrokePosition = Enum.BorderStrokePosition.Outer
	end)
	stroke.Parent = parent
	return stroke
end

local function Traceback(err)
	return tostring(err) .. "\n" .. debug.traceback("", 2)
end

-- Execucao sincrona protegida (xpcall + traceback).
local function SafeCall(callback, ...)
	if type(callback) ~= "function" then
		return false
	end
	local ok, err = xpcall(callback, Traceback, ...)
	if not ok then
		warn("[Frank Hub] Callback:", err)
	end
	return ok, err
end

-- Execucao assincrona protegida (callbacks do usuario).
local function Call(callback, ...)
	if type(callback) ~= "function" then
		return
	end
	local args = table.pack(...)
	task.spawn(function()
		SafeCall(callback, table.unpack(args, 1, args.n))
	end)
end

local function DeepCopy(value)
	if type(value) ~= "table" then
		return value
	end
	local copy = {}
	for key, item in pairs(value) do
		copy[key] = DeepCopy(item)
	end
	return copy
end

local function CopyList(list)
	local out = {}
	if type(list) == "table" then
		for _, value in ipairs(list) do
			table.insert(out, value)
		end
	end
	return out
end

-- Primeiro campo nao-nil da tabela (respeita false).
local function Pick(tbl, ...)
	for _, key in ipairs({ ... }) do
		if tbl[key] ~= nil then
			return tbl[key]
		end
	end
	return nil
end

local function IsJSONSafe(value, depth)
	depth = depth or 0
	if depth > 6 then
		return false
	end
	local kind = type(value)
	if kind == "boolean" or kind == "string" then
		return true
	end
	if kind == "number" then
		return value == value and value ~= math.huge and value ~= -math.huge
	end
	if kind == "table" then
		for k, v in pairs(value) do
			if type(k) ~= "string" and type(k) ~= "number" then
				return false
			end
			if not IsJSONSafe(v, depth + 1) then
				return false
			end
		end
		return true
	end
	return false
end

local function Slug(text)
	local s = string.lower(tostring(text))
	s = s:gsub("[^%w]+", "_")
	s = s:gsub("^_+", "")
	s = s:gsub("_+$", "")
	if s == "" then
		s = "item"
	end
	return s
end

local function NormalizeKey(key)
	if typeof(key) == "EnumItem" then
		if key.EnumType == Enum.KeyCode then
			return key
		end
		return nil
	end
	if type(key) == "string" then
		local ok, result = pcall(function()
			return Enum.KeyCode[key]
		end)
		if ok and typeof(result) == "EnumItem" then
			return result
		end
	end
	return nil
end

local function ToImage(id)
	if type(id) == "number" then
		return "rbxassetid://" .. tostring(id)
	end
	id = tostring(id or "")
	if id == "" then
		return "rbxasset://textures/ui/GuiImagePlaceholder.png"
	end
	if id:match("^%d+$") then
		return "rbxassetid://" .. id
	end
	return id
end

		end)
	end
	table.clear(list)
end

local function ToHex(color)
	return string.format(
		"#%02X%02X%02X",
		math.floor(color.R * 255 + 0.5),
		math.floor(color.G * 255 + 0.5),
		math.floor(color.B * 255 + 0.5)
	)
end

local function ParseColor(value)
	if typeof(value) == "Color3" then
		return value
	end
	if type(value) == "string" then
		local r, g, b = value:match("^#?(%x%x)(%x%x)(%x%x)$")
		if r then
			return Color3.fromRGB(tonumber(r, 16), tonumber(g, 16), tonumber(b, 16))
		end
	end
	if type(value) == "table" and #value == 3 then
		return Color3.fromRGB(
			math.clamp(tonumber(value[1]) or 0, 0, 255),
			math.clamp(tonumber(value[2]) or 0, 0, 255),
			math.clamp(tonumber(value[3]) or 0, 0, 255)
		)
	end
	return nil
end

function HubLibrary:_Themed(object, applyFn)
	table.insert(self._themed, { Object = object, Apply = applyFn })
	applyFn(self._theme)
	return object
end

function HubLibrary:_T(object, property, themeKey)
	return self:_Themed(object, function(theme)
		object[property] = theme[themeKey]
	end)
end

-- Remove entradas de objetos destruidos; opcionalmente reaplica o tema.
function HubLibrary:_PruneThemed(apply)
	if self._destroyed or not self.ScreenGui then
		return
	end
	local alive = {}
	for _, entry in ipairs(self._themed) do
		if entry.Object:IsDescendantOf(self.ScreenGui) then
			table.insert(alive, entry)
			if apply then
				local ok, err = xpcall(entry.Apply, Traceback, self._theme)
				if not ok then
					warn("[Frank Hub] Tema:", err)
				end
			end
		end
	end
	self._themed = alive
end

function HubLibrary:_ApplyTheme()
	self:_PruneThemed(true)
end

-- owner (opcional): tabela de controle/aba que guarda a conexao em
-- owner._conns. Assim ela e desconectada quando o dono for destruido
-- e NAO fica acumulada em hub._conns.
function HubLibrary:_Conn(signal, callback, owner)
	local connection = signal:Connect(callback)
	if owner then
		owner._conns = owner._conns or {}
		table.insert(owner._conns, connection)
	else
		table.insert(self._conns, connection)
	end
	return connection
end

function HubLibrary:_Hover(button, normalKey, hoverKey, owner)
	self:_Conn(button.MouseEnter, function()
		self:Tween(button, { BackgroundColor3 = self._theme[hoverKey] }, 0.1)
	end, owner)
	self:_Conn(button.MouseLeave, function()
		self:Tween(button, { BackgroundColor3 = self._theme[normalKey] }, 0.1)
	end, owner)
end

function HubLibrary:RegisterTheme(name, theme)
	if type(name) ~= "string" or type(theme) ~= "table" then
		return false
	end
	self._themes[name] = MergeTheme(THEMES.Crimson, theme)
	return true
end

function HubLibrary:GetThemes()
	local names = {}
	for name in pairs(self._themes) do
		table.insert(names, name)
	end
	table.sort(names)
	return names
end

function HubLibrary:SetTheme(themeOrName)
	if self._destroyed then
		return false
	end
	local nextTheme, name
	if type(themeOrName) == "string" then
		nextTheme = self._themes[themeOrName]
		name = themeOrName
	elseif type(themeOrName) == "table" then
		nextTheme = themeOrName
	end
	if type(nextTheme) ~= "table" then
		return false
	end
	self._theme = MergeTheme(THEMES.Crimson, nextTheme)
	self._themeName = name or "Custom"
	self:_ApplyTheme()
	self:Fire("ThemeChanged", self._theme, self._themeName)
	return true
end

function HubLibrary:GetTheme()
	return DeepCopy(self._theme)
end

--==================================================
-- EVENTOS (assincronos, via Call)
--==================================================

function HubLibrary:On(eventName, callback)
	if type(eventName) ~= "string" or type(callback) ~= "function" then
		return nil
	end
	local list = self._events[eventName]
	if not list then
		list = {}
		self._events[eventName] = list
	end
	local entry = { Callback = callback }
	table.insert(list, entry)
	return {
		Disconnect = function()
			local index = table.find(list, entry)
			if index then
				table.remove(list, index)
			end
		end,
	}
end

function HubLibrary:Fire(eventName, ...)
	local list = self._events and self._events[eventName]
	if not list then
		return
	end
	for _, entry in ipairs(table.clone(list)) do
		Call(entry.Callback, ...)
	end
end

--==================================================
-- STATE (listeners sincronos, protegidos com xpcall)
--==================================================

function HubLibrary:GetState(key, fallback)
	local value = self._state[key]
	if value == nil then
		return fallback
	end
	return value
end

function HubLibrary:SetState(key, value, fireEvent)
	local old = self._state[key]
	self._state[key] = value
	if old ~= value and fireEvent ~= false then
		for _, listener in ipairs(table.clone(self._stateListeners[key] or {})) do
			SafeCall(listener, value, old, key)
		end
		self:Fire("StateChanged", key, value, old)
	end
	return value
end

function HubLibrary:BindState(key, callback)
	if type(key) ~= "string" or type(callback) ~= "function" then
		return nil
	end
	local list = self._stateListeners[key]
	if not list then
		list = {}
		self._stateListeners[key] = list
	end
	table.insert(list, callback)
	return {
		Disconnect = function()
			local index = table.find(list, callback)
			if index then
				table.remove(list, index)
			end
		end,
	}
end

function HubLibrary:SubscribeState(key, callback)
	return self:BindState(key, callback)
end

function HubLibrary:SetAnimationsEnabled(enabled)
	self._animations = enabled ~= false
	return self._animations
end

function HubLibrary:Tween(object, properties, duration, easingStyle, easingDirection)
	if not object or not object.Parent then
		return nil
	end
	if self._animations == false then
		for property, value in pairs(properties or {}) do
			pcall(function()
				object[property] = value
			end)
		end
		return nil
	end
	local tween = TweenService:Create(
		object,
		TweenInfo.new(
			tonumber(duration) or 0.2,
			easingStyle or Enum.EasingStyle.Quad,
			easingDirection or Enum.EasingDirection.Out
		),
		properties or {}
	)
	tween:Play()
	return tween
end

		card.Position = UDim2.new(0, 0, 0, 0)
	end

	local function remove()
		if wrap.Parent then
			wrap:Destroy()
		end
		self:_PruneThemed(false)
	end

		if not wrap.Parent then
			self:_PruneThemed(false)
			return
		end
		local tween = self:Tween(card, { Position = UDim2.new(1, 40, 0, 0) }, 0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.In)
		if tween then
			tween.Completed:Connect(remove)
		else
			remove()
		end
	end)

	return card

-- TOOLTIP
--==================================================

function HubLibrary:_PlaceTooltip()
	local tooltip = self._tooltip
	if not tooltip or not self.ScreenGui or not tooltip.Visible then
		return
	end
	local mouse = UserInputService:GetMouseLocation()
	local view = self.ScreenGui.AbsoluteSize
	local size = tooltip.AbsoluteSize
	local x = mouse.X + 14
	local y = mouse.Y + 14
	if x + size.X > view.X then
		x = math.max(0, mouse.X - size.X - 8)
	end
	if y + size.Y > view.Y then
		y = math.max(0, mouse.Y - size.Y - 8)
	end
	tooltip.Position = UDim2.fromOffset(x, y)
end

function HubLibrary:AddTooltip(object, text, owner)
	if self._destroyed or not object or not text then
		return nil
	end
	local target = object
	if type(object) == "table" then
		target = object.Object
	end
	if typeof(target) ~= "Instance" or not target:IsA("GuiObject") then
		return nil
	end

	self:_Conn(target.MouseEnter, function()
		if self._destroyed or not self._tooltip then
			return
		end
		self._tooltipText.Text = tostring(text)
		self._tooltip.Visible = true
		self:_PlaceTooltip()
	end, owner)
	self:_Conn(target.MouseLeave, function()
		if self._tooltip then
			self._tooltip.Visible = false
		end
	end, owner)
	return target
end

function HubLibrary:_SetOpenPopup(control)
	local previous = self._openPopup
	if previous and previous ~= control then
		self._openPopup = nil
		if previous.Close then
			SafeCall(previous.Close)
		end
	end
	self._openPopup = control
end

function HubLibrary:_ClosePopup()
	local popup = self._openPopup
	self._openPopup = nil
	if popup and popup.Close then
		SafeCall(popup.Close)
	end
end

function HubLibrary:_ClosePopupIfOutside(input)
	local popup = self._openPopup
	local object = popup and popup.Object
	if not object or not object.Parent then
		self._openPopup = nil
		return
	end
	local point = ScreenPoint(input.Position)
	local pos, size = object.AbsolutePosition, object.AbsoluteSize
	if point.X < pos.X or point.X > pos.X + size.X or point.Y < pos.Y or point.Y > pos.Y + size.Y then
		self:_ClosePopup()
	end
end

-- Arraste generico (slider, color picker). onPos recebe input.Position.
function HubLibrary:_BindDrag(control, hit, scroll, onPos)
	control._dragConns = control._dragConns or {}
	self:_Conn(hit.InputBegan, function(input)
		if input.UserInputType ~= Enum.UserInputType.MouseButton1
			and input.UserInputType ~= Enum.UserInputType.Touch then
			return
		end

		DisconnectAll(control._dragConns)

		local isTouch = input.UserInputType == Enum.UserInputType.Touch
		if scroll then
			scroll.ScrollingEnabled = false
		end
		onPos(input.Position)

		local function stop()
			DisconnectAll(control._dragConns)
			if scroll and scroll.Parent then
				scroll.ScrollingEnabled = true
			end
		end

		table.insert(control._dragConns, UserInputService.InputChanged:Connect(function(changed)
			local isMove
			if isTouch then
				isMove = changed == input
			else
				isMove = changed.UserInputType == Enum.UserInputType.MouseMovement
			end
			if isMove then
				onPos(changed.Position)
			end
		end))

		table.insert(control._dragConns, UserInputService.InputEnded:Connect(function(ended)
			if ended == input or (not isTouch and ended.UserInputType == Enum.UserInputType.MouseButton1) then
				stop()
			end
		end))
	end, control)
end

-- Retorna uma tabela { Moved = bool }. Moved so vira true depois de
-- 4px de movimento, assim um clique simples nao e tratado como arraste.
function HubLibrary:_MakeDraggable(handle, getOrigin, setPosition)
	local dragInput, dragStart, dragOrigin
	local state = { Moved = false }

	self:_Conn(handle.InputBegan, function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then
			dragInput = input
			dragStart = input.Position
			dragOrigin = getOrigin()
			state.Moved = false
		end
	end)

	self:_Conn(UserInputService.InputChanged, function(input)
		if self._destroyed or not dragInput then
			return
		end
		local isMove
		if dragInput.UserInputType == Enum.UserInputType.Touch then
			isMove = input == dragInput
		else
			isMove = input.UserInputType == Enum.UserInputType.MouseMovement
		end
		if not isMove then
			return
		end
		local delta = input.Position - dragStart
		if delta.Magnitude > 4 then
			state.Moved = true
		end
		if state.Moved then
			setPosition(dragOrigin + Vector2.new(delta.X, delta.Y))
		end
	end)

	self:_Conn(UserInputService.InputEnded, function(input)
		if not dragInput then
			return
		end
		if input == dragInput
			or (dragInput.UserInputType == Enum.UserInputType.MouseButton1
				and input.UserInputType == Enum.UserInputType.MouseButton1) then
			dragInput = nil
		end
	end)

	return state
end

function HubLibrary:SetFabPosition(x, y)
	if self._destroyed or not self._fab then
		return false
	end
	x = tonumber(x)
	y = tonumber(y)
	if not x or not y then
		return false
	end
	self._fabPos = { X = math.clamp(x, 0, 0.97), Y = math.clamp(y, 0.03, 0.97) }
	self._fab.Position = UDim2.fromScale(self._fabPos.X, self._fabPos.Y)
	return true
end

-- Aceita: Carregar("Nome", opcoes) | Library:Carregar("Nome") |
--         Carregar({ Name = "Nome", ... })
function HubLibrary.Carregar(...)
	local args = { ... }
	if args[1] == HubLibrary then
		table.remove(args, 1)
	end
	local nomeDoHub, opcoes = args[1], args[2]
	if type(nomeDoHub) == "table" and opcoes == nil then
		opcoes = nomeDoHub
		nomeDoHub = opcoes.Name or opcoes.Title
	end
	opcoes = type(opcoes) == "table" and opcoes or {}
	nomeDoHub = tostring(nomeDoHub or "Frank Hub")

	-- Remove uma instancia anterior mesmo que tenha sido criada por
	-- outra execucao do loader.
	local previous = Global.FrankHubActive
	if type(previous) == "table" and type(previous.Destroy) == "function" then
		pcall(function()
			previous:Destroy()
		end)
	end
	Global.FrankHubActive = nil

	local guiParent = typeof(opcoes.Parent) == "Instance" and opcoes.Parent
		or LocalPlayer:WaitForChild("PlayerGui")
	local stale = guiParent:FindFirstChild("Menu Hub")
	if stale then
		stale:Destroy()
	end

	local self = setmetatable({}, HubLibrary)
	self.Abas = {}
	self.AbaAtiva = nil
	self.Version = HubLibrary.Version
	self.OnDiscord = nil
	self.ConfigCallback = nil
	self.ConfigTab = nil

	self._destroyed = false
	self._ready = false
	self._restoring = false
	self._conns = {}
	self._themed = {}
	self._themes = {}
	self._registry = {}
	self._registryById = {}
	self._state = {}
	self._stateListeners = {}
	self._events = {}
	self._profiles = {}
	self._notifCount = 0
	self._searchQuery = ""
	self._searchEnabled = true
	self._animations = opcoes.Animations ~= false
	self._responsiveEnabled = opcoes.Responsive ~= false
	self._fireOnRestore = opcoes.FireCallbacks ~= false
	self._minimized = false
	self._minToken = 0
	self._autoSave = opcoes.AutoSave == true
	self._capturingKey = false
	self._openPopup = nil
	-- Config separado por hub e por jogo.
	self._configFile = tostring(opcoes.ConfigFile or string.format("FrankHub_%s_%s.json", Slug(nomeDoHub), tostring(game.PlaceId)))
	self._menuKey = NormalizeKey(opcoes.MenuKey) or Enum.KeyCode.RightShift
	self._watermarkEnabled = opcoes.Watermark == true
	self._watermarkText = tostring(opcoes.WatermarkText or nomeDoHub)
	self._showSearch = opcoes.Search ~= false
	self._showDiscord = opcoes.Discord ~= false
	self._showConfig = opcoes.Config ~= false
	self._toastLimit = math.clamp(tonumber(opcoes.NotificationLimit) or 6, 1, 12)
	self._batchDepth = 0
	self._batchDirty = false
	self._destroying = false

	for name, theme in pairs(THEMES) do
		self._themes[name] = MergeTheme(THEMES.Crimson, theme)
	end
	for name, theme in pairs(EXTRA_THEMES) do
		self._themes[name] = MergeTheme(THEMES.Crimson, theme)
	end
	local themeName = (type(opcoes.Theme) == "string" and self._themes[opcoes.Theme]) and opcoes.Theme or "Crimson"
	self._themeName = themeName
	self._theme = self._themes[themeName]

	self:_BuildUI(nomeDoHub)
	self:_BuildV5Extras(nomeDoHub, opcoes)
	self.ScreenGui.Parent = guiParent
	self:SetResponsive(self._responsiveEnabled)

	Global.FrankHubActive = self
	HubLibrary._Active = self

	self:Fire("Loaded", nomeDoHub)
	return self
end

function HubLibrary.Unload()
	local hub = Global.FrankHubActive or HubLibrary._Active
	if hub and not hub._destroyed then
		hub:Destroy()
	end
	Global.FrankHubActive = nil
	HubLibrary._Active = nil
	return true
end

function HubLibrary:Destroy()
	if self._destroyed then
		return
	end
	self:Fire("Destroyed")
	self._destroyed = true

	-- Desconecta eventos globais de controles e abas.
	for _, entry in ipairs(self._registry) do
		local control = entry.Control
		if control then
			DisconnectAll(control._conns)
			DisconnectAll(control._dragConns)
			DisconnectAll(control._dyn)
		end
	end
	for _, tab in ipairs(self.Abas) do
		DisconnectAll(tab._conns)
		tab._destroyed = true
	end

	DisconnectAll(self._conns)

	if self.ScreenGui then
		self.ScreenGui:Destroy()
	end

	if Global.FrankHubActive == self then
		Global.FrankHubActive = nil
	end
	if HubLibrary._Active == self then
		HubLibrary._Active = nil
	end

	self.Abas = {}
	self.AbaAtiva = nil
	self._registry = {}
	self._registryById = {}
	self._themed = {}
	self._events = {}
	self._stateListeners = {}
	self._openPopup = nil
	self.ScreenGui = nil
	self.BackGround = nil
	self._tooltip = nil
	self._notifHolder = nil
end

function HubLibrary:_ApplySearch()
	if self._destroyed then
		return
	end
	local query = string.lower(tostring(self._searchQuery or ""))
	query = query:match("^%s*(.-)%s*$") or ""

	local counts = {}
	for _, entry in ipairs(self._registry) do
		local object = entry.Object
		if object and object.Parent then
			local matched = true
			if query ~= "" then
				if entry.Kind == "Section" or entry.Kind == "Spacer" then
					matched = false
				else
					local haystack = string.lower((entry.Title or "") .. " " .. (entry.Description or ""))
					matched = string.find(haystack, query, 1, true) ~= nil
				end
			end
			entry.Visible = matched
			object.Visible = matched and not entry.ForceHidden
			if matched and not entry.ForceHidden and entry.Kind ~= "Section" and entry.Kind ~= "Spacer" then
				counts[entry.Tab] = (counts[entry.Tab] or 0) + 1
			end
		end
	end

	-- Contagem por aba e estado "Nenhum resultado".
	for _, tab in ipairs(self.Abas) do
		local n = counts[tab] or 0
		if query ~= "" then
			tab.Btn.Text = tab.Name .. (n > 0 and (" (" .. n .. ")") or "")
			tab.Btn.TextTransparency = n > 0 and 0 or 0.55
		else
			tab.Btn.Text = tab.Name
			tab.Btn.TextTransparency = 0
		end
	end
	if self._noResults then
		local active = self.AbaAtiva
		self._noResults.Visible = query ~= "" and active ~= nil and (counts[active] or 0) == 0
	end
end

function HubLibrary:SetSearchEnabled(enabled)
	if self._destroyed then
		return false
	end
	enabled = enabled ~= false
	self._searchEnabled = enabled
	self._searchFrame.Visible = enabled

	self._pages.Position = UDim2.fromOffset(0, offset)
	self._pages.Size = UDim2.new(1, 0, 1, -offset)
	return enabled
end

	if self._destroyed then

	end
	self._searchInput.Text = tostring(query or "")
	return self._searchQuery
end

function HubLibrary:ClearSearch()
	return self:Search("")
end

function HubLibrary:Refresh()

	self:Fire("Refresh")

function HubLibrary:SelecionarAba(aba)
	if self._destroyed then
		return
	end
	if type(aba) == "string" then
		for _, candidate in ipairs(self.Abas) do
			if candidate.Name == aba then
				aba = candidate
				break
			end
		end
	end
	if type(aba) ~= "table" or aba._destroyed then
		return
	end

	self:_ClosePopup()

	local previous = self.AbaAtiva
	for _, other in ipairs(self.Abas) do
		local active = other == aba
		other.Scroll.Visible = active
		other:_StyleButton(active)
	end
	self.AbaAtiva = aba

	if self._tooltip then
		self._tooltip.Visible = false
	end

	if previous ~= aba and self._animations then
		aba.Scroll.Position = UDim2.fromOffset(16, 0)
		self:Tween(aba.Scroll, { Position = UDim2.fromOffset(0, 0) }, 0.18, Enum.EasingStyle.Quint)
	end

	self:_ApplySearch()
	self:Fire("TabChanged", aba, previous)
end

function Tab:Destroy()
	if self._destroyed then
		return
	end
	local hub = self._hub
	self._destroyed = true

	for _, entry in ipairs(table.clone(hub._registry)) do
		if entry.Tab == self then
			local control = entry.Control
			if control and type(control.Destroy) == "function" then
				pcall(control.Destroy)
			else
				hub:_Unregister(entry)
			end
		end
	end

	DisconnectAll(self._conns)

	local index = table.find(hub.Abas, self)
	if index then
		table.remove(hub.Abas, index)
	end
	self.Scroll:Destroy()
	self.Btn:Destroy()

	if hub.AbaAtiva == self then
		hub.AbaAtiva = nil
		if hub.Abas[1] then
			hub:SelecionarAba(hub.Abas[1])
		end
	end
end

function HubLibrary:_MakeId(tabName, title, custom)
	local base
	if custom then
		base = tostring(custom)
	else
		base = Slug(tabName) .. "_" .. Slug(title)
	end
	local id = base
	local counter = 1
	while self._registryById[id] do
		counter = counter + 1
		id = base .. "_" .. tostring(counter)
	end
	return id
end

function HubLibrary:_Unregister(entry)
	local index = table.find(self._registry, entry)
	if index then
		table.remove(self._registry, index)
	end
	self._registryById[entry.Id] = nil
end

function Tab:_Register(control, kind, title, opts)
	opts = opts or {}
	local hub = self._hub

	local entry = {
		-- Prioridade do id: Id explicito > Flag/StateKey > aba_titulo.
		Id = hub:_MakeId(self.Name, title, opts.Id or opts.StateKey),
		Kind = kind,
		Control = control,
		Object = control.Object,
		Title = tostring(title or ""),
		Description = opts.Description,
		Tab = self,
		Visible = true,
		ForceHidden = false,
		Persist = opts.Persist ~= false and self.Persist ~= false,
		Default = DeepCopy(opts.Default),
		CreatedAt = os.clock(),
	}

	control.Id = entry.Id
	control.Kind = kind
	control.Title = entry.Title
	control.Description = entry.Description
	control._entry = entry

	table.insert(hub._registry, entry)
	hub._registryById[entry.Id] = entry
	table.insert(self.Items, control)

	local stateBinding

	control.Destroy = function()
		hub:_Unregister(entry)
		DisconnectAll(control._conns)
		DisconnectAll(control._dragConns)
		DisconnectAll(control._dyn)
		if self.Scroll and self.Scroll.Parent then
			self.Scroll.ScrollingEnabled = true
		end
		if hub._openPopup == control then
			hub._openPopup = nil
		end
		if stateBinding then
			stateBinding.Disconnect()
			stateBinding = nil
		end
		local index = table.find(self.Items, control)
		if index then
			table.remove(self.Items, index)
		end
		if control.Object then
			control.Object:Destroy()
		end
	end

	-- Tooltip no card e tambem no botao interno (que pode cobrir o card).
	if opts.Description and control.Object then
		hub:AddTooltip(control.Object, opts.Description, control)
		if control.Button and control.Button ~= control.Object then
			hub:AddTooltip(control.Button, opts.Description, control)
		end
	end

	local stateKey = opts.StateKey and tostring(opts.StateKey) or nil
	if stateKey and control.Get and control.Set then
		control.StateKey = stateKey
		-- Valor ja existente no state: aplica e (por padrao) dispara o callback.
		if hub._state[stateKey] ~= nil then
			control:Set(hub._state[stateKey], not hub._fireOnRestore)
		else
			hub._state[stateKey] = control:Get()
		end
		stateBinding = hub:BindState(stateKey, function(value)
			-- Mudanca vinda de fora (SetState/outro controle): dispara callback.
			control:Set(value)
		end)
	end

	if hub._searchQuery ~= "" then
		hub:_ApplySearch()
	end
	return control
end

function Tab:_Changed(control, callback, value)
	local hub = self._hub
	if hub._destroyed then
		return
	end
	if control.StateKey then
		hub:SetState(control.StateKey, value)
	end
	Call(callback, value)
	hub:Fire("ControlChanged", control, value)
	hub:_AutoSaveTick()
end

		valorInicial = Pick(o, "Default", "Value")
		descricao = o.Description
		stateKey = Pick(o, "Flag", "StateKey")
		id = o.Id
	end

	local card = self:_NewCard(46, "Toggle")

	local hit = New("TextButton", {
		Name = "ToggleButton",
		Size = UDim2.fromScale(1, 1),
		BackgroundTransparency = 1,
		Text = "",
		AutoButtonColor = false,
	}, card)

		Name = "Track",
		AnchorPoint = Vector2.new(1, 0.5),
		Position = UDim2.new(1, -12, 0.5, 0),
		Size = UDim2.fromOffset(52, 26),
		BorderSizePixel = 0,

		Name = "ButtonLine",
		AnchorPoint = Vector2.new(0, 0.5),
		Size = UDim2.fromOffset(20, 20),
		BorderSizePixel = 0,
	}, track)
	Corner(knob, 10)

	local active = valorInicial == true
	local control = { Object = card, Button = hit }

	local function paint(animate)
		local t = hub._theme
		local trackColor = active and t.Success or t.Control
		local knobColor = active and t.Text or t.MutedText
		local knobPos = active and UDim2.new(1, -23, 0.5, 0) or UDim2.new(0, 3, 0.5, 0)
		trackStroke.Color = t.Border
		if animate then
			hub:Tween(track, { BackgroundColor3 = trackColor }, 0.12)
			hub:Tween(knob, { BackgroundColor3 = knobColor, Position = knobPos }, 0.12)
		else
			track.BackgroundColor3 = trackColor
			knob.BackgroundColor3 = knobColor
			knob.Position = knobPos
		end
	end
	hub:_Themed(track, function()
		paint(false)
	end)

	local function apply(value, silent)
		value = value == true
		if value == active then
			return
		end
		active = value
		paint(true)
		if not silent then
			self:_Changed(control, callback, active)
		end
	end

	control.Set = function(_, value, silent)
		apply(value, silent)
	end
	control.Get = function()
		return active
	end
	control.Toggle = function()
		apply(not active)
	end

	hub:_Conn(hit.MouseButton1Click, function()
		apply(not active)
	end, control)

	return self:_Register(control, "Toggle", texto, {
		Description = descricao,
		Default = active,
		StateKey = stateKey,
		Id = id,
	})
end

function Tab:AddSlider(texto, valorMin, valorMax, valorPadrao, callback, descricao, stateKey, passo)
	if self._destroyed then
		return nil
	end
	local hub = self._hub
	local id
	if type(texto) == "table" then
		local o = texto
		texto = Pick(o, "Name", "Title", "Text")
		valorMin = o.Min
		valorMax = o.Max
		valorPadrao = Pick(o, "Default", "Value")
		callback = o.Callback
		descricao = o.Description
		stateKey = Pick(o, "Flag", "StateKey")
		passo = Pick(o, "Step", "Increment")
		id = o.Id
	end

	local min = tonumber(valorMin) or 0
	local max = tonumber(valorMax) or 100
	if max < min then
		min, max = max, min
	end
	local step = tonumber(passo) or 1
	if step <= 0 then
		step = 1
	end

	local function needsDecimal(n)
		return math.abs(n - math.floor(n + 0.5)) > 1e-9
	end
	local decimals = 0
	while decimals < 6 and (needsDecimal(step * 10 ^ decimals) or needsDecimal(min * 10 ^ decimals)) do
		decimals = decimals + 1
	end

	local function normalize(v)
		v = math.clamp(tonumber(v) or min, min, max)
		v = min + math.floor((v - min) / step + 0.5) * step
		v = math.clamp(v, min, max)
		local m = 10 ^ decimals
		return math.floor(v * m + 0.5) / m
	end

	local value = normalize(valorPadrao or min)

	local card = self:_NewCard(60, "Slider")

	local title = New("TextLabel", {
		Name = "TitleLabel",
		BackgroundTransparency = 1,
		Position = UDim2.fromOffset(12, 6),
		Size = UDim2.new(1, -90, 0, 22),
		Text = tostring(texto),
		TextSize = 17,
		FontFace = FONT_BOLD,
		TextXAlignment = Enum.TextXAlignment.Left,
		TextTruncate = Enum.TextTruncate.AtEnd,
	}, card)
	hub:_T(title, "TextColor3", "Text")

	local valueLabel = New("TextLabel", {
		Name = "ValueLabel",
		AnchorPoint = Vector2.new(1, 0),
		BackgroundTransparency = 1,
		Position = UDim2.new(1, -12, 0, 6),
		Size = UDim2.fromOffset(70, 22),
		Text = "",
		TextSize = 17,
		FontFace = FONT_BOLD,
		TextXAlignment = Enum.TextXAlignment.Right,
	}, card)
	hub:_T(valueLabel, "TextColor3", "Accent")

	local track = New("Frame", {
		Name = "SliderTrack",
		Position = UDim2.fromOffset(12, 40),
		Size = UDim2.new(1, -24, 0, 8),
		BorderSizePixel = 0,
	}, card)
	Corner(track, 4)
	local trackStroke = Stroke(track, 1.5)

	local fill = New("Frame", {
		Name = "Progress",
		Size = UDim2.new(0, 0, 1, 0),
		BorderSizePixel = 0,
	}, track)
	Corner(fill, 4)

	local knob = New("Frame", {
		Name = "FillButton",
		AnchorPoint = Vector2.new(0.5, 0.5),
		Size = UDim2.fromOffset(16, 16),
		Position = UDim2.new(0, 0, 0.5, 0),
		BorderSizePixel = 0,
		ZIndex = 2,
	}, track)
	Corner(knob, 8)

	hub:_Themed(track, function(t)
		track.BackgroundColor3 = t.Control
		trackStroke.Color = t.Border
		fill.BackgroundColor3 = t.Success
		knob.BackgroundColor3 = t.Text
	end)

	local hit = New("TextButton", {
		Name = "Hit",
		Position = UDim2.fromOffset(0, 26),
		Size = UDim2.new(1, 0, 0, 34),
		BackgroundTransparency = 1,
		Text = "",
		AutoButtonColor = false,
	}, card)

	local control = { Object = card, Button = hit, Track = track, FillButton = knob, _dragConns = {} }

	local function render()
		local pct = 0
		if max ~= min then
			pct = (value - min) / (max - min)
		end
		fill.Size = UDim2.new(pct, 0, 1, 0)
		knob.Position = UDim2.new(pct, 0, 0.5, 0)
		valueLabel.Text = tostring(value)
	end

	local function apply(v, silent)
		v = normalize(v)
		if v == value then
			return
		end
		value = v
		render()
		if not silent then
			self:_Changed(control, callback, value)
		end
	end

	control.Set = function(_, v, silent)
		apply(v, silent)
	end
	control.Get = function()
		return value
	end

	hub:_BindDrag(control, hit, self.Scroll, function(position)
		local width = track.AbsoluteSize.X
		if width <= 0 then
			return
		end
		local pct = math.clamp((position.X - track.AbsolutePosition.X) / width, 0, 1)
		apply(min + (max - min) * pct)
	end)

	render()

	return self:_Register(control, "Slider", texto, {
		Description = descricao,
		Default = value,
		StateKey = stateKey,
		Id = id,
	})
end

--==================================================
-- DROPDOWN (simples e multiplo, com busca opcional)
--==================================================

function Tab:_BuildDropdown(multi, cfg)
	local hub = self._hub
	local texto = cfg.texto
	local extra = type(cfg.extra) == "table" and cfg.extra or {}
	local callback = cfg.callback
	local CARD_H, ROW_H, MAX_ROWS, SEARCH_ROW = 46, 28, 5, 34

	local searchPref = extra.Search
	if searchPref == nil then
		searchPref = extra.Searchable
	end

	local options = CopyList(cfg.opcoes)
	local current = nil
	local selected = {}
	if multi then
		local initial = cfg.valorInicial
		if type(initial) ~= "table" then
			initial = (initial ~= nil) and { initial } or {}
		end
		for _, value in ipairs(initial) do
			selected[value] = true
		end
	else
		current = cfg.valorInicial
		if current == nil then
			current = options[1]
		end
	end

	local card = self:_NewCard(CARD_H, multi and "MultiDropdown" or "Dropdown")
	card.ClipsDescendants = true
	self:_Title(card, texto, -176, CARD_H)

	local button = New("TextButton", {
		Name = "Frame",
		AnchorPoint = Vector2.new(1, 0),
		Position = UDim2.new(1, -12, 0, 8),
		Size = UDim2.fromOffset(150, 30),
		BorderSizePixel = 0,
		AutoButtonColor = false,
		Text = "",
	}, card)
	Corner(button, 4)
	local buttonStroke = Stroke(button, 2)

	local selectedLabel = New("TextLabel", {
		Name = "TitleLabel",
		BackgroundTransparency = 1,
		Position = UDim2.fromOffset(8, 0),
		Size = UDim2.new(1, -30, 1, 0),
		Text = "",
		TextSize = 16,
		FontFace = FONT_BOLD,
		TextXAlignment = Enum.TextXAlignment.Left,
		TextTruncate = Enum.TextTruncate.AtEnd,
	}, button)

	local arrow = New("TextLabel", {
		Name = "TextLabel",
		AnchorPoint = Vector2.new(1, 0.5),
		Position = UDim2.new(1, -6, 0.5, 0),
		Size = UDim2.fromOffset(18, 18),
		BackgroundTransparency = 1,
		Text = "v",
		TextSize = 16,
		FontFace = FONT_BOLD,
	}, button)

	local searchFrame = New("Frame", {
		Name = "SearchFrame",
		Position = UDim2.fromOffset(12, CARD_H),
		Size = UDim2.new(1, -24, 0, 28),
		BorderSizePixel = 0,
		Visible = false,
	}, card)
	Corner(searchFrame, 4)
	local searchStroke = Stroke(searchFrame, 2)
	local searchBox = New("TextBox", {
		Name = "Input",
		Position = UDim2.fromOffset(8, 0),
		Size = UDim2.new(1, -16, 1, 0),
		BackgroundTransparency = 1,
		ClearTextOnFocus = false,
		PlaceholderText = "Buscar...",
		Text = "",
		TextSize = 15,
		FontFace = FONT_REG,
		TextXAlignment = Enum.TextXAlignment.Left,
	}, searchFrame)

	local list = New("ScrollingFrame", {
		Name = "ContainerOpcoes",
		Position = UDim2.fromOffset(12, CARD_H),
		Size = UDim2.new(1, -24, 0, 0),
		BorderSizePixel = 0,
		ScrollBarThickness = 3,
		CanvasSize = UDim2.new(),
		AutomaticCanvasSize = Enum.AutomaticSize.Y,
		ScrollingDirection = Enum.ScrollingDirection.Y,
		Visible = false,
	}, card)
	Corner(list, 4)
	New("UIListLayout", { SortOrder = Enum.SortOrder.LayoutOrder }, list)

	local optionButtons = {}
	local opened = false
	local filter = ""
	local control = { Object = card, Button = button, _dyn = {} }

	local function isSelected(value)
		if multi then
			return selected[value] == true
		end
		return value == current
	end

	local function selectedList()
		local out = {}
		for _, option in ipairs(options) do
			if selected[option] then
				table.insert(out, option)
			end
		end
		return out
	end

	local function paintOptions()
		local t = hub._theme
		for _, item in ipairs(optionButtons) do
			item.Button.BackgroundColor3 = isSelected(item.Value) and t.Accent or t.Control
			item.Button.TextColor3 = t.Text
		end
	end

	hub:_Themed(button, function(t)
		button.BackgroundColor3 = t.Control
		buttonStroke.Color = t.Border
		selectedLabel.TextColor3 = t.Text
		arrow.TextColor3 = t.Accent
		searchFrame.BackgroundColor3 = t.Control
		searchStroke.Color = t.Border
		searchBox.TextColor3 = t.Text
		searchBox.PlaceholderColor3 = t.MutedText
		list.BackgroundColor3 = t.Control
		list.ScrollBarImageColor3 = t.Accent
		paintOptions()
	end)

	local function updateText()
		if multi then
			local items = selectedList()
			if #items == 0 then
				selectedLabel.Text = "Nenhum"
			elseif #items <= 2 then
				local names = {}
				for _, item in ipairs(items) do
					table.insert(names, tostring(item))
				end
				selectedLabel.Text = table.concat(names, ", ")
			else
				selectedLabel.Text = tostring(#items) .. " selecionados"
			end
		else
			selectedLabel.Text = current ~= nil and tostring(current) or "Select..."
		end
	end

	local function matches(option)
		return filter == "" or string.find(string.lower(tostring(option)), filter, 1, true) ~= nil
	end

	local function hasSearch()
		if searchPref == nil then
			return #options > 8
		end
		return searchPref == true
	end

	local function resize()
		local searchOn = opened and hasSearch()
		local visibleCount = 0
		for _, item in ipairs(optionButtons) do
			local show = matches(item.Value)
			item.Button.Visible = show
			if show then
				visibleCount = visibleCount + 1
			end
		end
		local listHeight = math.min(visibleCount, MAX_ROWS) * ROW_H
		local listY = CARD_H + (searchOn and SEARCH_ROW or 0)
		searchFrame.Visible = searchOn
		list.Position = UDim2.fromOffset(12, listY)
		list.Size = UDim2.new(1, -24, 0, listHeight)
		list.Visible = opened and visibleCount > 0
		arrow.Text = opened and "^" or "v"
		local target = CARD_H
		if opened then
			target = listY + listHeight + 10
		end
		hub:Tween(card, { Size = UDim2.new(1, 0, 0, target) }, 0.16)
		if hub._animations == false then
			card.Size = UDim2.new(1, 0, 0, target)
		end
	end

	local function setOpened(state)
		if state and #options == 0 then
			return
		end
		opened = state
		if opened then
			hub:_SetOpenPopup(control)
		elseif hub._openPopup == control then
			hub._openPopup = nil
		end
		if not opened and searchBox.Text ~= "" then
			searchBox.Text = ""
		end
		resize()
	end

	local function setCurrent(value, silent)
		if value == current then
			return
		end
		current = value
		updateText()
		paintOptions()
		if not silent then
			self:_Changed(control, callback, current)
		end
	end

	local function toggleOption(option)
		if selected[option] then
			selected[option] = nil
		else
			selected[option] = true
		end
		updateText()
		paintOptions()
		self:_Changed(control, callback, selectedList())
	end

	local function buildOptions()
		DisconnectAll(control._dyn)
		for _, item in ipairs(optionButtons) do
			item.Button:Destroy()
		end
		table.clear(optionButtons)

		for index, option in ipairs(options) do
			local optionButton = New("TextButton", {
				Name = "Option" .. tostring(index),
				LayoutOrder = index,
				Size = UDim2.new(1, 0, 0, ROW_H),
				BorderSizePixel = 0,
				AutoButtonColor = false,
				Text = tostring(option),
				TextSize = 16,
				FontFace = FONT_REG,
				TextXAlignment = Enum.TextXAlignment.Left,
				TextTruncate = Enum.TextTruncate.AtEnd,
			}, list)
			New("UIPadding", { PaddingLeft = UDim.new(0, 8) }, optionButton)
			table.insert(optionButtons, { Button = optionButton, Value = option })

			table.insert(control._dyn, optionButton.MouseEnter:Connect(function()
				optionButton.BackgroundColor3 = hub._theme.ControlHover
			end))
			table.insert(control._dyn, optionButton.MouseLeave:Connect(function()
				paintOptions()
			end))
			table.insert(control._dyn, optionButton.MouseButton1Click:Connect(function()
				if multi then
					toggleOption(option)
				else
					setCurrent(option)
					setOpened(false)
				end
			end))
		end

		paintOptions()
		resize()
	end

	hub:_Conn(button.MouseButton1Click, function()
		setOpened(not opened)
	end, control)

	hub:_Conn(searchBox:GetPropertyChangedSignal("Text"), function()
		filter = string.lower(searchBox.Text):match("^%s*(.-)%s*$") or ""
		resize()
	end, control)

	if multi then
		control.Set = function(_, value, silent)
			local list_ = type(value) == "table" and value or ((value ~= nil) and { value } or {})
			local newSet = {}
			for _, item in ipairs(list_) do
				if table.find(options, item) then
					newSet[item] = true
				end
			end
			local same = true
			for _, option in ipairs(options) do
				if (newSet[option] == true) ~= (selected[option] == true) then
					same = false
					break
				end
			end
			if same then
				return
			end
			selected = newSet
			updateText()
			paintOptions()
			if not silent then
				self:_Changed(control, callback, selectedList())
			end
		end
		control.Get = selectedList
		control.SelectAll = function()
			control:Set(options)
		end
		control.Clear = function()
			control:Set({})
		end
	else
		control.Set = function(_, value, silent)
			setCurrent(value, silent)
		end
		control.Get = function()
			return current
		end
	end

	control.Open = function()
		setOpened(true)
	end
	control.Close = function()
		setOpened(false)
	end
	control.SetOptions = function(_, newOptions)
		options = CopyList(newOptions)
		if multi then
			for value in pairs(selected) do
				if not table.find(options, value) then
					selected[value] = nil
				end
			end
		elseif current ~= nil and not table.find(options, current) then
			current = options[1]
		end
		updateText()
		if #options == 0 then
			opened = false
		end
		buildOptions()
	end
	control.GetOptions = function()
		return table.clone(options)
	end

	updateText()
	buildOptions()

	return self:_Register(control, multi and "MultiDropdown" or "Dropdown", texto, {
		Description = cfg.descricao,
		Default = multi and selectedList() or current,
		StateKey = cfg.stateKey,
		Id = cfg.id,
	})
end

--==================================================
-- INPUT (texto / numero com validacao)
--==================================================

function Tab:AddInput(texto, callback, valorInicial, descricao, stateKey, extra)
	if self._destroyed then
		return nil
	end
	local hub = self._hub
	local id
	if type(texto) == "table" then
		local o = texto
		extra = o
		texto = Pick(o, "Name", "Title", "Text")
		callback = o.Callback
		valorInicial = Pick(o, "Default", "Value")
		descricao = o.Description
		stateKey = Pick(o, "Flag", "StateKey")
		id = o.Id
	end
	extra = type(extra) == "table" and extra or {}

	local numeric = extra.Numeric == true
	local min, max = tonumber(extra.Min), tonumber(extra.Max)

	local function normalize(v)
		if numeric then
			local n = tonumber(v)
			if not n then
				return nil
			end
			if min then
				n = math.max(n, min)
			end
			if max then
				n = math.min(n, max)
			end
			return n
		end
		if v == nil then
			return ""
		end
		return tostring(v)
	end

	local value = normalize(valorInicial)
	if value == nil then
		value = min or 0
	end

	local card = self:_NewCard(46, "Input")
	self:_Title(card, texto, -176, 46)

	local frame = New("Frame", {
		Name = "InputFrame",
		AnchorPoint = Vector2.new(1, 0),
		Position = UDim2.new(1, -12, 0, 8),
		Size = UDim2.fromOffset(150, 30),
		BorderSizePixel = 0,
	}, card)
	Corner(frame, 4)
	local frameStroke = Stroke(frame, 2)

	local box = New("TextBox", {
		Name = "Box",
		Position = UDim2.fromOffset(8, 0),
		Size = UDim2.new(1, -16, 1, 0),
		BackgroundTransparency = 1,
		ClearTextOnFocus = extra.ClearOnFocus == true,
		PlaceholderText = tostring(extra.Placeholder or (numeric and "Numero..." or "Texto...")),
		Text = tostring(value),
		TextSize = 16,
		FontFace = FONT_BOLD,
		TextXAlignment = Enum.TextXAlignment.Left,
		TextTruncate = Enum.TextTruncate.AtEnd,
	}, frame)
	hub:_Themed(frame, function(t)
		frame.BackgroundColor3 = t.Control
		frameStroke.Color = t.Border
		box.TextColor3 = t.Text
		box.PlaceholderColor3 = t.MutedText
	end)

	local control = { Object = card, Button = box }

	local function apply(v, silent)
		local normalized = normalize(v)
		if normalized == nil then
			box.Text = tostring(value)
			return
		end
		box.Text = tostring(normalized)
		if normalized == value then
			return
		end
		value = normalized
		if not silent then
			self:_Changed(control, callback, value)
		end
	end

	control.Set = function(_, v, silent)
		apply(v, silent)
	end
	control.Get = function()
		return value
	end

	-- O callback roda ao confirmar (Enter) ou ao sair da caixa.
	hub:_Conn(box.FocusLost, function()
		apply(box.Text)
	end, control)

	return self:_Register(control, "Input", texto, {
		Description = descricao,
		Default = value,
		StateKey = stateKey,
		Id = id,
	})
end

		hexLabel.Text = ToHex(color)
		sv.BackgroundColor3 = Color3.fromHSV(h, 1, 1)
		svCursor.Position = UDim2.fromScale(s, 1 - v)
		hueCursor.Position = UDim2.fromScale(0.5, h)
	end

	local function resize()
		local target = opened and (CARD_H + PICK_H + 12) or CARD_H
		hub:Tween(card, { Size = UDim2.new(1, 0, 0, target) }, 0.16)
		if hub._animations == false then
			card.Size = UDim2.new(1, 0, 0, target)
		end
	end

	local function setOpened(state)
		opened = state
		if opened then
			hub:_SetOpenPopup(control)
		elseif hub._openPopup == control then
			hub._openPopup = nil
		end
		resize()
	end

	local function commitHSV()
		local newColor = Color3.fromHSV(h, s, v)
		render()
		if newColor ~= color then
			color = newColor
			render()
			self:_Changed(control, callback, color)
		end
	end

	control.Set = function(_, value, silent)
		local parsed = ParseColor(value)
		if not parsed or parsed == color then
			return
		end
		color = parsed
		local nh, ns, nv = parsed:ToHSV()
		-- Preserva o matiz quando ele e indefinido (cinza/preto).
		if ns > 0 and nv > 0 then
			h = nh
		end
		s, v = ns, nv
		render()
		if not silent then
			self:_Changed(control, callback, color)
		end
	end
	control.Get = function()
		return color
	end
	control.GetHex = function()
		return ToHex(color)
	end
	control.Open = function()
		setOpened(true)
	end
	control.Close = function()
		setOpened(false)
	end
	-- Color3 nao e JSON: salvamos o hexadecimal.
	control._serialize = function()
		return ToHex(color)
	end
	control._deserialize = function(value, silent)
		control:Set(value, silent)
	end

	hub:_Conn(swatch.MouseButton1Click, function()
		setOpened(not opened)
	end, control)

	hub:_BindDrag(control, svHit, self.Scroll, function(position)
		local point = ScreenPoint(position)
		local size = sv.AbsoluteSize
		if size.X <= 0 or size.Y <= 0 then
			return
		end
		s = math.clamp((point.X - sv.AbsolutePosition.X) / size.X, 0, 1)
		v = 1 - math.clamp((point.Y - sv.AbsolutePosition.Y) / size.Y, 0, 1)
		commitHSV()
	end)

	hub:_BindDrag(control, hueHit, self.Scroll, function(position)
		local point = ScreenPoint(position)
		local size = hueBar.AbsoluteSize
		if size.Y <= 0 then
			return
		end
		h = math.clamp((point.Y - hueBar.AbsolutePosition.Y) / size.Y, 0, 0.999)
		commitHSV()
	end)

	render()

	return self:_Register(control, "ColorPicker", texto, {
		Description = descricao,
		Default = ToHex(color),
		StateKey = stateKey,
		Id = id,
	})
end

function Tab:AddButton(texto, callback, descricao, confirmar)
	if self._destroyed then
		return nil
	end
	local hub = self._hub
	if type(texto) == "table" then
		local o = texto
		texto = Pick(o, "Name", "Title", "Text")
		callback = o.Callback
		descricao = o.Description
		confirmar = o.Confirm
	end

	local label = tostring(texto)
	local armed, token = false, 0

	self._order = self._order + 1
	local button = New("TextButton", {
		Name = "ButtonTemplate",
		LayoutOrder = self._order,
		Size = UDim2.new(1, 0, 0, 40),
		BorderSizePixel = 0,
		AutoButtonColor = false,
		Text = label,
		TextSize = 18,
		FontFace = FONT_BOLD,
	}, self.Scroll)
	Corner(button, 5)
	local stroke = Stroke(button, 2)
	hub:_Themed(button, function(t)
		button.BackgroundColor3 = t.Control
		button.TextColor3 = t.Text
		stroke.Color = armed and t.Danger or t.Border
	end)

	local control = { Object = button, Button = button }
	hub:_Hover(button, "Control", "ControlHover", control)

	local function disarm()
		armed = false
		token = token + 1
		button.Text = label
		stroke.Color = hub._theme.Border
	end

	control.SetText = function(_, text)
		label = tostring(text)
		if not armed then
			button.Text = label
		end
		if control._entry then
			control._entry.Title = label
		end
	end

	hub:_Conn(button.MouseButton1Click, function()
		if confirmar and not armed then
			armed = true
			token = token + 1
			local mine = token
			button.Text = "Confirmar? Clique de novo"
			stroke.Color = hub._theme.Danger
			task.delay(3, function()
				if armed and token == mine and button.Parent then
					disarm()
				end
			end)
			return
		end
		if armed then
			disarm()
		end
		Call(callback)
		hub:Fire("ButtonPressed", control)
	end, control)

	return self:_Register(control, "Button", texto, { Description = descricao, Persist = false })
end

-- callback: chamado quando a tecla e PRESSIONADA.
-- onChanged: chamado quando o usuario troca a tecla.
function Tab:AddKeybind(texto, teclaInicial, callback, descricao, onChanged)
	if self._destroyed then
		return nil
	end
	local hub = self._hub
	local id
	if type(texto) == "table" then
		local o = texto
		texto = Pick(o, "Name", "Title", "Text")
		teclaInicial = Pick(o, "Key", "Default")
		callback = o.Callback
		descricao = o.Description
		onChanged = Pick(o, "Changed", "OnChanged")
		id = Pick(o, "Flag", "Id")
	end

	local card = self:_NewCard(46, "Keybind")
	self:_Title(card, texto, -140)

	local keyButton = New("TextButton", {
		Name = "Key",
		AnchorPoint = Vector2.new(1, 0.5),
		Position = UDim2.new(1, -10, 0.5, 0),
		Size = UDim2.fromOffset(116, 30),
		BorderSizePixel = 0,
		AutoButtonColor = false,
		Text = "",
		TextSize = 15,
		FontFace = FONT_BOLD,
	}, card)
	Corner(keyButton, 4)
	local keyStroke = Stroke(keyButton, 2)
	hub:_Themed(keyButton, function(t)
		keyButton.BackgroundColor3 = t.Control
		keyButton.TextColor3 = t.Text
		keyStroke.Color = t.Border
	end)

	local key = NormalizeKey(teclaInicial) or Enum.KeyCode.RightShift
	local listening = false
	local control = { Object = card, Button = keyButton, _conns = {} }

	local function update()
		keyButton.Text = listening and "PRESSIONE..." or key.Name
	end

	local function setKey(newKey, silent)
		local normalized = NormalizeKey(newKey)
		if not normalized or normalized == key then
			return
		end
		key = normalized
		update()
		if not silent then
			Call(onChanged, key)
			hub:Fire("KeybindChanged", tostring(texto), key)
			hub:_AutoSaveTick()
		end
	end

	update()

	-- Se o controle for destruido durante a captura, libera o hub
	-- (senao a tecla do menu e os outros keybinds ficariam travados).
	table.insert(control._conns, {
		Disconnect = function()
			if listening then
				listening = false
				hub._capturingKey = false
			end
		end,
	})

	hub:_Conn(keyButton.MouseButton1Click, function()
		listening = true
		hub._capturingKey = true
		update()
	end, control)

	hub:_Conn(UserInputService.InputBegan, function(input, processed)
		if hub._destroyed or input.UserInputType ~= Enum.UserInputType.Keyboard then
			return
		end
		if listening then
			listening = false
			-- Libera a captura so no proximo frame, para que os outros
			-- handlers deste mesmo evento ainda ignorem a tecla.
			task.defer(function()
				hub._capturingKey = false
			end)
			if input.KeyCode ~= Enum.KeyCode.Escape and input.KeyCode ~= Enum.KeyCode.Unknown then
				setKey(input.KeyCode)
			end
			update()
			return
		end
		if processed or hub._capturingKey then
			return
		end
		if input.KeyCode == key then
			Call(callback, key)
			hub:Fire("KeybindPressed", tostring(texto), key)
		end
	end, control)

	control.Get = function()
		return key
	end
	control.Set = function(_, newKey, silent)
		setKey(newKey, silent)
		return key
	end
	-- Enum nao e serializavel em JSON: salvamos o nome da tecla.
	control._serialize = function()
		return key.Name
	end
	control._deserialize = function(value, silent)
		setKey(value, silent)
	end

	return self:_Register(control, "Keybind", texto, {
		Description = descricao,
		Default = key.Name,
		Id = id,
	})
end

-- Barra de progresso (valor de 0 ate Max; nao entra no config).
function Tab:AddProgress(texto, valorInicial, descricao, maximo)
	if self._destroyed then
		return nil
	end
	local hub = self._hub
	if type(texto) == "table" then
		local o = texto
		texto = Pick(o, "Name", "Title", "Text")
		valorInicial = Pick(o, "Value", "Default")
		descricao = o.Description
		maximo = o.Max
	end

	local max = tonumber(maximo) or 100
	if max <= 0 then
		max = 100
	end
	local value = math.clamp(tonumber(valorInicial) or 0, 0, max)

	local card = self:_NewCard(54, "Progress")
	local title = New("TextLabel", {
		Name = "TitleLabel",
		BackgroundTransparency = 1,
		Position = UDim2.fromOffset(12, 6),

		TextSize = 17,
		FontFace = FONT_BOLD,
		TextXAlignment = Enum.TextXAlignment.Left,
		TextTruncate = Enum.TextTruncate.AtEnd,
	}, card)
	hub:_T(title, "TextColor3", "Text")

	local percent = New("TextLabel", {
		Name = "Percent",
		AnchorPoint = Vector2.new(1, 0),
		BackgroundTransparency = 1,
		Position = UDim2.new(1, -12, 0, 6),
		Size = UDim2.fromOffset(70, 22),
		TextSize = 17,
		FontFace = FONT_BOLD,
		TextXAlignment = Enum.TextXAlignment.Right,
	}, card)
	hub:_T(percent, "TextColor3", "Accent")

	local track = New("Frame", {
		Name = "ProgressTrack",
		Position = UDim2.fromOffset(12, 36),
		Size = UDim2.new(1, -24, 0, 8),
		BorderSizePixel = 0,
	}, card)
	Corner(track, 4)
	local trackStroke = Stroke(track, 1.5)
	local fill = New("Frame", {
		Name = "Fill",
		Size = UDim2.new(0, 0, 1, 0),
		BorderSizePixel = 0,
	}, track)
	Corner(fill, 4)
	hub:_Themed(track, function(t)
		track.BackgroundColor3 = t.Control
		trackStroke.Color = t.Border
		fill.BackgroundColor3 = t.Success
	end)

	local control = { Object = card }

	local function render(animate)
		local pct = value / max
		percent.Text = tostring(math.floor(pct * 100 + 0.5)) .. "%"
		if animate then
			hub:Tween(fill, { Size = UDim2.new(pct, 0, 1, 0) }, 0.15)
		else
			fill.Size = UDim2.new(pct, 0, 1, 0)

		value = math.clamp(tonumber(newValue) or value, 0, max)

			render(false)
		end
	end

		title.Text = tostring(newText)
		if control._entry then
			control._entry.Title = tostring(newText)
		end
	end

	render(false)

	-- Sem StateKey/Get-Set persistente: nao participa do config.
	return self:_Register(control, "Progress", texto, { Description = descricao, Persist = false })
end

-- Mostra/oculta um item (ele continua respeitando a busca).
function HubLibrary:SetEnabled(control, enabled)

-- Verifica: itens removidos, ids duplicados, StateKey usada por tipos
-- diferentes e chaves de state sem nenhum controle ligado.
function HubLibrary:Validate()
	local issues = {}
	local seenIds, keyKinds, controlKeys = {}, {}, {}

	for _, entry in ipairs(self._registry) do
		if not entry.Object or not entry.Object.Parent then
			table.insert(issues, "Item removido: " .. tostring(entry.Id))
		end
		if seenIds[entry.Id] then
			table.insert(issues, "Id duplicado: " .. tostring(entry.Id))
		end
		seenIds[entry.Id] = true

		local key = entry.Control and entry.Control.StateKey
		if key then
			controlKeys[key] = true
			if keyKinds[key] and keyKinds[key] ~= entry.Kind then
				table.insert(issues, string.format(
					"StateKey '%s' usada por tipos diferentes (%s e %s)",
					key, keyKinds[key], entry.Kind
				))
			end
			keyKinds[key] = keyKinds[key] or entry.Kind
		end
	end

	for key in pairs(self._state) do
		if not controlKeys[key] then
			table.insert(issues, "State sem controle: " .. tostring(key))
		end
	end

	if #issues > 0 then
		warn("[Frank Hub] Validate encontrou " .. tostring(#issues) .. " problema(s):\n - " .. table.concat(issues, "\n - "))
	end
	return #issues == 0, issues
end

function HubLibrary:CleanupRegistry()
	local cleaned = {}
	for _, entry in ipairs(self._registry) do
		if entry.Object and entry.Object.Parent then
			table.insert(cleaned, entry)
		else
			self._registryById[entry.Id] = nil
		end
	end
	self._registry = cleaned
	return #cleaned
end

function HubLibrary:ResetAll()
	for _, entry in ipairs(table.clone(self._registry)) do
		local control = entry.Control
		if entry.Persist and entry.Default ~= nil and control and type(control.Set) == "function" then
			pcall(function()
				control:Set(DeepCopy(entry.Default))
			end)
		end
	end
	self:Fire("ConfigReset")
	return true
end

function HubLibrary:Snapshot()
	local data = {
		Version = 4,
		Theme = self._themeName,
		MenuKey = self._menuKey and self._menuKey.Name or nil,
		State = {},
		Values = {},
	}

	-- Tema custom e salvo por inteiro (o nome "Custom" nao e restauravel).
	if self._themeName == "Custom" then
		data.CustomTheme = SerializeTheme(self._theme)
	end
	if self._fabPos then
		data.FabPos = { X = self._fabPos.X, Y = self._fabPos.Y }
	end

	for key, value in pairs(self._state) do
		if IsJSONSafe(value) then
			data.State[key] = DeepCopy(value)
		end
	end

	for _, entry in ipairs(self._registry) do
		local control = entry.Control
		if entry.Persist and control and type(control.Get) == "function" then
			local ok, value = pcall(function()
				if control._serialize then
					return control._serialize()
				end
				return control:Get()
			end)
			if ok and IsJSONSafe(value) then
				data.Values[entry.Id] = value
			end
		end
	end

	return data
end

-- silent: true = nao dispara callbacks; nil = segue a opcao FireCallbacks.
function HubLibrary:Restore(snapshot, silent)
	if self._destroyed or type(snapshot) ~= "table" then
		return false
	end
	if silent == nil then
		silent = not self._fireOnRestore
	end

	self._restoring = true

	if type(snapshot.CustomTheme) == "table" then
		self:SetTheme(DeserializeTheme(snapshot.CustomTheme))
	elseif type(snapshot.Theme) == "string" and self._themes[snapshot.Theme] then
		self:SetTheme(snapshot.Theme)
	end
	if snapshot.MenuKey then
		self:SetMenuKeybind(snapshot.MenuKey)
	end
	if type(snapshot.FabPos) == "table" then
		self:SetFabPosition(snapshot.FabPos.X, snapshot.FabPos.Y)
	end

	for id, value in pairs(snapshot.Values or {}) do
		local entry = self._registryById[id]
		local control = entry and entry.Control
		if control then
			local ok, err = xpcall(function()
				if control._deserialize then
					control._deserialize(value, silent)
				elseif type(control.Set) == "function" then
					control:Set(value, silent)
				end
			end, Traceback)
			if not ok then
				warn("[Frank Hub] Restore:", id, err)
			end
		end
	end

	for key, value in pairs(snapshot.State or {}) do
		self:SetState(key, value, not silent)
	end

	self._restoring = false
	self:Refresh()
	self:Fire("ConfigRestored", snapshot)
	return true
end

function HubLibrary:ImportConfig(json, silent)
	local ok, data = pcall(function()
		return HttpService:JSONDecode(tostring(json or ""))
	end)
	if not ok or type(data) ~= "table" then
		warn("[Frank Hub] JSON invalido:", data)
		return false
	end
	return self:Restore(data, silent)
end

function HubLibrary:SetConfigFile(fileName)
	local safe = tostring(fileName or self._configFile)
	safe = safe:gsub("[^%w%._%-]", "_")
	if not safe:lower():match("%.json$") then
		safe = safe .. ".json"
	end
	self._configFile = safe
	return safe
end

function HubLibrary:SaveConfig(fileName, quiet)
	if fileName then
		self:SetConfigFile(fileName)
	end
	if not FileApiAvailable() then
		return false, "writefile/readfile/isfile indisponivel neste executor"
	end
	local payload = self:ExportConfig()
	if not payload then
		return false, "falha ao gerar JSON"
	end
	local ok, err = pcall(writefile, self._configFile, payload)
	if not ok then
		return false, err
	end
	if not quiet then
		self:Notify("Config", "Config salva: " .. self._configFile, 2.5, "Success")
	end
	self:Fire("ConfigSaved", self._configFile)
	return true
end

-- quiet: nao mostra notificacao de sucesso.
function HubLibrary:LoadConfig(fileName, silent, quiet)
	if fileName then
		self:SetConfigFile(fileName)
	end
	if not FileApiAvailable() then
		return false, "readfile/isfile indisponivel neste executor"
	end
	local okExists, exists = pcall(isfile, self._configFile)
	if not okExists or not exists then
		return false, "arquivo nao encontrado"
	end
	local ok, contents = pcall(readfile, self._configFile)
	if not ok then
		return false, contents
	end
	local restored = self:ImportConfig(contents, silent)
	if restored then
		if not quiet then
			self:Notify("Config", "Config carregada: " .. self._configFile, 2.5, "Success")
		end
		self:Fire("ConfigLoaded", self._configFile)
		return true
	end
	return false, "config invalida"
end

function HubLibrary:SetAutoSave(enabled, fileName)
	self._autoSave = enabled == true
	if fileName then
		self:SetConfigFile(fileName)
	end
	return self._autoSave
end

function HubLibrary:_AutoSaveTick()
	if not self._autoSave or self._destroyed or self._restoring then
		return
	end
	self._saveToken = (self._saveToken or 0) + 1
	local token = self._saveToken
	task.delay(1, function()
		if self._destroyed or token ~= self._saveToken then
			return
		end
		self:SaveConfig(nil, true)
	end)
end

-- Chame depois de criar TODOS os controles: carrega o config salvo
-- (se existir) e libera o hub. opcoes: AutoLoad (padrao true), AutoSave.
function HubLibrary:Ready(opcoes)
	if self._destroyed then
		return false
	end
	opcoes = type(opcoes) == "table" and opcoes or {}
	local loaded = false
	if opcoes.AutoLoad ~= false then
		local ok = self:LoadConfig(nil, nil, true)
		loaded = ok == true
		if loaded and opcoes.Notify ~= false then
			self:Notify("Config", "Config carregada: " .. self._configFile, 2.5, "Success")
		end
	end
	if opcoes.AutoSave ~= nil then
		self:SetAutoSave(opcoes.AutoSave)
	end
	self._ready = true
	self:Fire("Ready", loaded)
	return loaded
end

-- Registro aprimorado: dependencias, metadados e eventos por controle.
do
    local RegisterCore = Tab._Register

    function Tab:_Register(control, kind, title, opts)
        opts = opts or {}
        local result = RegisterCore(self, control, kind, title, opts)
        if not result then
            return result
        end

        result.DependsOn = opts.DependsOn or opts.Dependency
        result.DependsMode = opts.DependsMode or opts.DependencyMode or "all"
        result.Disabled = false
        if result.DependsOn then
            task.defer(function()
                if not self._destroyed then
                    self._hub:RefreshDependencies()
                end
            end)
        end
        result._changedListeners = {}

        function result:OnChanged(callback)
            if type(callback) ~= "function" then
                return nil
            end
            table.insert(self._changedListeners, callback)
            return {
                Disconnect = function()
                    local i = table.find(self._changedListeners, callback)
                    if i then
                        table.remove(self._changedListeners, i)
                    end
                end,
            }
        end

        function result:SetVisible(visible)
            if self.Object and self.Object.Parent then
                self.Object.Visible = visible ~= false
            end
            return visible ~= false
        end

        function result:IsVisible()
            return self.Object and self.Object.Visible == true
        end

        function result:SetDisabled(disabled)
            return self._hub:SetEnabled(self, disabled ~= true)
        end

        function result:IsDisabled()
            return self.Disabled == true
        end

        function result:FireChanged(value)
            for _, listener in ipairs(table.clone(self._changedListeners)) do
                SafeCall(listener, value, self)
            end
        end

        -- Faz o callback interno continuar existindo, mas tambem permite
        -- observers externos sem precisar alterar cada Add* individualmente.
        local oldSet = result.Set
        if type(oldSet) == "function" then
            result.Set = function(selfControl, value, silent)
                local before
                pcall(function()
                    before = selfControl:Get()
                end)
                local output = oldSet(selfControl, value, silent)
                if not silent then
                    local after
                    pcall(function()
                        after = selfControl:Get()
                    end)
                    if before ~= after then
                        selfControl:FireChanged(after)
                    end
                end
                return output
            end
        end

        return result
    end
end

end

    return json

function HubLibrary:ImportProfile(name, json, silent)
    name = tostring(name or "Default")
    local ok, snapshot = pcall(HttpService.JSONDecode, HttpService, tostring(json or ""))
    if not ok or type(snapshot) ~= "table" then
        return false, "perfil invalido"
    end
    self._profiles[name] = snapshot
    self._currentProfile = name
    if not silent then
        self:Restore(snapshot)
    end
    self:Fire("ProfileImported", name)
    return true
end

function HubLibrary:SaveProfileFile(name, fileName)
    name = tostring(name or "Default")
    if not FileApiAvailable() then
        return false, "API de arquivo indisponivel"
    end
    local json, err = self:ExportProfile(name)
    if not json then
        return false, err
    end
    fileName = tostring(fileName or ("FrankHub_Profile_" .. Slug(name) .. ".json"))
    local ok, writeErr = pcall(writefile, fileName, json)
    if not ok then
        return false, writeErr
    end
    return true
end

function HubLibrary:LoadProfileFile(name, fileName)
    name = tostring(name or "Default")
    if not FileApiAvailable() then
        return false, "API de arquivo indisponivel"
    end
    fileName = tostring(fileName or ("FrankHub_Profile_" .. Slug(name) .. ".json"))
    local okExists, exists = pcall(isfile, fileName)
    if not okExists or not exists then
        return false, "arquivo nao encontrado"
    end
    local okRead, json = pcall(readfile, fileName)
    if not okRead then
        return false, json
    end
    return self:ImportProfile(name, json)
end

function HubLibrary:CreateBackup(fileName)
    if not FileApiAvailable() then
        return false, "API de arquivo indisponivel"
    end
    fileName = tostring(fileName or (self._configFile .. ".bak"))
    local json = self:ExportConfig()
    if not json then
        return false, "falha ao exportar config"
    end
    local ok, err = pcall(writefile, fileName, json)
    if not ok then
        return false, err
    end
    return true
end

function HubLibrary:Diagnostics()
    local valid, issues = self:Validate()
    local info = {
        Version = self.Version,
        Theme = self._themeName,
        Tabs = #self.Abas,
        Controls = #self._registry,
        StateKeys = 0,
        Profiles = #self:GetProfiles(),
        AutoSave = self._autoSave == true,
        Ready = self._ready == true,
        Destroyed = self._destroyed == true,
        Valid = valid,
        Issues = issues,
    }
    for _ in pairs(self._state) do
        info.StateKeys = info.StateKeys + 1
    end
    return info
end

function HubLibrary:DebugReport(printReport)
    local d = self:Diagnostics()
    local report = string.format(
        "Frank Hub %s | Tabs=%d | Controls=%d | States=%d | Profiles=%d | Theme=%s | Valid=%s",
        tostring(d.Version), d.Tabs, d.Controls, d.StateKeys, d.Profiles,
        tostring(d.Theme), tostring(d.Valid)
    )
    if printReport ~= false then
        print("[Frank Hub] " .. report)
        if #d.Issues > 0 then
            warn("[Frank Hub] Issues:\n - " .. table.concat(d.Issues, "\n - "))
        end
    end
    return report, d
end

-- Atualiza dependencias automaticamente sempre que um State muda.
do
    local SetStateCore = HubLibrary.SetState
    function HubLibrary:SetState(key, value, fireEvent)
        local result = SetStateCore(self, key, value, fireEvent)
        if not self._restoring and self._dependencyRefreshLock ~= true then
            self._dependencyRefreshLock = true
            pcall(function()
                self:RefreshDependencies()
            end)
            self._dependencyRefreshLock = false
        end
        return result
    end
end

-- Dependencias: o controle pode depender de um ou varios StateKey/IDs.
function HubLibrary:_ResolveDependency(control)
    if not control or not control.DependsOn then
        return true
    end
    local deps = type(control.DependsOn) == "table" and control.DependsOn or { control.DependsOn }
    local results = {}
    for _, dep in ipairs(deps) do
        local depControl = self:FindItem(tostring(dep))
        local value
        if depControl and type(depControl.Get) == "function" then
            local ok, result = pcall(depControl.Get)
            if ok then value = result end
        else
            value = self:GetState(tostring(dep), nil)
        end
        table.insert(results, value == true or value ~= nil and value ~= false)
    end
    if control.DependsMode == "any" then
        for _, result in ipairs(results) do
            if result then return true end
        end
        return false
    end
    for _, result in ipairs(results) do
        if not result then return false end
    end
    return true
end

function HubLibrary:RefreshDependencies()
    for _, entry in ipairs(self._registry) do
        local control = entry.Control
        if control and control.DependsOn then
            self:SetEnabled(control, self:_ResolveDependency(control))
        end
    end
end

function HubLibrary:AddDependency(control, dependency, mode)
    if type(control) == "string" then
        control = self:FindItem(control)
    end
    if not control then return false end
    control.DependsOn = dependency
    control.DependsMode = mode or "all"
    self:RefreshDependencies()
    return true
end

-- Substitui SetEnabled por uma versao que realmente bloqueia entrada,
-- sem destruir o controle e sem interferir no config.
do
    local SetEnabledCore = HubLibrary.SetEnabled
    function HubLibrary:SetEnabled(control, enabled)
        if type(control) == "string" then
            control = self:FindItem(control)
        end
        local entry = type(control) == "table" and control._entry or nil
        if not entry or not entry.Object then
            return false
        end

        local active = enabled ~= false
        control.Disabled = not active
        entry.ForceHidden = false
        entry.Object.Visible = entry.Visible ~= false

        if control._disabledOverlay and control._disabledOverlay.Parent then
            control._disabledOverlay:Destroy()
            control._disabledOverlay = nil
        end

        if not active and entry.Object.Parent then
            local overlay = New("TextButton", {
                Name = "DisabledOverlay",
                Size = UDim2.fromScale(1, 1),
                BackgroundTransparency = 1,
                BorderSizePixel = 0,
                Text = "",
                AutoButtonColor = false,
                Active = true,
                ZIndex = 100,
            }, entry.Object)
            control._disabledOverlay = overlay
            self:_Conn(overlay.MouseButton1Click, function()
                self:Notify("Frank Hub", control.Title .. " esta desativado.", 1.5, "Info")
            end, control)
        end

        if entry.Object:IsA("GuiObject") then
            entry.Object.BackgroundTransparency = entry.Object.BackgroundTransparency
        end
        self:Fire("ControlEnabledChanged", control, active)
        return active
    end
end

-- Adiciona controles leves que nao existiam na v4.
function Tab:AddDivider(text)
    if self._destroyed then return nil end
    local hub = self._hub
    local frame = New("Frame", {
        Name = "Divider",
        Size = UDim2.new(1, 0, 0, text and 30 or 14),
        BackgroundTransparency = 1,
        BorderSizePixel = 0,
    }, self.Scroll)
    local line = New("Frame", {
        AnchorPoint = Vector2.new(0, 0.5),
        Position = UDim2.new(0, 8, 0.5, text and 7 or 0),
        Size = UDim2.new(1, -16, 0, 1),
        BorderSizePixel = 0,
    }, frame)
    hub:_T(line, "BackgroundColor3", "Border")
    if text then
        local label = New("TextLabel", {
            BackgroundTransparency = 1,
            Position = UDim2.fromOffset(12, 0),
            Size = UDim2.new(1, -24, 0, 20),
            Text = tostring(text),
            TextSize = 14,
            FontFace = FONT_BOLD,
            TextXAlignment = Enum.TextXAlignment.Left,
        }, frame)
        hub:_T(label, "TextColor3", "MutedText")
    end
    return self:_Register({ Object = frame }, "Divider", text or "divider", { Persist = false })
end

function Tab:AddBadge(text, kind)
    if self._destroyed then return nil end
    local hub = self._hub
    local card = self:_NewCard(34, "Badge")
    local label = New("TextLabel", {
        BackgroundTransparency = 1,
        Position = UDim2.fromOffset(10, 0),
        Size = UDim2.new(1, -20, 1, 0),
        Text = tostring(text or "Badge"),
        TextSize = 15,
        FontFace = FONT_BOLD,
        TextXAlignment = Enum.TextXAlignment.Left,
    }, card)
    local kindKey = KIND_KEY[tostring(kind or "Info")] or "Accent"
    hub:_T(label, "TextColor3", kindKey)
    local control = { Object = card, Label = label }
    control.SetText = function(_, value) label.Text = tostring(value or "") end
    control.Get = function() return label.Text end
    return self:_Register(control, "Badge", text or "Badge", { Persist = false })
end

-- Corrige o carregamento de configs antigas e adiciona migração simples.
do
    local RestoreCore = HubLibrary.Restore
    function HubLibrary:Restore(snapshot, silent)
        if type(snapshot) == "table" then
            local version = tonumber(snapshot.Version) or 1
            if version < 4 and snapshot.Values == nil and snapshot.State == nil then
                snapshot = { Version = 4, Values = snapshot }
            end
            snapshot.Version = 5
        end
        return RestoreCore(self, snapshot, silent)
    end
end
